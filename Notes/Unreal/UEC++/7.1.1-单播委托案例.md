# 案例描述
- Actor B中声明一个单播委托对象`PrintLocationDelegate`，一个调用委托函数`CallLocationActorPrint()`
- Actor A中有一个打印Actor位置函数
- 在Actor A的构造函数中创建一个根组件，用于查询Actor A的位置
- 在Actor A的`BeginPlay()`函数中利用`UGameplayStatics::GetActorOfClass()`获取Actor B的指针，并为该指针的单播委托对象绑定Actor A的打印函数用于打印Actor A的位置。
- 在Actor B的调用委托函数CallLocationActorPrint中执行委托函数`ExecuteIfBound()`,即可打印Actor A的位置

```cpp
/*位置类*/

	GENERATED_BODY()

	void PrintLocation();

	int32 Mana = 7;

{
	GENERATED_BODY()

	// Sets default values for this actor's properties

	/*实例化单播委托*/
	FSingleDelegetePrintLocation PrintLocationDelegate;//无参

	FSingleDelegeteGetLocation GetLocationDelegateWithPayLoad;//载荷数据

	UFUNCTION(BlueprintCallable, Category = "SingleDelegateTest")
	void CallLocationActorPrint();
	UFUNCTION(BlueprintCallable, Category = "SingleDelegateTest")
	void CallLocationActorGet();
	UFUNCTION(BlueprintCallable, Category = "SingleDelegateTest")
	void CallLocationActorGetWithPayLoad();
```

SingleDelegateActor.cpp
```cpp
#include "SingleDelegateActor.h"
#include "Kismet/GameplayStatics.h"

ALocationActor::ALocationActor()
{
	RootComponent = CreateDefaultSubobject<USceneComponent>(TEXT("LOcationRoot"));
}

void ALocationActor::BeginPlay()
{
	/* 在世界中查找指定类的第一个 Actor,返回的是Actor的指针 */
	AActor* ActorPtr = UGameplayStatics::GetActorOfClass(this, ASingleDelegateActor::StaticClass());

	//如果ActorPtr指针可以转换为SingleDelegateActorPtr,则单播委托绑定函数
	if (ASingleDelegateActor* SingleDelegateActorPtr = Cast<ASingleDelegateActor>(ActorPtr))
	{
		SingleDelegateActorPtr->PrintLocationDelegate.BindUObject(this, &ALocationActor::PrintLocation);
		SingleDelegateActorPtr->GetLocationDelegate.BindUObject(this, &ALocationActor::GetLocation);
		//载荷数据绑定
		int32 CurHealth = 100, CurMoney = 1200;
		SingleDelegateActorPtr->GetLocationDelegateWithPayLoad.BindUObject(this, &ALocationActor::GetLocationWithPayLoad, CurHealth, CurMoney);
	}
}
//委托调用函数
void ALocationActor::PrintLocation()
{
	FVector ThisActorLocaton = GetActorLocation();
	UE_LOG(LogTemp, Warning, TEXT("[%s] MyLocation: [%s]"), *FString(__FUNCTION__), *ThisActorLocaton.ToString());
}

FVector ALocationActor::GetLocation(FString InStr)
{
	FVector ThisActorLocaton = GetActorLocation();
	UE_LOG(LogTemp, Warning, TEXT("[%s] : [%s]"), *InStr, *ThisActorLocaton.ToString());
	return ThisActorLocaton;
}

FVector ALocationActor::GetLocationWithPayLoad(FString InStr, int32 Health, int32 Money)
{
	FVector ThisActorLocaton = GetActorLocation();
	//打印参数
	UE_LOG(LogTemp, Warning, TEXT("[%s] : [%s]"), *InStr, *ThisActorLocaton.ToString());
	//打印载荷数据
	UE_LOG(LogTemp, Warning, TEXT("[%s] : [%d]"), *InStr, Health);
	UE_LOG(LogTemp, Warning, TEXT("[%s] : [%d]"), *InStr, Money);
	//打印成员变量
	UE_LOG(LogTemp, Warning, TEXT("[%s] : [%d]"), *InStr, Mana);
	return ThisActorLocaton;
}

 	// Set this actor to call Tick() every frame.  You can turn this off to improve performance if you don't need it.

{
	Super::BeginPlay();
	
}

// Called every frame
void ASingleDelegateActor::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

}
//委托执行
void ASingleDelegateActor::CallLocationActorPrint()
{
	PrintLocationDelegate.ExecuteIfBound();
}

void ASingleDelegateActor::CallLocationActorGet()
{
	FVector ReceivedLocation = GetLocationDelegate.Execute(TEXT("SingleDelegateActor"));
}

void ASingleDelegateActor::CallLocationActorGetWithPayLoad()
{
	FVector ReceivedLocation = GetLocationDelegateWithPayLoad.Execute(TEXT("SingleDelegateActor"));
}
```
