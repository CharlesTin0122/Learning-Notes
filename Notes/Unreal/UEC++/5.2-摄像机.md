```cpp
#pragma once

/* 摄影机信息结构体 */
USTRUCT()
struct FToggleCameraInfo
{
	GENERATED_BODY()

	UPROPERTY(EditAnywhere, Category = "MyCamera")
	AActor* MyCamera;//摄像机指针
	float CameraBlendTime = 1.0f;//摄像机混合时间

};

UCLASS()
class QUICKSTART_API AMyCameraDirector : public AActor
{
	GENERATED_BODY()
	
public:	
	// Sets default values for this actor's properties
	AMyCameraDirector();

protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;

public:	
	// Called every frame
	virtual void Tick(float DeltaTime) override;

private:
	//摄影信息机结构体列表
	UPROPERTY(VisibleAnywhere, Category = "MyCamera")
	TArray<FToggleCameraInfo> CameraInfoList;
	//切换下个摄影界间隔时间
	float TimeToNextCameraChange = 0.0f;
	//摄像机索引
	int32 CameraIndex = -1;
};
```

MyCameraDirector.cpp
```cpp
// MyCameraDirector.cpp

#include "MyCameraDirector.h"
#include "Kismet/GameplayStatics.h"
#include "Camera/CameraActor.h"

// Sets default values
AMyCameraDirector::AMyCameraDirector()
{
    PrimaryActorTick.bCanEverTick = true;
}

// Called when the game starts or when spawned
void AMyCameraDirector::BeginPlay()
{
    Super::BeginPlay();

    TArray<AActor*> CameraList;//CameraActor数组
    /*获取世界中所有的CameraActor并将其填充CameraActor数组
    *参数1：世界上下文；参数2：要获取对象的类名；参数3：获取对象填充的数组*/
    UGameplayStatics::GetAllActorsOfClass(this, ACameraActor::StaticClass(), CameraList);

    CameraInfoList.Empty();//清空数组
    //检查摄影机数量是否大于1
    if (CameraList.Num() <= 1)
    {
        UE_LOG(LogTemp, Warning, TEXT("Not enough cameras found. At least 2 cameras are needed."));
        return;
    }
    // 遍历世界摄影机列表
    for (AActor* Cam : CameraList)
    {
        /* AddDefaulted_GetRef()在CameraInfoList数组中添加默认构造对象并返回对象的引用*/
        FToggleCameraInfo& CameraInfo = CameraInfoList.AddDefaulted_GetRef();
        //设置对象属性
        CameraInfo.MyCamera = Cam;
        CameraInfo.CameraBlendTime = 0.3f;
    }

    CameraIndex = 0;//从第一个摄影机开始
    TimeToNextCameraChange = 2.0f;  // 设置初始切换时间间隔
}

// Called every frame
void AMyCameraDirector::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);
    // 防止空数组访问
    if (CameraInfoList.Num() == 0) 
    {
        return;
    }
    //摄影机切换间隔
    const float TimeBetweenCameraChanges = 2.0f;

    // 每帧减少剩余的时间
    TimeToNextCameraChange -= DeltaTime;

    // 如果时间到了，就进行摄像机切换
    if (TimeToNextCameraChange <= 0.0f)
    {
        // 切换完后重置计时器为下次切换的间隔
        TimeToNextCameraChange = TimeBetweenCameraChanges;
        //获取本地玩家控制器
        APlayerController* PlayerController = UGameplayStatics::GetPlayerController(this, 0);
        //如果获取失败报错
        if (!PlayerController)
        {
            UE_LOG(LogTemp, Warning, TEXT("No PlayerController found."));
            return;
        }
        //获取摄像机信息属性
        AActor* CurrentCamera = CameraInfoList[CameraIndex].MyCamera;
        float CurrentBlendTime = CameraInfoList[CameraIndex].CameraBlendTime;
        //如果当前摄影机存在，切换摄影机
        if (CurrentCamera)
        {
            PlayerController->SetViewTargetWithBlend(CurrentCamera, CurrentBlendTime);// 平滑地混合到摄像机。
        }
        // 环形索引，确保循环切换
        CameraIndex = (CameraIndex + 1) % CameraInfoList.Num(); 
    }
}
```
