// Fill out your copyright notice in the Description page of Project Settings.


#include "FloatingActor.h"

// Sets default values
AFloatingActor::AFloatingActor()
{
 	//确保了该 Actor 会在每一帧中调用 Tick() 函数。如果不需要逐帧更新，可以将其设为 false 来提高性能。
	PrimaryActorTick.bCanEverTick = true;

	//创建静态模型组件子对象，命名为mesh
	VisualMesh = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("Mesh"));
	//将Mesh组建设置为根组件的子组件，根组件是 Actor 层级结构中的最上层组件，其他组件通常会附加到它上面。
	VisualMesh->SetupAttachment(RootComponent);

}

// BeginPlay() 是 Actor 生命周期的一部分，在游戏开始时或者 Actor 被生成时调用。
void AFloatingActor::BeginPlay()
{
	Super::BeginPlay();
	//如果立方体资产获取成功，则设置VisualMesh为立方体资产并设置相对位置为0,0,0
	if (NewMesh)
	{
		VisualMesh->SetStaticMesh(NewMesh);
		VisualMesh->SetRelativeLocation(FVector(0.0f, 0.0f, 0.0f));
	}
	
}

// 每帧调用的函数。DeltaTime 是自上次帧更新以来的时间间隔。通常用来进行逐帧的更新。
void AFloatingActor::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
	//使立方体在上下浮动的同时旋转
	FVector NewLocation = GetActorLocation();//获取Actor位置
	FRotator NewRotation = GetActorRotation();//获取Actor旋转
	float RunningTime = GetGameTimeSinceCreation();//获取游戏运行时间，Actor产生之后的秒数
	//计算浮动的高度变化。通过正弦函数 (FMath::Sin) 来产生一个基于时间的浮动效果。DeltaTime 用于保证浮动在每一帧都有细微变化。
	float DeltaHeight = (FMath::Sin(RunningTime + DeltaTime) - FMath::Sin(RunningTime));//正弦高度
	NewLocation.Z += DeltaHeight * FloatSpeed;//修改 Actor 的 Z 轴位置，通过将 DeltaHeight 乘以一个系数（20.0f），使其浮动的幅度变大。

	float DeltaRotation = DeltaTime * RotationSpeed;// 计算每一帧旋转的角度。每秒钟旋转 20 度
	NewRotation.Yaw += DeltaRotation;// 修改 Actor 的旋转角度，这里只改变 Yaw（绕 Z 轴的旋转）。

	SetActorLocationAndRotation(NewLocation, NewRotation);//设置Actor的位置和旋转

}
