#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Pawn.h"
#include "CollidingPawn.generated.h"

UCLASS()
class QUICKSTART_API ACollidingPawn : public APawn
{
	GENERATED_BODY()

public:
	// Sets default values for this pawn's properties
	ACollidingPawn();

protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;
	/*前向声明了一个指向 粒子系统组件 的指针
	* Unreal Engine 的代码风格推荐在头文件中尽可能使用前向声明，而不是 #include 具体的头文件。这样可以：
	* 减少编译时间：特别是在大型项目中，每次修改头文件都会触发大量的重编译。
	* 避免循环依赖：如果两个类互相引用，直接 #include 会导致循环依赖问题，而前向声明可以避免。*/
	UPROPERTY()
	class UParticleSystemComponent* OurParticleSystem;

	/*要使用自定义 Pawn移动组件，首先需将变量添加到 Pawn 类进行追踪,前向声明一个自定义的移动组件指针*/
	UPROPERTY()
	class UCollidingPawnMovementComponent* OurMovementComponent;

	//输入函数
	void MoveForward(float AxisValue);
	void MoveRight(float AxisValue);
	void Turn(float AxisValue);
	void ParticleToggle();

public:
	// Called every frame
	virtual void Tick(float DeltaTime) override;

	// Called to bind functionality to input
	virtual void SetupPlayerInputComponent(class UInputComponent* PlayerInputComponent) override;

	//用于提供引擎中其他类访问该Pawn当前所用Pawn移动组件的权限。
	virtual UPawnMovementComponent* GetMovementComponent() const override;



};

#include "CollidingPawn.h"
#include "UObject/ConstructorHelpers.h"
#include "Particles/ParticleSystemComponent.h"
#include "Components/SphereComponent.h"
#include "Camera/CameraComponent.h"
#include "GameFramework/SpringArmComponent.h"

#include "CollidingPawnMovementComponent.h"//自定义移动组件

// Sets default values
ACollidingPawn::ACollidingPawn()
{
 	// Set this pawn to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;

	/*创建组建*/
	/*USphereComponent：用于简单碰撞的球体。 边界在编辑器中呈现为线条。*/
	USphereComponent* SphereComponent = CreateDefaultSubobject<USphereComponent>(TEXT("RootComponent"));
	RootComponent = SphereComponent;//球组件设置为根组件
	SphereComponent->InitSphereRadius(40.0f);//球组件的初始化半径为40
	/** 设置碰撞配置文件名称
	* 构造函数在设置 ProfileName 时调用该函数
	* 这将把当前的 CollisionProfileName 改为 Pown，并覆盖碰撞设置。
	*/
	SphereComponent->SetCollisionProfileName(TEXT("Pawn"));//

	// 创建并放置网格体组件，以便查看球体位置
	UStaticMeshComponent* SphereVisual = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("VisualRepresentation"));
	//设置观察球为根组件的子组件
	SphereVisual->SetupAttachment(RootComponent);
	/*找到观察球的静态网格体位置,ConstructorHelpers 是一个命名空间，包含了一些辅助功能，用于在构造函数中进行资源初始化。
	* static关键字使 SphereVisualAsset 成为一个 静态局部变量。
	* 静态变量在第一次被访问时初始化，并且在整个程序生命周期中保持有效。
	* 在 Unreal Engine 中，静态加载资源时通常会用 static 来避免重复加载。
	*/
	static ConstructorHelpers::FObjectFinder<UStaticMesh>SphereVisualAsset(TEXT("/Game/StarterContent/Shapes/Shape_Sphere.Shape_Sphere"));
	//如果静态网格体可用，设置
	if (SphereVisualAsset.Succeeded())
	{
		SphereVisual->SetStaticMesh(SphereVisualAsset.Object);
		SphereVisual->SetRelativeLocation(FVector(0.0f, 0.0f, -40.0f));
		SphereVisual->SetWorldScale3D(FVector(0.8f));
	}

	/* 创建可激活或停止的粒子系统*/
	OurParticleSystem = CreateDefaultSubobject<UParticleSystemComponent>(TEXT("MovementParticles"));//创建粒子组件
	OurParticleSystem->SetupAttachment(SphereVisual);//粒子组件设置为观察球的子组件
	OurParticleSystem->bAutoActivate = false;//关闭自动激活
	OurParticleSystem->SetRelativeLocation(FVector(-20.0f, 0.0f, 20.0f));//设置粒子组件位置
	static ConstructorHelpers::FObjectFinder<UParticleSystem> ParticleAsset(TEXT("/Game/StarterContent/Particles/P_Fire.P_Fire"));//获取粒子资产
	if (ParticleAsset.Succeeded())//如果粒子资产获取成功，则更改此粒子系统组件使用的粒子系统为ParticleAsset
	{
		OurParticleSystem->SetTemplate(ParticleAsset.Object);
	}

	// 使用弹簧臂给予摄像机平滑自然的运动感。
	USpringArmComponent* SpringArm = CreateDefaultSubobject<USpringArmComponent>(TEXT("CameraAttachmentArm"));
	SpringArm->SetupAttachment(RootComponent);
	SpringArm->SetRelativeRotation(FRotator(-45.0f, 0.0f, 0.0f));
	SpringArm->TargetArmLength = 400.0f;
	SpringArm->bEnableCameraLag = true;
	SpringArm->CameraLagSpeed = 3.0f;
	// 创建摄像机并附加到弹簧臂
	UCameraComponent* Camera = CreateDefaultSubobject<UCameraComponent>(TEXT("ActualCamera"));
	/*摄像机绑定到弹簧臂，绑定位置为USpringArmComponent::SocketName = SpringEndpoint */
	Camera->SetupAttachment(SpringArm, USpringArmComponent::SocketName);

	// 控制默认玩家
	AutoPossessPlayer = EAutoReceiveInput::Player0;

	// 创建移动组件的实例，并要求其更新根。
	OurMovementComponent = CreateDefaultSubobject<UCollidingPawnMovementComponent>(TEXT("CustomMovementComponent"));
	OurMovementComponent->UpdatedComponent = RootComponent;

}

// Called when the game starts or when spawned
void ACollidingPawn::BeginPlay()
{
	Super::BeginPlay();
	
}

// Called every frame
void ACollidingPawn::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

}

// Called to bind functionality to input
void ACollidingPawn::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);

	PlayerInputComponent->BindAction("ParticleToggle", IE_Pressed, this, &ACollidingPawn::ParticleToggle);

	PlayerInputComponent->BindAxis("MoveForward", this, &ACollidingPawn::MoveForward);
	PlayerInputComponent->BindAxis("MoveRight", this, &ACollidingPawn::MoveRight);
	PlayerInputComponent->BindAxis("Turn", this, &ACollidingPawn::Turn);

}

UPawnMovementComponent* ACollidingPawn::GetMovementComponent() const
{
	return OurMovementComponent;
}

/*输入行为函数*/
void ACollidingPawn::MoveForward(float AxisValue)
{
	if (OurMovementComponent && OurMovementComponent->UpdatedComponent == RootComponent)
	{
		/**
		* AddInputVector() 是一个用于添加输入方向的方法，属于 UCharacterMovementComponent 或 UPawnMovementComponent 类的一部分。
		* 它的主要功能是将一个输入向量加入到当前帧的移动输入中，以实现角色或物体的累积移动。
		*/
		OurMovementComponent->AddInputVector(GetActorForwardVector() * AxisValue);
	}
}

void ACollidingPawn::MoveRight(float AxisValue)
{
	if (OurMovementComponent && OurMovementComponent->UpdatedComponent == RootComponent)
	{
		OurMovementComponent->AddInputVector(GetActorRightVector() * AxisValue);
	}
}

void ACollidingPawn::Turn(float AxisValue)
{
	FRotator NewRotation = GetActorRotation();
	NewRotation.Yaw += AxisValue;
	SetActorRotation(NewRotation);
}

void ACollidingPawn::ParticleToggle()
{
	if (OurParticleSystem && OurParticleSystem->Template)
	{
		OurParticleSystem->ToggleActive();
	}
}

// Fill out your copyright notice in the Description page of Project Settings.


#include "CollidingPawnMovementComponent.h"


/**重写TickComponent方法，
* 在Pawn上每帧调用的函数。重写这个函数以实现每帧执行的自定义逻辑。只有在组件注册时才会执行，并且PrimaryComponentTick.bCanEverTick必须设置为true。
* 参数1： DeltaTime - 表示当前帧与上一帧之间的时间间隔（以秒为单位）。你可以用它来确保逻辑以时间为基础（如移动、旋转）而不是帧数。
* 参数2：TickType - 表示当前 Tick 的类型，可能的值包括：LEVELTICK_TimeOnly：标准游戏时间更新。LEVELTICK_ViewportsOnly：仅当编辑器视口更新时触发（非游戏逻辑）。LEVELTICK_All：针对所有 Tick 类型。
* 参数3：ThisTickFunction - 结构体封装了与 Actor 组件 Tick 相关的各种属性和操作。包含有关当前组件 Tick 的额外数据，例如是否应该跳过或延迟执行等。
*/
void UCollidingPawnMovementComponent::TickComponent(float DeltaTime, ELevelTick TickType, FActorComponentTickFunction* ThisTickFunction)
{
	Super::TickComponent(DeltaTime, TickType, ThisTickFunction);

	/**确保所有事物持续有效，以便进行移动
	* PawnOwner：拥有这个组件的Pawn,是一个指向Pawn的指针
	* UpdatedComponent：我们移动和更新的组件。如果在启动时这个值为null，并且bAutoRegisterUpdatedComponent为true，拥有者Actor的根组件将在启动时自动设置为我们的UpdatedComponent。
	* ShouldSkipUpdate：如果移动的组件未渲染或无法移动，则可能跳过更新。如果应跳过组件移动更新，则返回 true
	*/
	if (!PawnOwner || !UpdatedComponent || ShouldSkipUpdate(DeltaTime))
	{
		return;
	}


	/**获取（然后清除）ACollidingPawn::Tick中设置的移动向量
	* ConsumeInputVector()：返回待处理的输入向量并将其重置为零。
	* 此操作应在移动更新期间（由Pawn或PawnMovementComponent执行）使用，以防止在帧之间累积控制输入。
	* 将待处理的输入向量复制到保存的输入向量（GetLastMovementInputVector()）。返回待处理的输入向量。
	* GetClampedToMaxSize()：创建这个向量的副本，最大二维幅度限制为1。Z保持不变。
	*/
	FVector DesiredMovementThisFrame = ConsumeInputVector().GetClampedToMaxSize(1.0f) * DeltaTime * 150.0f;
	//如果预期移动趋近于0，执行碰撞
	if (!DesiredMovementThisFrame.IsNearlyZero())
	{
		/*FHitResult包含有关撞击信息的结构体，例如撞击点和该点的表面法线。*/
		FHitResult Hit;
		/** SafeMoveUpdatedComponent()是一个与物理和移动逻辑密切相关的重要函数，是 UMovementComponent的一部分，主要用途是安全地移动组件，同时避免穿透障碍物或导致不期望的物理行为。
		* 参数1： 描述组件在本次移动中的位移向量
		* 参数2： 目标旋转（以四元数表示）。如果只移动位置而不改变旋转，可以传递当前旋转。
		* 参数3： 指定是否启用碰撞检测（Sweep）。当设置为 true 时，组件会在移动路径中检查是否有障碍物，并在检测到碰撞时停止或调整移动。
		* 参数4：一个可选的 FHitResult 指针。如果组件在移动过程中发生碰撞，OutHit 将填充碰撞的详细信息（如位置、法线、命中对象等）。
		*/
		SafeMoveUpdatedComponent(DesiredMovementThisFrame, UpdatedComponent->GetComponentRotation(), true, Hit);

		// 若发生碰撞，尝试滑过去
		/*IsValidBlockingHit()：如果发生了撞击，但不是由于在穿透中启动造成的，则返回 true*/
		if (Hit.IsValidBlockingHit())
		{
			/**SlideAlongSurface() 是一个与物体的移动和碰撞处理相关的重要方法。它被广泛应用于角色、移动组件等需要复杂碰撞处理的场景，主要功能是让对象沿着碰撞表面滑动，而不是直接停止或反弹。
			* 参数1：示物体希望移动的向量（位移向量）。
			* 参数2：碰撞检测的比例（0.0 到 1.0），表示滑动发生的时间点。例如，0.5 表示在路径中途发生的滑动。
			* 参数3：表示与物体发生碰撞的表面法线。
			* 参数4：表示碰撞检测的详细信息，使用一个 FHitResult 来存储
			* 返回：返回计算后的修正向量，即物体在沿表面滑动后的实际移动方向。
			* Hit.Time:如果有撞击，沿跟踪方向的撞击 "时间"（范围从 0.0 到 1.0），表示从跟踪开始到跟踪结束之间的时间。
			*/
			SlideAlongSurface(DesiredMovementThisFrame, 1.0f - Hit.Time, Hit.Normal, Hit);//
		}
	}
}
