# 概述
- 该文件主要用于封装拾取组件，继承USphereComponent。包含一个OnPickUp委托事件，一个OnSphereBeginOverlap回调函数。
- 在BeginPlay中给OnComponentBeginOverlap绑定函数OnSphereBeginOverlap
- 在OnSphereBeginOverlap函数中广播OnPickUp委托
- OnPickUp委托在蓝图中实现
# 代码
MyPickUpComponent.h
```cpp
// Fill out your copyright notice in the Description page of Project Settings.

#pragma once

#include "CoreMinimal.h"
#include "Components/SphereComponent.h"
#include "MyCharacter.h"
#include "MyPickUpComponent.generated.h"

//声明碰撞后触发的委托
DECLARE_DYNAMIC_MULTICAST_DELEGATE_OneParam(FOnPickUp, AMyCharacter*, PickUpCharacter);

/**武器拾取组件类
 * Blueprintable：允许该类可以被用作蓝图的基类。意味着你可以在编辑器中基于该类创建蓝图派生类。
 * BlueprintType：允许该类在蓝图中被用作变量、参数或返回值。意味着你可以在蓝图中声明这种类型的变量，或者将其作为函数参数类型。
 * ClassGroup = (Custom):组件会分组到 "Custom" 类别中。
 * meta = (BlueprintSpawnableComponent):允许该组件可以直接从蓝图中被添加到 Actor 中。
 */
UCLASS(Blueprintable, BlueprintType, ClassGroup = (Custom), meta = (BlueprintSpawnableComponent))
class FPCSAMPLE_API UMyPickUpComponent : public USphereComponent
{
	GENERATED_BODY()

public:
	UMyPickUpComponent();

protected:

	virtual void BeginPlay() override;

	/**事件回调函数，当其他的 Actor 或组件与一个球形碰撞体（Sphere Component）发生重叠时会被触发。
	* OnComponentBeginOverlap.AddDynamic()绑定该函数，必须为UFUNCTION(),否则无法调用。
	* OnComponentBeginOverlap是一个FComponentBeginOverlapSignature结构体对象，FComponentBeginOverlapSignature是一个DECLARE_DYNAMIC_MULTICAST_SPARSE_DELEGATE_SixParams宏
	*/
	UFUNCTION()
	void OnSphereBeginOverlap(
		UPrimitiveComponent* OverlapComponent, //发生重叠的球形碰撞体组件
		AActor* OtherActor, //与球形碰撞体发生重叠的另一个 Actor
		UPrimitiveComponent* OtherComp, //与当前球形碰撞体发生重叠的另一个组件
		int32 OtherBodyIndex, //发生重叠的对方物理体的索引值，索引可以用来区分具体的物理体。
		bool bFromSweep, //此次重叠是否是通过 “Sweep” 操作检测到的
		const FHitResult& SweepResult //如果 bFromSweep 为 true，此参数包含详细的碰撞信息。包括碰撞位置、法线、冲击点等数据。
	);

public:
	//定义委托
	UPROPERTY(BlueprintAssignable, Category = "Interaction")
	FOnPickUp OnPickUp;
	
};

```

MyPickUpComponent.cpp
```cpp
// Fill out your copyright notice in the Description page of Project Settings.


#include "MyPickUpComponent.h"

UMyPickUpComponent::UMyPickUpComponent()
{
	//设置碰撞球半径
	SphereRadius = 32.f;
}

void UMyPickUpComponent::BeginPlay()
{
	Super::BeginPlay();
	//注册碰撞触发事件,当重叠发生时，此对象调用OnSphereBeginOverlap函数。
	OnComponentBeginOverlap.AddDynamic(this, &UMyPickUpComponent::OnSphereBeginOverlap);
}
//当重叠发生时调用函数实现
void UMyPickUpComponent::OnSphereBeginOverlap(
	UPrimitiveComponent* OverlapComponent, 
	AActor* OtherActor, 
	UPrimitiveComponent* OtherComp, 
	int32 OtherBodyIndex, 
	bool bFromSweep, 
	const FHitResult& SweepResult
)
{
	//重叠发生时的对方Actor转化为MyCharacter
	AMyCharacter* Character = Cast<AMyCharacter>(OtherActor);
	//如果Character不为空，则OnPickUp动态多播委托广播
	if (Character)
	{
		OnPickUp.Broadcast(Character);
	}
	//功能完成后移除重叠事件，避免重复触发。
	OnComponentBeginOverlap.RemoveAll(this);
}
```
