```cpp
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Pawn.h"
#include "PawnWithCamera.generated.h"

UCLASS()
class QUICKSTART_API APawnWithCamera : public APawn
{
	GENERATED_BODY()

public:
	// Sets default values for this pawn's properties
	APawnWithCamera();

protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;

public:
	// Called every frame
	virtual void Tick(float DeltaTime) override;

	// 设置玩家输入组件，参数是一个玩家输入组件对象
	virtual void SetupPlayerInputComponent(class UInputComponent* PlayerInputComponent) override;

protected:
	// 输入函数
	void MoveForward(float AxisValue);
	void MoveRight(float AxisValue);
	void PitchCamera(float AxisValue);
	void YawCamera(float AxisValue);
	void ZoomIn();
	void ZoomOut();
protected:
	//前向声明类指针变量
	UPROPERTY(EditAnywhere)
	class USpringArmComponent* SpringArmComp;

	UPROPERTY(EditAnywhere)
	class UCameraComponent* CameraComp;

	UPROPERTY(EditAnywhere)
	class UStaticMeshComponent* StaticMeshComp;

	//输入变量
	FVector2D MovementInput;//移动输入
	FVector2D CameraInput;//转向输入
	float ZoomFactor;//推拉因子
	bool bZoomingIn;//是否进行推拉
};
```

PawnWithCamera.cpp
```cpp

#include "PawnWithCamera.h"
#include "GameFramework/SpringArmComponent.h"
#include "Camera/CameraComponent.h"
#include "Components/StaticMeshComponent.h"



// Sets default values
APawnWithCamera::APawnWithCamera()
{
	// Set this pawn to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;

	//创建组件
	RootComponent = CreateDefaultSubobject<USceneComponent>(TEXT("RootSceneComponent"));
	StaticMeshComp = CreateDefaultSubobject <UStaticMeshComponent>(TEXT("StaticMeshComponent"));
	SpringArmComp = CreateDefaultSubobject<USpringArmComponent>(TEXT("SpringArmComponent"));
	CameraComp = CreateDefaultSubobject<UCameraComponent>(TEXT("CameraComponent"));

	//绑定组件
	StaticMeshComp->SetupAttachment(RootComponent);
	SpringArmComp->SetupAttachment(StaticMeshComp);
	CameraComp->SetupAttachment(SpringArmComp, USpringArmComponent::SocketName);

	//为SpringArm类的变量赋值。
	SpringArmComp->SetRelativeLocationAndRotation(FVector(0.0f, 0.0f, 50.0f), FRotator(-60.0f, 0.0f, 0.0f));
	SpringArmComp->TargetArmLength = 400.f;
	SpringArmComp->bEnableCameraLag = true;
	SpringArmComp->CameraLagSpeed = 3.0f;

	//控制默认玩家:自动拥有玩家变量 自动接收 Player0 的输入
	AutoPossessPlayer = EAutoReceiveInput::Player0;

}

// Called when the game starts or when spawned
void APawnWithCamera::BeginPlay()
{
	Super::BeginPlay();

}

// Called every frame
void APawnWithCamera::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

	/*处理动作映射*/
	//如果按下了鼠标右键按钮则推进，否则就拉远
	if (bZoomingIn)
	{
		//如果按下了鼠标右键推拉值不断加大，在按下鼠标右键（触发缩放）后，ZoomFactor 需要大约 0.5 秒才能从 0.0f 增加到 1.0f。
		ZoomFactor += DeltaTime / 0.5f;
	}
	else 
	{
		//在松开鼠标右键后，ZoomFactor 需要大约 0.25 秒才能从 1.0f 减小到 0.0f。
		ZoomFactor -= DeltaTime / 0.25f;//如果松开鼠标右键，推拉值不断减小
	}
	//将推拉值值限制在0-1
	ZoomFactor = FMath::Clamp<float>(ZoomFactor, 0.0f, 1.0f);
	//根据ZoomFactor来设置摄像机的视场和弹簧臂的长度，Lerp线性插值（A + Alpha * (B - A)），Alpha范围是0-1
	CameraComp->FieldOfView = FMath::Lerp<float>(90.0f, 60.0f, ZoomFactor);//对摄影机FOV进行90-60的线性插值
	SpringArmComp->TargetArmLength = FMath::Lerp<float>(400.0f, 300.0f, ZoomFactor);//对弹簧臂长度进行400-300的线性插值

	/*处理轴向映射*/
	//旋转Actor的偏转角度，这样摄像机也能旋转，因为摄像机与Actor相互绑定
	{
		FRotator NewRotation = GetActorRotation();//获取Actor的旋转值
		NewRotation.Yaw += CameraInput.X;//Actor的旋转值加上输入旋转值就得到了新的Actor旋转值
		SetActorRotation(NewRotation);//用新旋转值赋值Actor旋转
	}
	//旋转摄像机的俯仰角度，但要对其进行限制，这样我们就能始终俯视Actor
	{
		FRotator NewRotation = SpringArmComp->GetComponentRotation();//获取弹簧臂的旋转值
		NewRotation.Pitch = FMath::Clamp<float>(NewRotation.Pitch + CameraInput.Y, -80.0f, -15.0f);//对这个旋转值加上输入旋转值并限制，就得到了新的弹簧臂的旋转值
		SpringArmComp->SetWorldRotation(NewRotation);//新的弹簧臂的旋转值赋值弹簧臂旋转
	}
	// 基于"MoveX"和 "MoveY"坐标轴来处理移动
	{
		if (!MovementInput.IsZero())
		{
			// GetSafeNormal获取向量的标准化副本，检查根据长度进行标准化是否安全。如果向量长度太小而无法安全标准化，则返回零向量。
			MovementInput = MovementInput.GetSafeNormal() * 100.0f; //把移动轴的输入数值放大100倍
			FVector NewLocation = GetActorLocation();//获取Actor位置
			/*借助 GetActorForwardVector 和 GetActorRightVector，我们可以朝着Actor的朝向移动。由于摄像机朝的向和Actor一样，这使得我们按下"前进键"后，始终能朝着玩家所看的方向移动。
			* Actor向前向量（方向单位向量） * 键盘wasd键输入值（100倍） * 每帧时间
			*/
			NewLocation += GetActorForwardVector() * MovementInput.X * DeltaTime;
			NewLocation += GetActorRightVector() * MovementInput.Y * DeltaTime;
			SetActorLocation(NewLocation);//设置Actor位置
		}
	}

}

// 设置玩家输入组件，参数是一个玩家输入组件对象
void APawnWithCamera::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);

	/*绑定ZoomIn事件
	*InputComponent是AActor的一个UInputComponent指针
	*BindAction是UInputComponent的一个方法，将委托函数绑定到项目设置中定义的 Action。返回的引用只保证在绑定另一个动作之前有效。
	* 参数1：动作名称；参数2：按键事件枚举；参数3：绑定对象指针；参数4：绑定函数。
	*/
	InputComponent->BindAction("ZoomIn", IE_Pressed, this, &APawnWithCamera::ZoomIn);
	InputComponent->BindAction("ZoomIn", IE_Released, this, &APawnWithCamera::ZoomOut);
	/* 轴向映射
	* 将委托函数与项目设置中定义的轴绑定。返回的引用只保证在绑定另一个轴之前有效。
	* 参数1：编辑器中input设置的动作名称；参数2：绑定对象指针；参数3：绑定函数。
	*/
	InputComponent->BindAxis("MoveForward", this, &APawnWithCamera::MoveForward);
	InputComponent->BindAxis("MoveRight", this, &APawnWithCamera::MoveRight);
	InputComponent->BindAxis("CameraPitch", this, &APawnWithCamera::PitchCamera);
	InputComponent->BindAxis("CameraYaw", this, &APawnWithCamera::YawCamera);
}

//前后移动
void APawnWithCamera::MoveForward(float AxisValue)
{
	MovementInput.X = FMath::Clamp<float>(AxisValue, -1.0f, 1.0f);//限制为-1~1
}
//左右移动
void APawnWithCamera::MoveRight(float AxisValue)
{
	MovementInput.Y = FMath::Clamp<float>(AxisValue, -1.0f, 1.0f); // 限制为-1~1

}
//Pitch转向
void APawnWithCamera::PitchCamera(float AxisValue)
{
	CameraInput.Y = AxisValue;
}
//Yaw转向
void APawnWithCamera::YawCamera(float AxisValue)
{
	CameraInput.X = AxisValue;

}
//推拉操作
void APawnWithCamera::ZoomIn()
{
	bZoomingIn = true;
}
void APawnWithCamera::ZoomOut()
{
	bZoomingIn = false;
}
```
