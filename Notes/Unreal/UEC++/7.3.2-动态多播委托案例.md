在 **Unreal Engine** 中，**Dynamic Multicast Delegate** 是一种强大的事件机制，它允许对象广播事件并让多个监听器订阅和响应这些事件。与普通的 C++ 函数回调相比，动态多播委托具有更灵活的绑定机制，并且可以在运行时动态添加或移除监听器。

DynamicMultiDelegateActor.h
```cpp


#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Actor.h"
#include "DynamicMultiDelegateActor.generated.h"

//声明动态多播
DECLARE_DYNAMIC_MULTICAST_DELEGATE_ThreeParams(FMyDynamicDelegate, FString, InName, int32, InHelth, int32, InMana);


UCLASS()
class QUICKSTART_API ADynamicMultiDelegateActor : public AActor
{
	GENERATED_BODY()
	
public:	
	// Sets default values for this actor's properties
	ADynamicMultiDelegateActor();

protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;

public:	
	// Called every frame
	virtual void Tick(float DeltaTime) override;

public:
	/** 委托实例属性
	* BlueprintAssignable：仅限多播委托使用，该属性可以在蓝图中进行绑定和分配 
	*/
	UPROPERTY(BlueprintAssignable)
	FMyDynamicDelegate MyDelegate;

	/*初始化委托函数*/
	void InitDelegate(FMyDynamicDelegate InDelegate);

	/*执行委托函数*/
	UFUNCTION(BlueprintCallable)
	void ExcuteDelegate(FString InName, int32 InHelth, int32 InMana);

	/*解绑委托函数*/
	UFUNCTION(BlueprintCallable)
	void ReleaseDelegate();

	/*要绑定的函数*/
	UFUNCTION(BlueprintCallable)
	void Work(FString InName, int32 InHelth, int32 InMana);
	
};

```
DynamicMultiDelegateActor.cpp
```cpp



#include "DynamicMultiDelegateActor.h"


// Sets default values
ADynamicMultiDelegateActor::ADynamicMultiDelegateActor()
{
 	// Set this actor to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;

}

// Called when the game starts or when spawned
void ADynamicMultiDelegateActor::BeginPlay()
{
	Super::BeginPlay();
	/*将 UObject 实例和 UFUNCTION 成员绑定到动态多播委托*/
	MyDelegate.AddDynamic(this, &ADynamicMultiDelegateActor::Work);
	
}

// Called every frame
void ADynamicMultiDelegateActor::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

}

void ADynamicMultiDelegateActor::InitDelegate(FMyDynamicDelegate InDelegate)
{
	MyDelegate = InDelegate;
}

void ADynamicMultiDelegateActor::ExcuteDelegate(FString InName, int32 InHelth, int32 InMana)
{
	MyDelegate.Broadcast(InName, InHelth, InMana);
}

void ADynamicMultiDelegateActor::ReleaseDelegate()
{
	MyDelegate.Clear();
}

void ADynamicMultiDelegateActor::Work(FString InName, int32 InHelth, int32 InMana)
{
	UE_LOG(LogTemp, Warning, TEXT("Work!"));
}
```
