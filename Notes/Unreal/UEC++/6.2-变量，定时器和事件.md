#include "Countdown.h"


// Sets default values
ACountdown::ACountdown()
{
	//将此Actor设为逐帧调用Tick()。如无需此功能，可关闭以提高性能。
	PrimaryActorTick.bCanEverTick = false;//关闭Tick

	CountdownText = CreateDefaultSubobject<UTextRenderComponent>(TEXT("CountdownNumber"));//创建文本渲染组件
	CountdownText->SetHorizontalAlignment(EHTA_Center);//设置文本渲染组件位置为中央（EHorizTextAligment：Center）
	CountdownText->SetWorldSize(150.0f);//改变文字的世界大小
	RootComponent = CountdownText;//设置根组件

	CountdownTime = 3;//倒数时间为3s
}

// Called when the game starts or when spawned
void ACountdown::BeginPlay()
{
	Super::BeginPlay();

	UpdateTimerDispaly();//更新时间显示
	/**设置一个定时器，在设定的时间间隔内调用给定的本地函数
	 * 参数1：定时器句柄
	 * 参数2：定时器对象
	 * 参数3：调用函数		
	 * 参数4：调用时间间隔
	 * 参数5：是否循环
	*/
	GetWorldTimerManager().SetTimer(CountdownTimerHandle, this, &ACountdown::AdvanceTimer, 1.0f, true);
	
}

// Called every frame
void ACountdown::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

}

void ACountdown::UpdateTimerDispaly()
{
	/**设置文本
	* FMath::Max()返回参数中更大的值
	* FString::FromInt()将整数转换为字符串
	*/
	CountdownText->SetText(FText::FromString(FString::FromInt(FMath::Max(CountdownTime, 0))));
}
//时间倒数并显示
void ACountdown::AdvanceTimer()
{
	--CountdownTime;//时间倒数
	UpdateTimerDispaly();//时间显示
	//倒数完成，停止运行定时器并执行倒数完成函数
	if (CountdownTime < 1)
	{
		GetWorldTimerManager().ClearTimer(CountdownTimerHandle);
		CountdownHasFinished();
	}
}
//倒数完成函数
void ACountdown::CountdownHasFinished_Implementation()
{
	CountdownText->SetText(FText::FromString(TEXT("GO!")));
}
