#pragma once

#include "CoreMinimal.h"
#include "HAL/Runnable.h"
#include <atomic>

/** 线程任务类
 * 
 */
class FMySimpleRunnable : public FRunnable
{
public:
	FMySimpleRunnable(FString InThreadName) : MyTaskName(InThreadName)
	{}
	~FMySimpleRunnable() {}

	virtual bool Init() override;
	virtual uint32 Run() override;
	virtual void Exit() override;
	virtual void Stop() override;

	//获取线程任务名称
	FString GetThreadName();

protected:
	FString MyTaskName;//线程任务名称
	std::atomic<bool> bRunning = true;//线程是否在运行,原子类型布尔

	/** FCriticalSection 是 Unreal Engine 中的一种同步原语，用于在多线程环境中管理对共享资源的访问。它确保在任意时刻只有一个线程可以访问特定的代码段，从而防止竞争条件并确保数据完整性。
	* 用途：FCriticalSection 通常用于保护可以被多个线程访问的共享数据。
•	* 锁定和解锁：当一个线程进入临界区时，它会锁定 FCriticalSection 对象。其他尝试进入相同临界区的线程将被阻塞，直到第一个线程退出并解锁 FCriticalSection。
•	* 基于作用域的锁定：FScopeLock 类通常与 FCriticalSection 一起使用，以确保即使发生异常，临界区也能正确锁定和解锁。
	*/
	FCriticalSection MyCriticalSection;


	//打印信息
	void PrintThreadInfo();
};

#include "MySimpleRunnable.h"
#include "MyThreadSubsystem.h"//用于调用异步打印函数PrintInfo
#include "HAL/ThreadManager.h"//用于获取线程名称FThreadManager::Get().GetThreadName()

//初始化任务，新线程执行
bool FMySimpleRunnable::Init()
{
	bRunning = true;
	UMyThreadSubsystem::PrintInfo(TEXT("FMySimpleRunnable Is Init!"));
	PrintThreadInfo();
	return true;
}
//运行任务，新线程执行
uint32 FMySimpleRunnable::Run()
{

	PrintThreadInfo();
	//每隔0.04秒打印一次
	while (bRunning)
	{
		//睡眠0.04秒后打印
		FPlatformProcess::Sleep(0.04);
		/** 实现了一个作用域锁。
		* 这是一个处理作用域级别锁定的工具类。它非常有用
		* 以防止由于捕获异常而导致的死锁，并了解给定线程在资源上拥有的锁的数量
		*/
		FScopeLock Lock(&MyCriticalSection);
		if (!bRunning)
		{
			break;
		}
		UMyThreadSubsystem::PrintInfo(TEXT("FMySimpleRunnable Is Working"));
	}
	return 0;
}
//退出任务，新线程执行
void FMySimpleRunnable::Exit()
{
	UMyThreadSubsystem::PrintInfo(TEXT("FMySimpleRunnable Is Exit"));
	PrintThreadInfo();
}
//停止任务，外部调用方线程执行
void FMySimpleRunnable::Stop()
{
	FScopeLock Lock(&MyCriticalSection);
	bRunning = false;
	UMyThreadSubsystem::PrintInfo(TEXT("FMySimpleRunnable Is Stop"));
	PrintThreadInfo();

}

FString FMySimpleRunnable::GetThreadName()
{
	return MyTaskName;
}

void FMySimpleRunnable::PrintThreadInfo()
{
	uint32 CurrentID = FPlatformTLS::GetCurrentThreadId();//获取线程ID
	FString CurrentThreadName = FThreadManager::Get().GetThreadName(CurrentID);//获取线程名称
	// 以类似于经典 sprintf 的方式构造字符串
	FString MyInfo = FString::Printf(TEXT("CurrentID:[%d], CurrentThreadName:[%s]"), CurrentID, *CurrentThreadName);
	//打印信息
	UMyThreadSubsystem::PrintInfo(MyInfo);
}

#include "MyThreadSubsystem.h"


//是否创建子系统
bool UMyThreadSubsystem::ShouldCreateSubsystem(UObject* Outer) const
{
	return true;
}
//初始化子系统，子系统初始化之后创建线程
void UMyThreadSubsystem::Initialize(FSubsystemCollectionBase& Collection)
{
	Super::Initialize(Collection);
}
//停用子系统，同时释放线程
void UMyThreadSubsystem::Deinitialize()
{
	Super::Deinitialize();
	ReleaseMySimpleThread();//释放线程
}
//初始化线程
void UMyThreadSubsystem::InitMySimpleThread()
{
	//线程任务共享指针
	MySimpleTask = MakeShared<FMySimpleRunnable>(TEXT("MyTask01"));
	/** 工厂方法，用于创建具有指定堆栈大小和线程优先级的线程。
	* @param InRunnable 要执行的可运行对象
	* @param ThreadName 线程名称
	* @param InStackSize 要创建的堆栈大小。0表示使用当前线程的堆栈大小
	* @param InThreadPri 告诉线程是否需要调整其优先级。默认为正常优先级
	* @return 新创建的线程，如果失败则返回nullptr
	*/
	MyThreadPtr = FRunnableThread::Create(MySimpleTask.Get(), *(MySimpleTask->GetThreadName()));
}
//释放线程
void UMyThreadSubsystem::ReleaseMySimpleThread()
{
	//如果线程任务可用，则停止
	if (MySimpleTask.IsValid())
	{
		MySimpleTask->Stop();
	}
	//如果线程指针可用，则停止调用，直到该线程完成工作
	if (MyThreadPtr)
	{
		MyThreadPtr->WaitForCompletion();
	}
}

void UMyThreadSubsystem::PrintInfo(FString InString)
{
	/**
	* AsyncTask()用于在任务图上异步执行代码的函数。
	* @param Thread 要运行的线程名称。
	* @param Function 要执行的函数。
	*/
	AsyncTask(ENamedThreads::GameThread, [InString]() {UE_LOG(LogTemp, Warning, TEXT("ThreadLog:[%s]"), *InString); });

}
