#pragma once

#include "CoreMinimal.h"
#include "MyEnum.generated.h"

/**
* 默认情况下，C++ 枚举的底层类型是 int32，这意味着每个枚举值都占用 4 字节的内存。但如果你知道枚举值的范围非常小（例如，只需要 0 到 255 之间的值），使用 uint8（1 字节）会减少内存占用
* BlueprintType使其可以在蓝图中调用
* 注意：枚举类命名为E开头
*/
UENUM(BlueprintType)
enum class EMYUENUM : uint8
{
	None,
	Basse,
	Max
};
