MyCharacter.cpp
```cpp
// Fill out your copyright notice in the Description page of Project Settings.


#include "MyCharacter.h"
#include "MyProjectile.h"

#include "Animation/AnimInstance.h"
#include "Camera/CameraComponent.h"
#include "Components/CapsuleComponent.h"
#include "Components/SkeletalMeshComponent.h"
#include "EnhancedInputComponent.h"
#include "EnhancedInputSubsystems.h"
#include "InputActionValue.h"
#include "Engine/LocalPlayer.h"



//定义日志
DEFINE_LOG_CATEGORY(LogTemplateCharacter);

// Sets default values
AMyCharacter::AMyCharacter()
{
	//调整胶囊体大小
	GetCapsuleComponent()->InitCapsuleSize(55.0f, 96.0f);

	//创建摄像机组件
	FirstPersonCameraComponent = CreateDefaultSubobject<UCameraComponent>(TEXT("FirstPersonCamera"));//创建摄像机组建
	FirstPersonCameraComponent->SetupAttachment(GetCapsuleComponent());//设置摄像机父对象
	FirstPersonCameraComponent->SetRelativeLocation(FVector(-10.f, 0.f, 60.f));//设置摄像机相对位置
	FirstPersonCameraComponent->bUsePawnControlRotation = true;//使用角色（胶囊体）控制摄像机旋转

	//创建角色网格体组件
	Mesh1P = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("CharacterMesh1P"));//创建角色网格体组件
	Mesh1P->SetOnlyOwnerSee(true);//设置为仅自己可见
	Mesh1P->SetupAttachment(FirstPersonCameraComponent);//设置父对象为摄像机
	Mesh1P->bCastDynamicShadow = false;//关闭动态阴影
	Mesh1P->CastShadow = false;//关闭阴影
	Mesh1P->SetRelativeLocation(FVector(-30.f, 0.f, -150.f));//设置网格体相对位置
}

// Called when the game starts or when spawned
void AMyCharacter::BeginPlay()
{
	Super::BeginPlay();
	
}
//移动函数
void AMyCharacter::Move(const FInputActionValue& value)
{
	// 以FVector2D的形式获取FInputActionValue
	FVector2D MovementVector = value.Get<FVector2D>();

	//如果拥有此对象的控制器(APawn::Controller)不为空，则添加移动
	if (Controller != nullptr)
	{
		AddMovementInput(GetActorForwardVector(), MovementVector.Y);
		AddMovementInput(GetActorRightVector(), MovementVector.X);
	}
}
//观察函数
void AMyCharacter::Look(const FInputActionValue& value)
{
	// 以FVector2D的形式获取FInputActionValue
	FVector2D LookAxisVector = value.Get<FVector2D>();

	//如果拥有此对象的控制器(APawn::Controller)不为空，则添加控制器旋转
	if (Controller != nullptr)
	{
		AddControllerYawInput(LookAxisVector.X);
		AddControllerPitchInput(LookAxisVector.Y);
	}

}

// Called every frame
void AMyCharacter::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);

}

// Called to bind functionality to input
void AMyCharacter::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);

	//设置行为绑定,如果PlayerInputComponent参数是一个UEnhancedInputComponent类型，则绑定行为。
	if (UEnhancedInputComponent* EnhancedInputComponent = Cast<UEnhancedInputComponent>(PlayerInputComponent))
	{
		EnhancedInputComponent->BindAction(JumpAction, ETriggerEvent::Started, this, &ACharacter::Jump);
		EnhancedInputComponent->BindAction(JumpAction, ETriggerEvent::Completed, this, &ACharacter::StopJumping);
		EnhancedInputComponent->BindAction(MoveAction, ETriggerEvent::Triggered, this, &AMyCharacter::Move);
		EnhancedInputComponent->BindAction(LookAction, ETriggerEvent::Triggered, this, &AMyCharacter::Look);

		UE_LOG(LogTemplateCharacter, Display, TEXT("%s is BindAction !"), *GetNameSafe(this));
	}
	else
	{
		UE_LOG(LogTemplateCharacter, Error, TEXT("%s Failed to find an Enhanced Input Component!"), *GetNameSafe(this));
	}
}
```
