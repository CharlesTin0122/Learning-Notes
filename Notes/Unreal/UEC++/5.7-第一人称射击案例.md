#include "FPSCharacter.h"

// Sets default values
AFPSCharacter::AFPSCharacter()
{
 	// Set this character to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;

	//创建一个 UCameraComponent 并将其附加到角色的 CapsuleComponent
	FPSCameraComponent = CreateDefaultSubobject<UCameraComponent>(TEXT("FirstPersonCamera"));
	check(FPSCameraComponent != nullptr);
	/**
	* GetCapsuleComponent()：获取奖囊体组件
	* CastChecked<>()：将返回的胶囊组件转换为 USceneComponent 类型。USceneComponent 是所有可以被附加到其他组件的组件的基类。
	* CastChecked 会在转换失败时引发错误，因此可以确保在运行时类型安全。
	*/
	FPSCameraComponent->SetupAttachment(CastChecked<USceneComponent, UCapsuleComponent>(GetCapsuleComponent()));
	//设置摄影机位置，BaseEyeHeight是胶囊体中心上方的眼睛位置高度
	FPSCameraComponent->SetRelativeLocation(FVector(0.0f, 0.0f, 50.0f + BaseEyeHeight));
	//允许 pawn 控制摄像机的旋转
	FPSCameraComponent->bUsePawnControlRotation = true;
	//第一人称网格体
	FPSMesh = CreateDefaultSubobject<USkeletalMeshComponent>(TEXT("FirstPersonMesh"));//创建骨骼网格体
	check(FPSMesh != nullptr);//检查是否创建成功
	FPSMesh->SetOnlyOwnerSee(true);//表示该网格物体仅对拥有该角色的PlayerController可见。
	FPSMesh->SetupAttachment(FPSCameraComponent);//附加到摄像机组件
	FPSMesh->bCastDynamicShadow = false;//关闭动态阴影
	FPSMesh->CastShadow = false;//关闭阴影
	GetMesh()->SetOwnerNoSee(true);//隐藏第三人称角色

}

// Called to bind functionality to input
/*InputComponent 是定义如何处理输入数据的组件。 InputComponent 可以附加到想要接收输入的参与者。*/
void AFPSCharacter::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);

	//绑定移动输入
	PlayerInputComponent->BindAxis("MoveForward", this, &AFPSCharacter::MoveForward);
	PlayerInputComponent->BindAxis("MoveRight", this, &AFPSCharacter::MoveRight);
	// 绑定旋转输入，Character基类为我们定义了两个必要的函数
	PlayerInputComponent->BindAxis("Turn", this, &AFPSCharacter::AddControllerYawInput);
	PlayerInputComponent->BindAxis("LookUp", this, &AFPSCharacter::AddControllerPitchInput);
	//绑定跳跃输入
	PlayerInputComponent->BindAction("Jump", IE_Pressed, this, &AFPSCharacter::StartJump);
	PlayerInputComponent->BindAction("Jump", IE_Released, this, &AFPSCharacter::StopJump);
	//绑定发射输入
	PlayerInputComponent->BindAction("Fire", IE_Pressed, this, &AFPSCharacter::Fire);


}

/*移动函数*/
void AFPSCharacter::MoveForward(float Value)
{
	/*找出哪个方向是“前进”，并记录玩家想要朝那个方向移动。*/
	FRotator ControlRotation = Controller->GetControlRotation();//获取当前控制器（通常是玩家的视角或控制器）的旋转角度（以欧拉角表示）。
	FRotationMatrix ControlRotationMatrix = FRotationMatrix(ControlRotation);//将获取的旋转角度转换为一个旋转矩阵。这个矩阵用于描述如何在三维空间中旋转。
	FVector Direction = ControlRotationMatrix.GetScaledAxis(EAxis::X);//从旋转矩阵中提取 X 轴的方向向量。返回的是一个带有缩放值的向量。
	/** 这是一个用于处理角色移动的方法。它根据给定的方向和输入值（通常是一个浮点数，表示移动的强度）来更新角色的移动状态。
	 * Base Pawn 类不会自动应用移动，在 Tick 事件中由用户来执行此操作。像 Character 和 DefaultPawn 这样的子类会自动处理这个输入和移动。
	 */
	AddMovementInput(Direction, Value);
}

void AFPSCharacter::MoveRight(float Value)
{
	FVector Direction = FRotationMatrix(Controller->GetControlRotation()).GetScaledAxis(EAxis::Y);
	AddMovementInput(Direction, Value);
}

void AFPSCharacter::StartJump()

	bPressedJump = true;

void AFPSCharacter::Fire()
{
	//尝试发射一枚抛射物
	if (ProjectileClass)
	{
		//获取摄影机位置
		FVector CameraLocation;//摄像机位置
		FRotator CameraRotation;//摄像机旋转
		GetActorEyesViewPoint(CameraLocation, CameraRotation);//获取摄像机视觉中心填充摄像机位置和旋转

		//设置枪口与摄像机偏移
		MuzzleOffset.Set(100.0f, 0.0f, 0.0f);
		/**设置枪口位置
		* FTransform(CameraRotation)：将摄像机局部旋转转换为局部矩阵
		* TransformVector()：会将一个向量按照变换矩阵中的旋转和缩放部分进行线性变换，但不会应用位移部分（即平移）。
		将 MuzzleOffset 向量应用 CameraRotation 的旋转，得到一个新的向量，表示在世界空间中枪口的位置。
		*/
		FVector MuzzleLocation = CameraLocation + FTransform(CameraRotation).TransformVector(MuzzleOffset);
		//将瞄准稍微向上倾斜
		FRotator MuzzleRotation = CameraRotation;
		MuzzleRotation.Pitch += 10.0f;

		UWorld* World = GetWorld();
		if (World)
		{
			//传递给 SpawnActor 函数的可选参数结构体
			FActorSpawnParameters SpawnParams;
			SpawnParams.Owner = this;
			// 对所生成的 Actor 造成的伤害负责的 APawn。 (可以为空）
			SpawnParams.Instigator = GetInstigator();
			//在枪口发射抛射物
			AFPSProjectile* Projectile = World->SpawnActor<AFPSProjectile>(ProjectileClass, MuzzleLocation, MuzzleRotation, SpawnParams);
			if (Projectile)
			{
				//设置投射物的初始轨迹。
				FVector LaunchDirector = MuzzleRotation.Vector();
				Projectile->FireInDirection(LaunchDirector);
			}
		}
	}
}

// Fill out your copyright notice in the Description page of Project Settings.


#include "FPSProjectile.h"

// Sets default values
AFPSProjectile::AFPSProjectile()
{
 	// Set this actor to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;
	//创建根节点
	if (!RootComponent)
	{
		RootComponent = CreateDefaultSubobject<USceneComponent>(TEXT("ProjectileSceneComponent"));
	}
	//创建投射物并设为根节点
	if (!CollisionComponent)
	{
		CollisionComponent = CreateDefaultSubobject<USphereComponent>(TEXT("SphereComponent"));
		//将球体的碰撞配置文件名称设置为"Projectile".
		CollisionComponent->BodyInstance.SetCollisionProfileName(TEXT("Projectile"));
		// 组件撞击到某物时调用的事件，AddDynamic()将 UObject 实例和成员 UFUNCTION 绑定到动态多投委托的辅助宏。
		/**使用委托（Delegate）来处理事件。它注册了一个函数，来响应某个组件的碰撞事件。
		* CollisionComponent 是一个指向碰撞组件（比如一个球体碰撞体，胶囊碰撞体等）的指针
		* OnComponentHit 是 UPrimitiveComponent 的一个委托，专门用于监听碰撞事件。
		* AddDynamic 是用于动态绑定一个函数到委托的函数，它将一个对象的成员函数（在此案例中是当前对象的 OnHit 函数）与事件关联起来
		* this 指向当前类的实例对象。
		* &AFPSProjectile::OnHit 是一个成员函数指针，指向名为 OnHit 的函数，负责处理碰撞事件的逻辑。
		* 当 CollisionComponent 检测到碰撞时，会调用 AFPSProjectile 类中的 OnHit 函数。
		*/
		CollisionComponent->OnComponentHit.AddDynamic(this, &AFPSProjectile::OnHit);

		CollisionComponent->InitSphereRadius(15.0f);
		RootComponent = CollisionComponent;
	}
	//设置抛射物运动组件
	if (!ProjectileMovementComponent)
	{
		//创建抛射物运动组件
		ProjectileMovementComponent = CreateDefaultSubobject<UProjectileMovementComponent>(TEXT("ProjectileMovementComponent"));
		//设置抛射物运动组件的更新组件为CollisionComponent
		ProjectileMovementComponent->SetUpdatedComponent(CollisionComponent);
		//设置运动组件属性
		ProjectileMovementComponent->InitialSpeed = 3000.0f;//初始速度为3000cm/s
		ProjectileMovementComponent->MaxSpeed = 3000.0f;//最大速度为3000cm/s
		ProjectileMovementComponent->bRotationFollowsVelocity = true;//在每一帧更新其旋转方向，使之与其速度方向一致
		ProjectileMovementComponent->bShouldBounce = true;//是否弹跳
		ProjectileMovementComponent->Bounciness = 0.3f;//反弹后沿冲击法线方向保持的速度百分比（恢复系数）。1.0 = 无速度损失，0.0 = 无反弹
		ProjectileMovementComponent->ProjectileGravityScale = 1.0f;//重力缩放
	}
	//设置抛射物模型
	if (!ProjectileMeshComponent)
	{
		ProjectileMeshComponent = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("ProjectileMeshComponent"));
		static ConstructorHelpers::FObjectFinder<UStaticMesh>Mesh(TEXT("/Game/Character/Sphere.Sphere"));
		if (Mesh.Succeeded())
		{
			ProjectileMeshComponent->SetStaticMesh(Mesh.Object);
		}
	}
	//弹药材质
	static ConstructorHelpers::FObjectFinder<UMaterial>Material(TEXT("/Game/Character/SphereMaterial.SphereMaterial"));
	if (Material.Succeeded())
	{
		ProjectileMaterialInstance = UMaterialInstanceDynamic::Create(Material.Object, ProjectileMeshComponent);
	}
	ProjectileMeshComponent->SetMaterial(0, ProjectileMaterialInstance);
	ProjectileMeshComponent->SetRelativeScale3D(FVector(0.09f, 0.09f, 0.09f));
	ProjectileMeshComponent->SetupAttachment(RootComponent);

	//3s后销毁抛射物
	InitialLifeSpan = 3.0f;
}

//初始化弹丸在发射方向上的速度。
void AFPSProjectile::FireInDirection(const FVector ShootDirection)
{
	/* 抛射物速度向量= 发射方向 * 发射物初始速度 */
	ProjectileMovementComponent->Velocity = ShootDirection * ProjectileMovementComponent->InitialSpeed;
}
/**该方法在投射物与其他物体发生碰撞时被调用。这是被绑定到 CollisionComponent->OnComponentHit 的回调函数
* 这个函数的参数是由 CollisionComponent->OnComponentHit 的委托签名所决定的。OnComponentHit 是一个 委托（Delegate），它定义了碰撞事件触发时所需要的参数结构。
* 为了正确绑定函数，绑定的回调函数必须与委托的签名完全匹配。
* UPrimitiveComponent* HitComponent：表示当前投射物上发生碰撞的组件。
* AActor* OtherActor：表示与投射物发生碰撞的其他物体（Actor）。
* UPrimitiveComponent* OtherComponent：表示被击中的物体的具体组件。
* FVector NormalImpulse：碰撞的法线冲量，用于描述碰撞中产生的瞬时力。
* const FHitResult& Hit：碰撞的详细信息，包括碰撞点、法线、以及相关的物理数据。
*/
void AFPSProjectile::OnHit(UPrimitiveComponent* HitComponent, AActor* OtherActor, UPrimitiveComponent* OtherComponent, FVector NormalImpulse, const FHitResult& Hit)
{
	/*确保碰撞的对象不是自身（避免自我交互）,检查 OtherComponent 是否启用了物理模拟（IsSimulatingPhysics()），因为只有启用了物理的组件才能响应冲量。*/
	if (OtherActor != this && OtherComponent->IsSimulatingPhysics())
	{
		/**在碰撞点添加冲量
		* 表示投射物的速度向量，乘以 100.0f 来放大冲量的效果。
		* 将计算出的冲量添加到碰撞点（Hit.ImpactPoint）。
		*/
		OtherComponent->AddImpulseAtLocation(ProjectileMovementComponent->Velocity * 100.0f, Hit.ImpactPoint);
	}
	//投射物完成作用后销毁
	Destroy();
}

// Fill out your copyright notice in the Description page of Project Settings.


#include "FPSHUD.h"

/*DrawHUD 是 Unreal Engine 中 AHUD 类的一个虚函数，用于绘制 HUD（Head-Up Display）。在每一帧都会被调用，负责更新并绘制 HUD 元素。*/
void AFPSHUD::DrawHUD()
{
    //调用父类方法
	Super::DrawHUD();
    //检查 CrosshairTexture 是否存在
    if (CrosshairTexture)
    {
        // 找到画布的中心.Canvas->ClipX 是画布的宽度。Canvas->ClipY 是画布的高度。
        FVector2D Center(Canvas->ClipX * 0.5f, Canvas->ClipY * 0.5f);

        /**调整绘制位置以居中准星贴图
        * 为了让准星贴图的中心对齐画布中心，需要将绘制位置向左和向上偏移半个贴图的宽度和高度。
        */
        FVector2D CrossHairDrawPosition(Center.X - (CrosshairTexture->GetSurfaceWidth() * 0.5f), Center.Y - (CrosshairTexture->GetSurfaceHeight() * 0.5f));

        /**创建一个 FCanvasTileItem
        * CrossHairDrawPosition：绘制位置，前面已经计算为准星贴图的左上角点。
        * CrosshairTexture->GetResource()：准星贴图的资源，用于指定绘制的纹理。
        * FLinearColor::White：颜色为白色，表示不改变贴图本身的颜色。
        */
        FCanvasTileItem TileItem(CrossHairDrawPosition, CrosshairTexture->GetResource(), FLinearColor::White);
        //设置 TileItem 的混合模式为 SE_BLEND_Translucent，确保绘制的准星具有透明度。
        TileItem.BlendMode = SE_BLEND_Translucent;
        //将准星绘制到画布上。
        Canvas->DrawItem(TileItem);
    }

}
