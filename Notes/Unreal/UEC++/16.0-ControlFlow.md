ControlFlows是Unreal Engine 5（UE5）中新增的插件，旨在简化复杂的流程控制，特别适用于需要顺序执行多个步骤的业务逻辑。它提供了一种类似状态机的框架，但更为简化，方便开发者管理异步操作和复杂的控制流。
* 系统/工具用于排队（异步或同步）函数，以通过委托实现模块化。
* 允许代码更易于阅读，因此减少了“Alt+G”来弄清楚类的功能和位置。
* 该系统目前仅支持非const函数。
**主要功能：**

- **流程定义：**允许开发者定义一系列按顺序执行的步骤，每个步骤可以是异步操作。
    
- **流程执行：**通过调用`ExecuteFlow()`方法启动已定义的流程。
    
- **步骤控制：**每个步骤可以根据需要决定何时继续执行下一个步骤，提供灵活的流程控制。
    

**使用示例：**

在UE5的Lyra示例项目中，`ULyraFrontendStateComponent`类使用了ControlFlows插件来管理前端状态的加载流程。

```cpp
void ULyraFrontendStateComponent::OnExperienceLoaded(const ULyraExperienceDefinition* Experience)
{
    FControlFlow& Flow = FControlFlowStatics::Create(this, TEXT("FrontendFlow"))
        .QueueStep(TEXT("Wait For User Initialization"), this, &ThisClass::FlowStep_WaitForUserInitialization)
        .QueueStep(TEXT("Try Show Press Start Screen"), this, &ThisClass::FlowStep_TryShowPressStartScreen)
        .QueueStep(TEXT("Try Join Requested Session"), this, &ThisClass::FlowStep_TryJoinRequestedSession)
        .QueueStep(TEXT("Try Show Main Screen"), this, &ThisClass::FlowStep_TryShowMainScreen);

    Flow.ExecuteFlow();

    FrontEndFlow = Flow.AsShared();
}
```

在上述代码中，`FControlFlow`对象被创建，并依次添加了多个步骤（如等待用户初始化、显示开始屏幕等）。调用`ExecuteFlow()`方法后，这些步骤将按顺序执行，每个步骤完成后继续下一个。

通过使用ControlFlows插件，开发者可以更清晰地组织和管理复杂的流程控制，提升代码的可读性和维护性。

# 实例
MyControlFlowsSubsystem.h
```cpp


#pragma once

#include "CoreMinimal.h"
#include "Subsystems/GameInstanceSubsystem.h"
#include "ControlFlow.h"
#include "ControlFlowManager.h"

#include "MyControlFlowsSubsystem.generated.h"

//声明动态多播委托
DECLARE_DYNAMIC_MULTICAST_DELEGATE_TwoParams(FMyInitProgress, FGuid, InitAsyncID, double, ProgressValue);//进度更新
DECLARE_DYNAMIC_MULTICAST_DELEGATE_ThreeParams(FMyInitResult, FGuid, InitAsyncID, bool, bResult, FString, Message);//初始化结果


UCLASS()
class QUICKSTART_API UMyControlFlowsSubsystem : public UGameInstanceSubsystem
{
	GENERATED_BODY()

public:
	//是否应当创建子系统，有可能服务器不需要子系统
	virtual bool ShouldCreateSubsystem(UObject* Outer) const override;
	//初始化子系统示例，FSubsystemCollectionBase 是一个基础类，用于管理子系统（Subsystem）的注册和依赖。
	virtual void Initialize(FSubsystemCollectionBase& collection) override;
	//取消子系统实例的初始化
	virtual void Deinitialize() override;

public:
	//创建动态多播实例
	UPROPERTY(BlueprintAssignable)
	FMyInitProgress InitProgress;
	UPROPERTY(BlueprintAssignable)
	FMyInitResult InitResult;

	//构建场景函数
	UFUNCTION(BlueprintCallable)
	void InitLevel();

protected:
	FGuid InitAsyncID = FGuid();//全局唯一的标识符。
	bool bIsIniting = false;//是否正在初始化场景
	//构建场景所需任务函数，FControlFlowNodeRef是ControlFlowNode的共享引用
	void InitLocalAsset(FControlFlowNodeRef SubFlow, double InProgressValue);
	void InitNetInfo(FControlFlowNodeRef SubFlow, double InProgressValue);
	void InitUserInfo(FControlFlowNodeRef SubFlow, double InProgressValue);
	void NotifyMainUI(FControlFlowNodeRef SubFlow, double InProgressValue);
	void FinishLevelInit(FControlFlowNodeRef SubFlow, double InProgressValue);
};
```

MyControlFlowsSubsystem.cpp
```cpp



#include "MyControlFlowsSubsystem.h"
#include "Misc/FileHelper.h"
#include "Async/Async.h"
#include "Misc/Paths.h"


bool UMyControlFlowsSubsystem::ShouldCreateSubsystem(UObject* Outer) const
{
	return true;
}

void UMyControlFlowsSubsystem::Initialize(FSubsystemCollectionBase& collection)
{
	Super::Initialize(collection);
}

void UMyControlFlowsSubsystem::Deinitialize()
{
	Super::Deinitialize();
}

void UMyControlFlowsSubsystem::InitLevel()
{
	//如果场景正在初始化，则返回；如果没有在初始化，则开始初始化。
	if (bIsIniting)
	{
		UE_LOG(LogTemp, Warning, TEXT("Level is initing!"));
		return;
	}

	InitAsyncID = FGuid::NewGuid();//初始化Guid
	bIsIniting = true;


	UE_LOG(LogTemp, Warning, TEXT("[%s]--InitFlow--FrameNum:[%d]"), *FString(__FUNCTION__), GFrameCounter);

	/**创建ControlFlow,是一个用于排队（异步或同步）的类，以通过委托实现模块化。
	* Create()通过静态方法创建实例。参数1：持有者；参数2：名称。
	* 所有流应具有唯一的 ID,一个Actor最好只有一个Flow实例。如果拥有多个Flow实例，可以调用 MakeShared<FControlFlow> 而不是使用 FControlFlowStatics！
	* 返回值是FControlFlow的引用
	*/
	FControlFlow& Flow = FControlFlowStatics::Create(this, TEXT("MyControlFlow"));

	/**初始化任务流
	* 参数1：为任务名称；参数2：任务持有者；参数3：任务函数指针；参数4：函数的荷载参数进度值Progress
	*/
	Flow.QueueStep(TEXT("InitLocalAsset"), this, &UMyControlFlowsSubsystem::InitLocalAsset, 0.1);
	Flow.QueueStep(TEXT("InitNetInfo"), this, &UMyControlFlowsSubsystem::InitNetInfo, 0.2);
	Flow.QueueStep(TEXT("InitUserInfo"), this, &UMyControlFlowsSubsystem::InitUserInfo, 0.5);
	Flow.QueueStep(TEXT("NotifyMainUI"), this, &UMyControlFlowsSubsystem::NotifyMainUI, 0.8);
	Flow.QueueStep(TEXT("FinishLevelInit"), this, &UMyControlFlowsSubsystem::FinishLevelInit, 1.0);
	
	UE_LOG(LogTemp, Warning, TEXT("[%s]--ExecuteFlow--FrameNum:[%d]"), *FString(__FUNCTION__), GFrameCounter);
	//按照任务的注册顺序逐个执行
	Flow.ExecuteFlow();

}
//任务函数
void UMyControlFlowsSubsystem::InitLocalAsset(FControlFlowNodeRef SubFlow, double InProgressValue)
{
	//函数逻辑
	UE_LOG(LogTemp, Warning, TEXT("Init Local Asset--FrameNum:[%d]"), GFrameCounter);
	//线程池中任意线程拉起任务，在主线程中广播并继续
	AsyncTask(ENamedThreads::AnyThread, 
		[this, SubFlow, InProgressValue]()
		{
			FPlatformProcess::Sleep(0.2); 

			AsyncTask(ENamedThreads::GameThread, 
				[this, SubFlow, InProgressValue]()
				{
					//动态多播广播
					InitProgress.Broadcast(InitAsyncID, InProgressValue);
					//继续工作流
					SubFlow->ContinueFlow();
				}
			);
		}
	);
}

void UMyControlFlowsSubsystem::InitNetInfo(FControlFlowNodeRef SubFlow, double InProgressValue)
{
	UE_LOG(LogTemp, Warning, TEXT("Init Net Info--FrameNum:[%d]"), GFrameCounter);
	AsyncTask(ENamedThreads::AnyThread,
		[this, SubFlow, InProgressValue]()
		{
			FPlatformProcess::Sleep(0.2);

			AsyncTask(ENamedThreads::GameThread,
				[this, SubFlow, InProgressValue]()
				{
					//动态多播广播
					InitProgress.Broadcast(InitAsyncID, InProgressValue);
					//继续工作流
					SubFlow->ContinueFlow();
				}
			);
		}
	);
}

void UMyControlFlowsSubsystem::InitUserInfo(FControlFlowNodeRef SubFlow, double InProgressValue)
{
	UE_LOG(LogTemp, Warning, TEXT("Init User Info--FrameNum:[%d]"), GFrameCounter);
	AsyncTask(ENamedThreads::AnyThread,
		[this, SubFlow, InProgressValue]()
		{
			FPlatformProcess::Sleep(0.2);

			FString UserInfoPath = FPaths::ConvertRelativePathToFull(FPaths::ProjectSavedDir() / TEXT("MyControlFlowInfo.txt"));
			FString UserInitAsyncID = InitAsyncID.ToString();
			FString UseInitDateTime = FDateTime::Now().ToString();

			TArray<FString> MyStringInfo;
			MyStringInfo.Add(UserInitAsyncID);
			MyStringInfo.Add(UseInitDateTime);

			FFileHelper::SaveStringArrayToFile(MyStringInfo, *UserInfoPath, FFileHelper::EEncodingOptions::ForceUTF8WithoutBOM);

			AsyncTask(ENamedThreads::GameThread,
				[this, SubFlow, InProgressValue]()
				{
					//动态多播广播
					InitProgress.Broadcast(InitAsyncID, InProgressValue);
					//继续工作流
					SubFlow->ContinueFlow();
				}
			);
		}
	);
}

void UMyControlFlowsSubsystem::NotifyMainUI(FControlFlowNodeRef SubFlow, double InProgressValue)
{
	UE_LOG(LogTemp, Warning, TEXT("Notify Main UI--FrameNum:[%d]"), GFrameCounter);
	InitProgress.Broadcast(InitAsyncID, InProgressValue);
	SubFlow->ContinueFlow();
}

void UMyControlFlowsSubsystem::FinishLevelInit(FControlFlowNodeRef SubFlow, double InProgressValue)
{
	UE_LOG(LogTemp, Warning, TEXT("Finish Level Init--FrameNum:[%d]"), GFrameCounter);
	InitProgress.Broadcast(InitAsyncID, InProgressValue);
	InitResult.Broadcast(InitAsyncID, true, TEXT("Init Finished"));

	SubFlow->ContinueFlow();
	bIsIniting = false;
	InitAsyncID.Invalidate();
}
```
