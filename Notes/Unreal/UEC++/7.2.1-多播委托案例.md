#include "MultiDelegateActor.h"
#include "Kismet/GameplayStatics.h"

//执行Actor函数实现 
void AExcuteActor1::BeginPlay()
{
	AActor* ActorPtr = UGameplayStatics::GetActorOfClass(this, AMultiDelegateActor::StaticClass());
	if (AMultiDelegateActor* DelegeteActorPtr = Cast<AMultiDelegateActor>(ActorPtr))
	{
		//添加基于 UObject 的成员函数委托。
		MyDelegateHandle = DelegeteActorPtr->MultiDelegate.AddUObject(this, &AExcuteActor1::Work);
	}
}
//Actor销毁时调用
void AExcuteActor1::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	Super::EndPlay(EndPlayReason);
	AActor* ActorPtr = UGameplayStatics::GetActorOfClass(this, AMultiDelegateActor::StaticClass());
	if (AMultiDelegateActor* DelegeteActorPtr = Cast<AMultiDelegateActor>(ActorPtr))
	{
		//添加基于 UObject 的成员函数委托。
		DelegeteActorPtr->MultiDelegate.Remove(MyDelegateHandle);
	}
}
//执行Actor构造函数实现
AExcuteActor1::AExcuteActor1()
{
	PrimaryActorTick.bCanEverTick = false;
	RootComponent = CreateDefaultSubobject<USceneComponent>(TEXT("RootComponent"));
	MyActorID = FGuid::NewGuid();
}

{
	UE_LOG(LogTemp, Warning, TEXT("ID:[%s], Instr:[%s]"), *MyActorID.ToString(), *InStr);
}
