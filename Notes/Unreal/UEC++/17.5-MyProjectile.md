# 概述
- 该文件主要用于封装抛射物，继承自AActor。包含一个碰撞组件一个抛射物运动组件，一个命中回调函数OnHit
- 回调函数OnHit用于给被碰撞组件添加冲量
- 在碰撞组件的OnComponentHit委托中绑定回调函数OnHit
MyProjectile.h
```cpp
// Fill out your copyright notice in the Description page of Project Settings.

#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Actor.h"
#include "MyProjectile.generated.h"

class USphereComponent;
class UProjectileMovementComponent;


UCLASS()
class FPCSAMPLE_API AMyProjectile : public AActor
{
	GENERATED_BODY()

	//碰撞组件
	UPROPERTY(VisibleDefaultsOnly, Category = "Projectile")
	USphereComponent* CollisionComp;
	//抛射物移动组件
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Movement", meta = (AllowPrivateAccess = "true"))
	UProjectileMovementComponent* ProjectileMovement;
	
public:	
	// Sets default values for this actor's properties
	AMyProjectile();

protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;

public:	
	//命中函数
	UFUNCTION()
	void OnHit(UPrimitiveComponent* HitComp, AActor* OtherActor, UPrimitiveComponent* OtherComp, FVector NormalImpulse, const FHitResult& Hit);

	//获取组件函数
	USphereComponent* GetCollision() const {return CollisionComp;}
	UProjectileMovementComponent* GetProjectileMovement() const {return ProjectileMovement;}
};

```

MyProjectile.cpp
```cpp
// Fill out your copyright notice in the Description page of Project Settings.


#include "MyProjectile.h"

#include "GameFramework/ProjectileMovementComponent.h"
#include "Components/SphereComponent.h"

// Sets default values
AMyProjectile::AMyProjectile()
{
	//设置碰撞组件
	CollisionComp = CreateDefaultSubobject<USphereComponent>(TEXT("SphereComp"));//生成组件
	CollisionComp->InitSphereRadius(5.0f);//设置碰撞球半径
	CollisionComp->BodyInstance.SetCollisionProfileName("Projectile");//设置碰撞预设
	CollisionComp->OnComponentHit.AddDynamic(this, &AMyProjectile::OnHit);//为OnComponentHit委托绑定函数

	//角色不能走上抛射物
	CollisionComp->SetWalkableSlopeOverride(FWalkableSlopeOverride(WalkableSlope_Unwalkable, 0.f));//为此组件实例设置新的斜率覆盖,不能行走
	CollisionComp->CanCharacterStepUpOn = ECB_No;//确定一个角色是否可以踩上这个组件。
	RootComponent = CollisionComp;//设置碰撞组件为根组件

	//抛射物运动组件
	ProjectileMovement = CreateDefaultSubobject<UProjectileMovementComponent>(TEXT("ProjectileComp"));//创建组件
	ProjectileMovement->UpdatedComponent = CollisionComp;//设置更新组件
	ProjectileMovement->InitialSpeed = 3000.f;//设置初始速度
	ProjectileMovement->MaxSpeed = 3000.f;//设置最大速度
	ProjectileMovement->bRotationFollowsVelocity = true;//是否跟随速度矢量旋转
	ProjectileMovement->bShouldBounce = true;//是否弹跳

	//3秒生命周期
	InitialLifeSpan = 3.0f;
}

// Called when the game starts or when spawned
void AMyProjectile::BeginPlay()
{
	Super::BeginPlay();
	
}
//抛射物命中执行函数
void AMyProjectile::OnHit(UPrimitiveComponent* HitComp, AActor* OtherActor, UPrimitiveComponent* OtherComp, FVector NormalImpulse, const FHitResult& Hit)
{
	//如果对方Actor存在且不是自身，对方组件存在且计算物理
	if ((OtherActor != nullptr) && (OtherActor != this) && (OtherComp != nullptr) && OtherComp->IsSimulatingPhysics())
	{
		//对方组件添加冲量，获取自身矢量并放大100倍作用于自身位置
		OtherComp->AddImpulseAtLocation(GetVelocity() * 100.0f, GetActorLocation());
		Destroy();//销毁自身
	}
}
```
