# 概述
- 该文件主要用于获取玩家增强输入子系统，给子系统添加输入映射上下文
# 代码
MyPlayerController.h
```cpp
// Fill out your copyright notice in the Description page of Project Settings.

#pragma once

#include "CoreMinimal.h"
#include "GameFramework/PlayerController.h"
#include "MyPlayerController.generated.h"

class UInputMappingContext;
/**
 * 
 */
UCLASS()
class FPCSAMPLE_API AMyPlayerController : public APlayerController
{
	GENERATED_BODY()

protected:
	virtual void BeginPlay() override;

	//前向声明一个输入上下文映射指针
	UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input")
	UInputMappingContext* InputMappingContext;
	
};

```

MyPlayerController.cpp
```cpp
// Fill out your copyright notice in the Description page of Project Settings.


#include "MyPlayerController.h"
#include "EnhancedInputSubsystems.h"
#include "Engine/LocalPlayer.h"

void AMyPlayerController::BeginPlay()
{
	//调用父类BeginPlay()函数，一定要调用否则会导致蓝图beginPlay不执行。
	Super::BeginPlay();

	/** 如果获取玩家增强输入子系统成功，则给子系统添加输入映射上下文
	* GetLocalPlayer()获取本地玩家
	* GetSubsystem(UEnhancedInputLocalPlayerSubsystem)获取玩家增强输入子系统,返回一个增强输入子系统的指针，如果未找到返回nullptr
	*/
	if (UEnhancedInputLocalPlayerSubsystem* subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(GetLocalPlayer()))
	{
		//给子系统添加输入映射上下文,0为优先级
		subsystem->AddMappingContext(InputMappingContext, 0);
		UE_LOG(LogTemp, Display, TEXT("%s is AddMappingContext !"), *GetNameSafe(this));
	}
	else
	{
		UE_LOG(LogTemp, Error, TEXT("%s Failed to get Enhanced Input Subsystem!"), *GetNameSafe(this));
	}
}
```
