# 概述
- 该文件主要用于封装武器类，包含抛射物类，射击音效，射击动画，枪口偏移属性，增强输入系统输入上下文映射，附加武器到角色，开火输入行为，开火发射函数生成抛射物，持有一个角色指针
# 代码
MyWeaponComponent.h
```cpp
// Fill out your copyright notice in the Description page of Project Settings.

#pragma once

#include "CoreMinimal.h"
#include "Components/SkeletalMeshComponent.h"
#include "MyWeaponComponent.generated.h"

class AMyProjectile;
class UInputMappingContext;
class UInputAction;
class AMyCharacter;

/**武器组件
 * 
 */
UCLASS(Blueprintable, BlueprintType, ClassGroup = (Custom), meta = (BlueprintSpawnableComponent))
class FPCSAMPLE_API UMyWeaponComponent : public USkeletalMeshComponent
{
	GENERATED_BODY()

public:
	UMyWeaponComponent();

protected:
	UFUNCTION()
	virtual void EndPlay(const EEndPlayReason::Type EndPLayReason) override;

public:
	//抛射物类属性，TSubclassOf<>()允许以类型安全方式传递 UClass 类型的模板
	UPROPERTY(EditDefaultsOnly, Category = "Projectile")
	TSubclassOf<AMyProjectile> ProjectileClass;

	//音效属性
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "GamePlay")
	USoundBase* FireSound;
	
	//动画蒙太奇属性
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "GamePlay")
	UAnimMontage* FireAnimation;

	//枪口位置偏移
	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "GamePlay")
	FVector MuzzleOffset;

	//增强输入系统输入上下文映射指针属性
	UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input", meta = (AllowPrivateAccess = "true"))
	UInputMappingContext* FireMappingContext;

	//开火输入行为
	UPROPERTY(EditAnywhere, BlueprintReadOnly, Category = "Input", meta = (AllowPrivateAccess = "true"))
	UInputAction* FireAction;

	//附加武器到角色
	UFUNCTION(BlueprintCallable, Category = "Weapon")
	bool AttachWeapon(AMyCharacter* TargetCharacter);

	//开火发射函数
	UFUNCTION(BlueprintCallable, Category = "Weapon")
	void Fire();

private:
	AMyCharacter* Character;
};
```

MyWeaponComponent.cpp
```cpp
// Fill out your copyright notice in the Description page of Project Settings.


#include "MyWeaponComponent.h"

#include "GameFramework/PlayerController.h"
#include "Camera/PlayerCameraManager.h"
#include "Kismet/GameplayStatics.h"
#include "EnhancedInputSubsystems.h"
#include "EnhancedInputComponent.h"
#include "Animation/AnimInstance.h"
#include "Engine/LocalPlayer.h"
#include "Engine/World.h"

#include "MyCharacter.h"
#include "MyProjectile.h"

UMyWeaponComponent::UMyWeaponComponent()
{
	//设置枪口偏移向量
	MuzzleOffset = FVector(100.0f, 0.0f, 0.0f);
}

void UMyWeaponComponent::EndPlay(const EEndPlayReason::Type EndPLayReason)
{
	if (Character == nullptr)
	{
		return;
	}
	if (APlayerController* PlayerController = Cast<APlayerController>(Character->GetController()))
	{
		if (UEnhancedInputLocalPlayerSubsystem* Subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PlayerController->GetLocalPlayer()))
		{
			//添加输入映射上下文，并将优先级设置为1，使得射击行为可以覆盖跳跃行为
			Subsystem->RemoveMappingContext(FireMappingContext);
		}
	}
}

bool UMyWeaponComponent::AttachWeapon(AMyCharacter* TargetCharacter)
{
	//使用类属性接收重叠对方角色参数
	Character = TargetCharacter;
	//如果角色不存在或者角色实例组件已经有武器组件了，返回false
	if (Character == nullptr || Character->GetInstanceComponents().FindItemByClass<UMyWeaponComponent>())
	{
		return false;
	}
	//附加武器组件到角色
	FAttachmentTransformRules AttachmentRules(EAttachmentRule::SnapToTarget, true);//附加规则，吸附到目标，附加时将模拟实体焊接在一起
	AttachToComponent(Character->GetMesh(), AttachmentRules, FName(TEXT("GripPoint")));//附加到组件，附加的目标组件，附加规则，附加插槽名称

	//在角色上添加武器组件作为实例组件
	Character->AddInstanceComponent(this);
	//设置行为绑定，先获取PlayerController
	if (APlayerController* PlayerController = Cast<APlayerController>(Character->GetController()))
	{
		//通过PlayerController获取LocalPlayer，再通过ULocalPlayer::GetSubsystem获取UEnhancedInputLocalPlayerSubsystem
		if (UEnhancedInputLocalPlayerSubsystem* Subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(PlayerController->GetLocalPlayer()))
		{
			//添加输入映射上下文，并将优先级设置为1，使得射击行为可以覆盖跳跃行为
			Subsystem->AddMappingContext(FireMappingContext, 1);
		}
		//通过PlayerController获取输入组件，转化为增强输入组件
		if (UEnhancedInputComponent* EnhancedInputComponent = Cast<UEnhancedInputComponent>(PlayerController->InputComponent))
		{
			//为增强输入组件绑定射击行为函数
			EnhancedInputComponent->BindAction(FireAction, ETriggerEvent::Triggered, this, &UMyWeaponComponent::Fire);
		}
	}

	return true;
}
//开火行为方法，生成抛射物
void UMyWeaponComponent::Fire()
{
	//如果角色为空或者控制器为空，则返回
	if (Character == nullptr || Character->GetController() == nullptr)
	{
		return;
	}
	//如果抛射物类不为空，则发射
	if (ProjectileClass != nullptr)
	{
		//获取世界
		UWorld* const World = GetWorld();
		//如果世界不为空
		if (World != nullptr)
		{
			//获取PlayerController
			APlayerController* PlayerController = Cast<APlayerController>(Character->GetController());
			//通过PlayerController获取PlayerCameraManager获取GetCameraRotation获取摄像机当前旋转
			const FRotator SpawnRotation = PlayerController->PlayerCameraManager->GetCameraRotation();
			//通过GetOwner()获取当前持有此组件的Actor，获取Actor位置，Actor位置加上通过摄像机旋转过的枪口偏移向量，获取开火点位置
			const FVector SpawnLocation = GetOwner()->GetActorLocation() + SpawnRotation.RotateVector(MuzzleOffset);

			//ActorSpawnParams是一个用于生成Actor的结构体，是传递给 SpawnActor 函数的可选参数的结构体。
			FActorSpawnParameters ActorSpawnParams;
			/** SpawnCollisionHandlingOverride：是FActorSpawnParameters结构体处理生成点碰撞的方法
			* ESpawnActorCollisionHandlingMethod：定义了可用的策略，以处理演员以穿透阻挡碰撞的方式生成的情况。
			* AdjustIfPossibleButDontSpawnIfColliding：Actor 将尝试查找附近的非碰撞位置（基于形状组件），但除非找到，否则不会生成。
			*/
			ActorSpawnParams.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AdjustIfPossibleButDontSpawnIfColliding;

			//生成抛射物,参数：抛射物类，生成Actor的位置，生成Actor的朝向，生成规则
			World->SpawnActor<AMyProjectile>(ProjectileClass, SpawnLocation, SpawnRotation, ActorSpawnParams);
		}
	}
	//播放音效
	if (FireSound != nullptr)
	{
		UGameplayStatics::PlaySoundAtLocation(this, FireSound, Character->GetActorLocation());
	}
	//播放射击动画,如果射击动画存在
	if (FireAnimation != nullptr)
	{
		//获取AnimInstance
		UAnimInstance* AnimInstance = Character->GetMesh()->GetAnimInstance();
		//如果AnimInstance不为空，则1倍速播放动画蒙太奇
		if (AnimInstance != nullptr)
		{
			AnimInstance->Montage_Play(FireAnimation, 1.0f);
		}
	}
}
```
