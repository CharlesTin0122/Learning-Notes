# string的基本概念
本质：
- `string`是C++风格的字符串，而string本质上是个类。
`string`和`char*`的区别：
- `char*`是一个指针
- string是一个类，类内部封装和`char*`，管理这个字符串，是一个`char*`的容器
特点：
- string类封装了很多成员方法，如`find`，`copy`，`delete`，`replace`，`insert`
- string管理`char*`所分配的内存，不用担心越界，由类内部进行控制。
# string的构造函数
构造函数原型：
- `string();` //创建一个空字符串，例如：`string str;`
- `string(const char* s);` //使用C风格字符串`s`初始化
- `string(const string& str);` //使用一个string对象初始化另一个string对象
- `string(int n, char c);` //使用多个字符初始化，参数1为字符个数，参数2为字符
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//string容器

void test01()
{
	//创建空字符串
	string str1;
	//使用C风格字符串初始化
	const char* str = "Hello World";
	string str2(str);
	cout << "str2: " << str2 << endl;
	//调用拷贝构造函数创建字符串
	string str3(str2);
	cout << "str3: " << str3 << endl;
	//使用多个字符初始化,10个'a'
	string str4(10, 'a');
	cout << "str4: " << str4 << endl;
}
// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# string的赋值操作
赋值的函数原型：
- `string& operator=(const char* s);` // `char*`类型字符串赋值给当前字符串
- `string& operator=(const string s);` // string类型赋值给当前字符串
- `string& operator=(char c);` // 字符类型赋值给当前字符串
- `string& assign(const char* s);` // 把字符串s赋值给当前字符串
- `string& assign(const char* s, int n);` // 把字符串s的前n个字符赋值给当前字符串
- `string& assign(const string &s);` // 把字符串s赋值给当前字符串
- `string& assign(int n, char c);` // 用n个字符c赋值给当前字符串
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//string容器赋值

void test01()
{
	//`string& operator=(const char* s); //`char*`类型字符串赋值给当前字符串
	string str1;
	str1 = "Hello World";
	//`string& operator=(const string s); ` // string类型赋值给当前字符串
	string str2;
	str2 = str1;
	//`string& operator=(char c); ` // 字符类型赋值给当前字符串
	string str3;
	str3 = 'a';
	//`string& assign(const char* s); ` // 把字符串s赋值给当前字符串
	string str4;
	str4.assign("Hello World");
	//`string& assign(const char* s, int n); ` // 把字符串s的前n个字符赋值给当前字符串
	string str5;
	str5.assign("Hello World", 5);
	//`string& assign(const string & s); ` // 把字符串s赋值给当前字符串
	string str6;
	str6.assign(str5);
	//`string& assign(int n, char c); ` // 用n个字符c赋值给当前字符串
	string str7;
	str7.assign(5, 'b');

}
// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# string 字符串拼接
函数原型：
- `string& operator+=(const char* str);` // 重载+=运算符，拼接c字符串
- `string& operator+=(const char c);` // 重载+=运算符，拼接字符
- `string& operator+=(const string& str);` // 重载+=运算符， 拼接字符串
- `string& append(const char* s);` // 把字符串s拼接到当前字符串结尾
- `string& append(const char* s, int n);` //把字符串s的前n个字符拼接到字符串结尾
- `string& append(const string& s);` // 拼接字符串
- `string& append(const string& s, int pos, int n);` // 字符串s从索引pos开始到第n个字符串拼接到结尾
```cpp
#include <iostream>
#include <string>
using namespace std;

//string字符串拼接

void test01()
{
	//+=拼接

	string str1 = "我";
	str1 += "爱玩游戏";
	cout << str1 << endl;

	str1 += ':';
	cout << str1 << endl;

	string str2 = "DNF LOL";
	str1 += str2;
	cout << str1 << endl;
	//append拼接

	string str3 = "I ";
	str3.append("Love ");
	cout << str3 << endl;
	//把字符串的前5个字符拼接到字符串结尾
	str3.append("Game GTA", 5);
	cout << str3 << endl;

	//字符串str2从索引4开始到第3个字符串拼接到结尾
	str3.append(str2, 4, 3);
	cout << str3 << endl;

}
// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 字符串查找和替换
## 特点
- find从左向右查找字符串第一次出现位置。有返回索引，没有返回-1
- rfind从右向左查找字符串第一次出现位置。有返回索引，没有返回-1
## 函数定义
- `int find(const string& str, int pos = 0)const;` // 查找str第一次出现位置,从pos开始查找
- `int find(const char*s, int pos =0)const;` // 查找s第一次出现位置,从pos开始查找
- `int find(const char*s, int pos, int n)const;` // 从pos位置查找s的前n个字符第一次位置
- `int find(const char c,int pos =0)const;` // 查找字符c第一次出现位置
- `int rfind(const string& str, int pos = npos)const;` // 查找str最后一次位置,从pos开始查找
- `int rfind(const char* s,int pos=npos)const;` // 查找s最后一次出现位置,从pos开始查找
- `int rfind(const char* s,int pos, int n)const;` // 从pos查找s的前n个字符最后一次位置
- `int rfind(const char c,int pos = 0)const;` // 查找字符c最后一次出现位置
- `string& replace(int pos,int n,const string& str);` // 替换从pos开始n个字符为字符串str
- `string& replace(int pos,int n,const char* s);` // 替换从pos开始的n个字符为字符串s
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//string字符串查找和替换

//string字符串查找
void test01()
{
	//find从左向右查找字符串第一次出现位置。有返回索引，没有返回-1
	string str1 = "abcdefgde";
	int pos1 = str1.find("de");
	cout << pos1 << endl;
	//rfind从右向左查找字符串第一次出现位置。有返回索引，没有返回-1
	int pos2 = str1.rfind("de");
	cout << pos2 << endl;
}
//string字符串替换
void test02()
{
	string str1 = "abcdefg";
	//参数1：替换位置索引，参数2：替换字符个数，参数3：要替换的字符串
	str1.replace(1, 3, "1111");
	cout << str1 << endl;
}
// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 字符串比较
## 对比方式
字符串的比较是按照字符发热ASCII码进行对比的：
- = 返回 0；
- > 返回 1；
- < 返回 -1
## 函数原型
- `int compare(const string& s) const;`//与字符串s比较
- `int compare(const char* s) const;`//与C风格字符串s比较
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//string字符串比较


void test01()
{
	string str1 = "Hello";
	string str2 = "Hello";
	//比较两个字符串的ASCII码，相等返回0，大于返回1，小于返回-1.
	if (str1.compare(str2) == 0)
	{
		cout << "str1 = str2" << endl;
	}
	else if (str1.compare(str2) > 0)
	{
		cout << "str1 > str2" << endl;
	}
	else if (str1.compare(str2) < 0)
	{
		cout << "str1 < str2" << endl;
	}

}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# string字符存取
string中单个字符存取方式有两种
- `char& operator[](int n);` //通过[ ]的方式取字符
- `char& at(int n);`//通过at方法获取字符
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//string字符存取


void test01()
{
	string str1 = "Hello World";
	//1.通过[]访问单个字符
	for (int i = 0; i < str1.size(); i++)
	{
		cout << str1[i] << endl;
	}
	//2.通过at方法访问单个字符
	for (int i = 0; i < str1.size(); i++)
	{
		cout << str1.at(i) << endl;
	}
	//3.修改字符串单个字符
	str1[0] = 'X';
	str1.at(6) = 'Z';
	cout << str1 << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 字符串插入和删除
函数原型：
- `string& insert(int pos, const char* s);`//插入字符串
- `string& insert(int pos, const string& str);`//插入字符串
- `string& insert(int pos, int n, char c);`//在指定位置插入n个字符c
- `string& erase(int pos, int n=npose);`//删除从pos位置开始的n个字符
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//string字符插入和删除

void test01()
{
	string str1 = "Hello World";
	//插入,在索引1位置插入222
	str1.insert(1, "222");
	cout << str1 << endl;
	//删除，在索引1的位置移除3个字符
	str1.erase(1, 3);
	cout << str1 << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 子串获取
## 功能描述
- 从字符串中获取想要的子串
## 函数原型
- `string substr(int pos=0, int n=npos) const;`//返回由索引pos开始的n个字符组成新的字符串
## 示例
```cpp
#include <iostream>
#include <string>
using namespace std;

//string获取子串

void test01()
{
	string str1 = "Hello World";
	//返回由索引pos开始的n个字符组成新的字符串
	string str2 = str1.substr(0, 5);
	cout << str2 << endl;
}
//电子邮箱获取用户名
void test02()
{
	string email = "CharlesTin@163.com";
	int pos = email.find("@");//pos = 10
	string name = email.substr(0, pos);
	cout << "name: " << name << endl;
}
// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
