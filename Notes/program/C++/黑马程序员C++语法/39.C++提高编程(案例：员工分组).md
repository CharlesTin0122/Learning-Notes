# 案例描述
- 公司招聘了10个员工（ABCDEFGHIJ），指派员工进入部门
- 员工信息有：姓名，工资；部门为：策划，美术，研发
- 随机给十名员工分配部门和工资
- 通过multimap进行信息插入：key：部门编号；value：员工
- 分部门显示员工信息
# 实现步骤
1. 创建10名员工放入vector中
2. 遍历vector容器，取出每个员工进行部门分组
3. 分组后将部门编号作为key，具体员工作为value，放入multimap容器中
4. 分部门显示员工信息
# 案例代码
```cpp
#include <iostream>
#include <vector>
#include <map>
#include <ctime>
using namespace std;


//STL容器案例：员工分组

//员工类
class Worker
{
public:
	string m_name = "";
	int m_salary = 0;
};
//打印函数
void print_vector(const vector<Worker>& v)
{
	for (const Worker& elem : v)
	{
		cout << elem.m_name << " ";
		cout << elem.m_salary << endl;
	}
}
void print_multimap(const multimap<int, Worker>& m)
{
	cout << "策划部门： " << endl;
	for (const auto& elem : m)
	{
		if (elem.first == 0)
		{
			cout << elem.second.m_name << " ";
			cout << elem.second.m_salary << endl;
		}
	}
	cout << "------------" << endl;
	cout << "美术部门： " << endl;
	for (const auto& elem : m)
	{
		if (elem.first == 1)
		{
			cout << elem.second.m_name << " ";
			cout << elem.second.m_salary << endl;
		}
	}
	cout << "------------" << endl;
	cout << "研发部门： " << endl;
	for (const auto& elem : m)
	{
		if (elem.first == 2)
		{
			cout << elem.second.m_name << " ";
			cout << elem.second.m_salary << endl;
		}
	}
}
//创建员工函数
void create_workers(vector<Worker>& v)
{
	for (int i = 0; i < 10; i++)
	{
		//姓名种子
		string name_seed = "ABCDEFGHIJ";
		//员工对象
		Worker worker;
		//对象赋值
		worker.m_name = name_seed[i];
		worker.m_salary = rand() % 10000 + 3000;//随机数10000-19999
		//添加到容器
		v.push_back(worker);
	}
}
//员工分组函数
void set_group(vector<Worker>& v, multimap<int, Worker>& m)
{
	//遍历vector
	for (const Worker& elem : v)
	{
		int dep_id = rand() % 3;//0,1,2之间的随机数
		m.insert(make_pair(dep_id, elem));//填充map容器，key为部门编号，value为员工对象
	}
}

//测试函数
void test01()
{
	srand((unsigned int)time(NULL));
	// 创建10名员工放入vector中
	vector<Worker> v_worker;//创建容器
	create_workers(v_worker);//添加元素
	//print_vector(v_worker);//打印容器

	// 遍历vector容器，部门编号作为key，具体员工作为value，放入multimap容器中
	multimap<int, Worker> m_worker;//创建容器
	set_group(v_worker, m_worker);
	// 分部门显示员工信息
	print_multimap(m_worker);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
