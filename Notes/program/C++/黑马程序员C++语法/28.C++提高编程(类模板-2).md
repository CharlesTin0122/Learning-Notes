# 类模板成员函数类外实现
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;
// 类模板成员函数类外实现

// 定义父类模板
template <class T1, class T2>
class Person
{
public:
	T1 m_name;
	T2 m_age;

	Person(T1 name, T2 age);
	void show_info();
};
//构造函数类外实现,1.添加类模板参数列表，2.添加类参数
template <class T1, class T2>
Person<T1, T2>::Person(T1 name, T2 age)
{
	this->m_name = name;
	this->m_age = age;
}
//成员函数
template <class T1, class T2>
void Person<T1, T2>::show_info()
{
	cout << "name:" << this->m_name << endl;
	cout << "age:" << this->m_age << endl;
}

//测试函数
void test01()
{
	Person<string, int> p("Tom", 19);
	p.show_info();
}

// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");
	// 执行测试函数
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 类模板分文件编写
- 问题：类模板成员函数创建时机是在调用阶段，导致分文件编写时链接不到
- 解决方案：
	1. main.cpp文件执行代码直接包含源文件person.cpp(不建议)
	2. 将头文件和源文件写到同一个文件中，并更改文件名为.hpp,hpp为约定名称，并不强制。
- 示例：
- Person.hpp文件
```cpp
#pragma once
#include <iostream>
#include <string>
using namespace std;
//将头文件和源文件写在一起，后缀名改为.hpp

//声明
template <class T1, class T2>
class Person
{
public:
	T1 m_name;
	T2 m_age;

	Person(T1 name, T2 age);
	void show_info();
};

//实现
template <class T1, class T2>
Person<T1, T2>::Person(T1 name, T2 age)
{
	this->m_name = name;
	this->m_age = age;
}
template <class T1, class T2>
void Person<T1, T2>::show_info()
{
	cout << "name:" << this->m_name << endl;
	cout << "age:" << this->m_age << endl;
}
```
- main.cpp文件
```cpp
#include "Person.hpp"
//测试函数
void test01()
{
	Person<string, int> p("Tom", 19);
	p.show_info();
}

// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");
	// 执行测试函数
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 类模板和友元
- 友元全局函数类内实现：直接在类内声明友元即可
- 友元全局函数类外实现：需要让编辑器知道全局函数的存在
- 示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

//类模板与友元

//提前让编译器知道Person类存在
template <class T1, class T2>
class Person;

//友元全局函数类外实现，需要让编译器提前知道函数存在，故此函数前置
template<class T1, class T2>
void print_info2(Person<T1, T2> p)
{
	cout << "name:" << p.m_name << endl;
	cout << "age:" << p.m_age << endl;
}

//创建类
template <class T1, class T2>
class Person
{
	//友元全局函数类内实现
	friend void print_info1(Person<T1, T2> p)
	{
		cout << "name:" << p.m_name << endl;
		cout << "age:" << p.m_age << endl;
	}
	//友元全局函数类内声明,需增加空模板参数列表<>.
	friend void print_info2<>(Person<T1, T2> p);

private:
	T1 m_name;
	T2 m_age;
public:
	Person(T1 name, T2 age)
	{
		this->m_name = name;
		this->m_age = age;
	}
};



void test01()
{
	Person<string, int> p("Jerry", 7);
	print_info1(p);
	print_info2(p);
}

// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");
	// 执行测试函数
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
>[!INFO] 建议
>建议全局函数做类内实现，用法简单，编译器直接识别。
# 类模板案例
案例描述：实现一个通用的数组类，要求如下：
- 可以对内置数据类型和自定义数据类型进行储存
- 讲数组中的数据储存到堆区
- 构造函数可以传入数组的容量
- 提供对应的拷贝构造函数以及operator=防止浅拷贝问题
- 提供尾插法和尾删法对数组的数据进行增加和删除
- 可通过下标访问数组
- 可以获取数组中的元素个数和数组容量
示例:
- `MyArray.hpp`
```cpp
#pragma once
#include <iostream>
#include <string>
using namespace std;

//实现通用的数组类

//创建数组类
template<class T> //类模板
class MyArray
{
private:
	T* p_address;//数组堆区内存地址
	int m_capacity; //数组容量
	int m_size; //数组大小
public:
	//构造函数
	MyArray(int capacity)
	{
		this->m_capacity = capacity;//初始化数组容量
		this->m_size = 0;//初始化数组个数
		this->p_address = new T[this->m_capacity];//初始化数组
	}
	//拷贝构造
	MyArray(const MyArray& arr)
	{
		this->m_capacity = arr.m_capacity;
		this->m_size = arr.m_size;
		this->p_address = new T[arr.m_capacity];//深拷贝，重新在堆区开辟数据。防止数据重复释放。
		//如果arr中已有数据，则拷贝数据到新对象
		for (int i = 0; i < this->m_size; i++)
		{
			this->p_address[i] = arr.p_address[i];
		}
	}
	//重载operator=，防止浅拷贝问题。因为会有a=b=c的情况，所以传入一个MyArray，返回一个MyArray
	MyArray& operator=(const MyArray& arr)
	{
		//1.清空数据
		if (this->p_address != NULL)
		{
			delete[] this->p_address;
			this->p_address = NULL;
			this->m_capacity = 0;
			this->m_size = 0;
		}
		//拷贝数据
		this->m_capacity = arr.m_capacity;
		this->m_size = arr.m_size;
		this->p_address = new T[arr.m_capacity];//深拷贝，重新在堆区开辟数据。防止数据重复释放。
		//拷贝数据到新对象
		for (int i = 0; i < this->m_size; i++)
		{
			this->p_address[i] = arr.p_address[i];
		}
		return *this;//返回自身对象的指针
	}
	//尾插法：尾部插入数据功能
	void push_back(const T& val)
	{
		//判断容量是否等于大小
		if (this->m_capacity == this->m_size)
		{
			cout << "数组容量已满" << endl;
			return;
		}
		this->p_address[this->m_size] = val;//数值插入数组末尾，m_size为最后索引
		this->m_size++;//更新数组大小
	}
	//尾删法
	void pop_back()
	{
		//让用户访问不到最后一个数据
		if (this->m_size == 0)
		{
			cout << "数组为空，不必删除" << endl;
			return;
		}
		this->m_size--;//数组大小自减，用户访问不到最后一个数据
	}
	//通过索引访问数组元素，重载[]运算符,返回值为T数据类型引用，返回数据本身以方便赋值。
	T& operator[](int index)
	{
		return this->p_address[index];
	}
	//返回数组容量
	int get_capacity()
	{
		return this->m_capacity;
	}
	//返回数组大小
	int get_size()
	{
		return this->m_size;
	}
	void print_arr()
	{
		cout << "数组容量为：" << this->get_capacity() << endl;
		cout << "数组大小为：" << this->get_size() << endl;
		cout << "数组成员为：" << endl;
		for (int i = 0; i < this->m_size; i++)
		{
			cout << this->p_address[i] << endl;
		}
	}
	//析构函数
	~MyArray()
	{
		if (this->p_address != NULL)
		{
			delete[] this->p_address;
			this->p_address = NULL;
		}
	}
};
```
- `main.cpp`
```cpp
#include "MyArray.hpp" 
//测试函数
void test01()
{
	MyArray<int> arr1(5);//有参构造函数调用

	//尾插法向数组中插入数据
	for (int i = 0; i < 5; i++)
	{
		arr1.push_back(i);
	}
	arr1.print_arr();//数组输出

	MyArray<int> arr2(arr1);//拷贝构造函数调用
	arr2.print_arr();//数组输出
	arr2.pop_back();//尾删法
	arr2.print_arr();//数组输出
	//重载运算符"="调用
	MyArray<int> arr3(7);
	arr3 = arr1;
}
//测试自定义数据类型

//创建类
class Person
{
public:
	string m_name;
	int m_age;

	Person() {};
	Person(string name, int age)
	{
		this->m_name = name;
		this->m_age = age;
	}

};
//创建打印自定义类型函数
void print_person_arr(MyArray<Person>& arr)
{
	cout << "数组容量为：" << arr.get_capacity() << endl;
	cout << "数组大小为：" << arr.get_size() << endl;

	for (int i = 0; i < arr.get_size(); i++)
	{
		cout << "name:" << arr[i].m_name << endl;
		cout << "age:" << arr[i].m_age << endl;
	}
}
//测试函数
void test02()
{
	MyArray<Person> arr(10);
	Person p1("Tom", 17);
	Person p2("Jerry", 13);
	Person p3("Jim", 36);
	Person p4("Lucy", 27);
	Person p5("Lily", 26);

	arr.push_back(p1);
	arr.push_back(p2);
	arr.push_back(p3);
	arr.push_back(p4);
	arr.push_back(p5);

	print_person_arr(arr);
}



// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
