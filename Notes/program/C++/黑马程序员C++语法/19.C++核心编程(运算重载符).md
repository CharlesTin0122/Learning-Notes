对已有的运算符重新进行定义，赋予其其他功能，以适应不同的数据类型。
# 加号运算符重载
- 作用：实现两个自定义类型相加的运算
- 示例：
```cpp
#include <iostream>
#include <string>
using namespace std;
//加号运算符重载
class Person
{
public:
	//1.通过成员函数重载加号运算符
	//Person operator+ (Person &p)
	//{
	//	Person temp;
	//	temp.m_a = this->m_a + p.m_a;
	//	temp.m_b = this->m_b + p.m_b;
	//	return temp;
	//}
	int m_a;
	int m_b;
};

//2.通过全局函数重载加号运算符
Person operator+(Person& p1, Person& p2)
{
	Person temp;
	temp.m_a = p1.m_a + p2.m_a;
	temp.m_b = p1.m_b + p2.m_b;
	return temp;
}
//3.加号运算符函数重载
Person operator+(Person& p1, int num)
{
	Person temp;
	temp.m_a = p1.m_a + num;
	temp.m_b = p1.m_b + num;
	return temp;
}
void test01()
{
	Person p1;
	p1.m_a = 10;
	p1.m_b = 10;

	Person p2;
	p2.m_a = 10;
	p2.m_b = 10;
	//成员函数重载加号运算符的本质：Person p3 = p1.operator+(p2);
	//全局函数重载加号运算符的本质：Person p3 = operator+(p1, p2);
	Person p3 = p1 + p2;
	cout << "p3.m_a = " << p3.m_a << endl;
	cout << "p3.m_b = " << p3.m_b << endl;
	//运算符重载也可以发生函数重载
	Person p4 = p1 + 100;
	cout << "p4.m_a = " << p4.m_a << endl;
	cout << "p4.m_b = " << p4.m_b << endl;
}

int main()
{
	test01();

	system("pause");

	return 0;
}
```
# 左移运算符重载
- 作用：可以输出自定义数据类型
- 示例
```cpp
#include <iostream>
#include <string>
using namespace std;
//左移运算符重载
class Person
{
	friend ostream& operator<<(ostream& out, Person& p);//全局函数做友元，可以访问私有成员
public:
	Person(int a, int b)
	{
		m_a = a;
		m_b = b;
	}
private:
	int m_a;
	int m_b;
};
// 1.通过成员函数重载左移运算符, 
// `p.operator<<(cout)`,简化后`p << cout`,不符合使用习惯,所以不用成员函数重载左移运算符，而使用全局函数重载

// 2.通过全局函数重载左移运算符
// cout 是一个 ostream 类的对象，它被定义在 std 命名空间中。ostream 类是一个模板类，用于表示输出流。
// 此重载函数接受一个ostream对象的引用和 Person对象的引用作为参数，返回ostream&的对象引用,实现链式编程，可以一直 << 下去。
ostream& operator<<(ostream &out, Person &p)
{
	out << "m_a = " << p.m_a << endl;
	out << "m_b = " << p.m_b << endl;
	return out;
}

void test01()
{
	Person p1(3, 6);

	cout << p1 << endl;
}

int main()
{
	test01();

	system("pause");

	return 0;
}
```
# 递增运算符重载
- 通过重载递增运算符，实现自己的整型数据
```cpp
#include<iostream>
#include<string>
using namespace std;
//创建自定义整型类
class MyInteger
{
	friend ostream& operator<<(ostream& out, MyInteger my_int);
private:
	int m_Num;
public:
	MyInteger()
	{
		m_Num = 0;
	}
	//重载前置自增运算符
	/*返回值是一个MyInteger类的引用，++（++ my_int）是一直对一个数据进行自增。
	如果不用引用，而用值返回，则对不同的数据进行自增*/
	MyInteger& operator++()
	{
		//先运行自增运算，再将自身返回。
		m_Num++;
		return *this;
	}
	//重载后置自增运算符
	/*
	int代表展位参数，可用于区分前置和后置递增。
	重载后置自增返回的是值而不是引用，因为temp是个成员函数的局部变量，函数运行完毕temp内存数据会被释放，返回引用会报错。
	*/
	MyInteger operator++(int)
	{
		//先记录结果，后运行自增。
		MyInteger temp = *this;
		m_Num++;
		return temp;
	}
};

//重载左移运算符
ostream& operator<<(ostream& out, MyInteger my_int)
{
	out << my_int.m_Num << endl;
	return out;
}

//测试函数
void test01()
{
	MyInteger my_int;
	cout << my_int << endl;//0
	cout << ++my_int << endl;//1
	cout << ++(++my_int) << endl;//3
}
void test02()
{
	MyInteger my_int;
	cout << my_int++ << endl;//0
	cout << my_int << endl;//1
}
//主函数
int main()
{
	test01();
	test02();

	system("pause");
	return 0;
}
```
# 赋值运算符重载
C++编译器至少给一个类添加四个函数
- 默认构造函数（无参，函数体为空）
- 默认析构函数（无参，函数体为空）
- 默认拷贝构造函数，对属性进行值拷贝
- 赋值运算符(operator=)，对属性进行值拷贝
> [!WARNING] WARNING
>编译器提供的类默认赋值函数是个浅拷贝操作。
>如果类中有属性指向堆区，做赋值操作是也会出现深浅拷贝问题。
>需要重载赋值运算符为深拷贝操作。
- 示例：
```cpp
#include<iostream>
#include<string>
using namespace std;
//重载赋值运算符
class Person
{
public:
	int* m_Age;
	//构造函数
	Person(int age)
	{
		m_Age = new int(age);
	}
	//析构函数
	~Person()
	{
		if (m_Age != NULL)
		{
			delete m_Age;
			m_Age = NULL;
		}
	}
	//重载赋值运算符
	Person& operator=(Person& p)
	{
		//先判断是否有属性在堆区，如果有，先释放。
		if (m_Age != NULL)
		{
			delete m_Age;
			m_Age = NULL;
		}
		//然后再进行深拷贝。
		m_Age = new int(*p.m_Age);

		return *this;
	}
};


//测试函数
void test01()
{
	Person p1(4);
	Person p2(36);
	Person p3(89);
	/*
	* 编译器提供的类默认赋值函数是个浅拷贝操作，只是将p2的属性值赋值给p1
	* 但是，如果属性值中有指针指向堆区数值，浅拷贝只是将内存地址进行了拷贝操作，堆区数值还是一份
	* 那么程序运行结束运行析构函数时，会导致堆区内存重复释放的错误。
	* 解决方案是重载赋值运算符(operator=)，对属性进行深拷贝。
	*/
	p3 = p2 = p1;//赋值操作

	cout << "p1的年龄为：" << *p1.m_Age << endl;
	cout << "p2的年龄为：" << *p2.m_Age << endl;
	cout << "p3的年龄为：" << *p3.m_Age << endl;
}

//主函数
int main()
{
	test01();

	system("pause");
	return 0;
}
```
# 关系运算符重载
- 作用：重载关系运算符，可以让两个自定义类型对象进行对比操作
- 示例：
```cpp
#include<iostream>
#include<string>
using namespace std;
//重载关系运算符
class Person
{
public:
	string m_Name;
	int m_Age;
	//构造函数
	Person(string name, int age)
	{
		m_Name = name;
		m_Age = age;
	}
	//重载关系运算符==
	bool operator==(Person& p)
	{
		if (this->m_Name == p.m_Name && this->m_Age == p.m_Age)
		{
			return true;
		}
		else 
		{
			return false;
		}
	}
	//重载关系运算符!=
	bool operator!=(Person& p)
	{
		if (this->m_Name == p.m_Name && this->m_Age == p.m_Age)
		{
			return false;
		}
		else
		{
			return true;
		}
	}
};


//测试函数
void test01()
{
	Person p1("Jerry", 18);
	Person p2("Tom", 18);
	//判断两个对象是否相等
	if (p1 == p2)
	{
		cout << "p1 = p2" << endl;
	}
	else {
		cout << "p1 != p2" << endl;
	}
	//判断两个对象是否不等
	if (p1 != p2)
	{
		cout << "p1 != p2" << endl;
	}
	else {
		cout << "p1 = p2" << endl;
	}
}

//主函数
int main()
{
	test01();

	system("pause");
	return 0;
}
```
# 函数调用运算符重载
- 概念：仿函数（functor），又称函数对象（function object），是C++中一种可以像函数一样调用的类。它通过重载了类的 `()` 操作符来实现。
- 特点：
	- 类的函数调用运算符()也可以重载
	- 由于重载后使用的方式很像函数的调用，因此成为仿函数
	- 仿函数没有固定的写法，非常灵活
- 示例：要定义一个仿函数，只需要创建一个类并重载其 `()` 操作符。
```cpp
#include<iostream>
#include<string>
using namespace std;
//重载函数调用运算符
class MyPrint
{
public:
	//重载函数调用运算符
	void operator()(string test)
	{
		cout << test << endl;
	}
};

//仿函数非常灵活，没有固定的写法
//实现加法类
class MyAdd 
{
public:
	int operator()(int num1, int num2)
	{
		return num1 + num2;
	}
};

//测试函数
void test01()
{
	MyPrint myprint;
	myprint("Hello World");//由于使用起来非常像函数调用，因此称为仿函数。

	MyAdd myadd;
	int res = myadd(1, 2);
	cout << res << endl;

	//匿名函数对象,MyAdd()创建一个匿名对象，该对象运行完毕就释放。匿名对象(3, 6)调用函数调用运算符，执行operator()。
	cout << MyAdd()(3, 6) << endl;
}

//主函数
int main()
{
	test01();

	system("pause");
	return 0;
}
```
