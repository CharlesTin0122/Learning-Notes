# 常用查找算法
学习目标：
- 掌握常用的查找算法
算法简介：
- `find();` //查找元素
- `find_if();` //按条件查找元素
- `adjacent_find();`//查找相邻重复元素
- `binary_search();`//二分法查找
- `count();`//统计元素个数
- `count_if();`//按条件统计元素个数
## find
### 功能描述
- 查找指定元素，找到返回指定元素的迭代器，否则返回迭代器`end()`
### 函数原型
- `find(begin, end, val);`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 查找算法

//创建类
class Person
{
public:
	string m_name;
	int m_age;

	Person(string name, int age)
	{
		this->m_name = name;
		this->m_age = age;
	}
	//重载==让底层的find函数知道如何对比Person数据类型
	bool operator==(const Person& p)
	{
		if (this->m_name == p.m_name && this->m_age == p.m_age)
		{
			return true;
		}
		else
		{
			return false;
		}
	}
};

//测试函数
void test01()
{
	//查找内置数据类型

	vector<int> vec1 = { 1,4,7,2,5,8,3,6,9 };//创建源容器
	vector<int>::iterator it = find(vec1.begin(), vec1.end(), 7);
	if (it == vec1.end())
	{
		cout << "元素未找到" << endl;
	}
	else
	{
		cout << "元素已找到： " << *it << endl;
	}
}
void test02()
{
	//查找自定义数据类型

	//创建对象
	Person p1("Jim", 19);
	Person p2("Tom", 45);
	Person p3("Lucy", 27);
	Person p4("Lily", 7);
	Person p5("Jerry", 16);
	//创建容器
	vector<Person> vec2 = { p1, p2 ,p3 ,p4 ,p5 };
	//查找对象
	Person p6("Tom", 45);//创建新对象
	//查找容器中有没有属性与p6对象一致的对象
	vector<Person>::iterator pit = find(vec2.begin(), vec2.end(), p6);
	if (pit == vec2.end())
	{
		cout << "元素未找到" << endl;
	}
	else
	{
		cout << "元素已找到：" << pit->m_name << " " << pit->m_age << endl;

	}

}

// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## find_if
### 功能描述
- 按条件查找元素，找到返回指定元素的迭代器，否则返回结束位置迭代器
### 函数原型
- `find_if(begin, end, _Pred);`//参数1,2：容器区间；参数3：函数或谓词
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 find_if

//创建类
class Person
{
public:
	string m_name;
	int m_age;

	Person(string name, int age)
	{
		this->m_name = name;
		this->m_age = age;
	}
	//重载==让底层的find函数知道如何对比Person数据类型
	bool operator==(const Person& p)
	{
		if (this->m_name == p.m_name && this->m_age == p.m_age)
		{
			return true;
		}
		else
		{
			return false;
		}
	}
};

//对比函数对象，用于find_if调用
class Greater_Five
{
public:
	bool operator()(int a)
	{
		return a > 5;
	}
};
class Greater_20
{
public:
	bool operator()(Person& a)
	{
		return a.m_age > 20;
	}
};
//测试函数
void test01()
{
	//查找内置数据类型

	vector<int> vec1 = { 1,4,7,2,5,8,3,6,9 };//创建源容器
	//循环打印大于5的数
	for (vector<int>::iterator i = vec1.begin(); i != vec1.end(); i++)
	{
		//从begin开始查找大于5的数，找到之后索引变为已找到数的索引，再继续查找
		vector<int>::iterator it = find_if(i, vec1.end(), Greater_Five());
		i = it;
		cout << *i << endl;
	}
}
void test02()
{
	//查找自定义数据类型

	//创建对象
	Person p1("Jim", 19);
	Person p2("Tom", 45);
	Person p3("Lucy", 27);
	Person p4("Lily", 7);
	Person p5("Jerry", 16);
	//创建容器
	vector<Person> vec2 = { p1, p2 ,p3 ,p4 ,p5 };

	//查找容器中年龄大于20的对象
	vector<Person>::iterator pit = find_if(vec2.begin(), vec2.end(), Greater_20());
	if (pit == vec2.end())
	{
		cout << "元素未找到" << endl;
	}
	else
	{
		cout << "元素已找到：" << pit->m_name << " " << pit->m_age << endl;

	}

}

// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## adjacent_find
### 功能描述
- 查找相邻重复元素,返回第一个相邻重复元素的位置迭代器,否则返回`end()`
### 函数原型
- `adjacent_find(begin, end)`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 adjacent_find


//测试函数
void test01()
{
	vector<int> vec1 = { 1,4,7,7,5,1,3,3,9 };//创建源容器
	auto pos = adjacent_find(vec1.begin(), vec1.end());
	if (pos == vec1.end())
	{
		cout << "未找到相邻重复元素" << endl;
	}
	else
	{
		cout << "找到相邻重复元素" << *pos << endl;
	}
}
// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## binary_search
### 功能描述
- 查找指定元素是否存在，查到返回真，否则返回假。
- binary_search查找元素的容器必须是有序序列，升序或降序，无序序列不可用
- 默认为升序查找，添加参数内置函数对象`greater<int>()`可改为降序查找
### 函数原型
- `bool binary_search(begin, end, val, _Pred);`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 binary_search


//测试函数
void test01()
{
	vector<int> vec1 = { 9,8,7,6,5,4,3,2,1 };//创建源容器
	//binary_search查找元素的容器必须是有序序列，升序或降序，无序序列不可用
	//默认为升序查找，添加参数内置函数对象greater<int>()可改为降序查找
	bool res = binary_search(vec1.begin(), vec1.end(), 7, greater<int>());
	if (res)
	{
		cout << "找到元素" << endl;
	}
	else
	{
		cout << "未找到元素" << endl;
	}
}
// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## count
### 功能描述
- 统计元素出现次数
### 函数原型
- `count(begin, end, val);`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 count

//创建类
class Person
{
public:
	string m_name;
	int m_age;

	Person(string name, int age)
	{
		this->m_name = name;
		this->m_age = age;
	}
	//重载==让底层的find函数知道如何对比Person数据类型
	bool operator==(const Person& p)
	{
		if (p.m_age == this->m_age)
		{
			return true;
		}
		else
		{
			return false;
		}
	}
};
//测试函数
void test01()
{
	//内置数据类型
	vector<int> v = { 9,8,7,6,7,4,3,7,1 };//创建源容器
	int num = count(v.begin(), v.end(), 7);
	cout << "7的元素个数为：" << num << endl;
}
void test02()
{
	//自定义数据类型

	//创建对象
	Person p1("Tom", 19);
	Person p2("Jerry", 16);
	Person p3("Lucy", 27);
	Person p4("Lily", 19);
	//创建容器
	vector<Person> v2 = { p1, p2 ,p3 ,p4 };
	//查找跟Jim年龄相同的人有几个
	Person p5("Jim", 19);
	int num = count(v2.begin(), v2.end(), p5);
	cout << "与Jim同岁的人有：" << num << "个" << endl;
}
// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## count_if
### 功能描述
- 按条件统计元素出现次数
### 函数原型
- `count_if(begin, end, _Pred);`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 count_if

//创建类
class Person
{
public:
	string m_name;
	int m_age;

	Person(string name, int age)
	{
		this->m_name = name;
		this->m_age = age;
	}
};
//谓词函数
bool greater_6(const int a)
{
	return a > 6;
}
bool greater_18(const Person& p)
{
	return p.m_age > 18;
}
//测试函数
void test01()
{
	//内置数据类型
	vector<int> v = { 9,8,7,6,7,4,3,7,1 };//创建源容器
	int num = count_if(v.begin(), v.end(), greater_6);
	cout << "大于6的元素个数为：" << num << endl;
}
void test02()
{
	//自定义数据类型

	//创建对象
	Person p1("Tom", 19);
	Person p2("Jerry", 16);
	Person p3("Lucy", 27);
	Person p4("Lily", 18);
	//创建容器
	vector<Person> v2 = { p1, p2 ,p3 ,p4 };
	//查找年龄大于18的有几个
	int num = count_if(v2.begin(), v2.end(), greater_18);
	cout << "成年的有：" << num << "个" << endl;
}
// 主函数
int main()
{
	test01();
	test02();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
