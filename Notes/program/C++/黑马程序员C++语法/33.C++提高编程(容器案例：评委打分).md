# 案例描述
- 有5名选手：ABCDE，10个评委分别对每一名选手打分，分别去除最高分和最低分，去平均分
# 实现步骤
1. 创建5名选手，放置到vector容器中
2. 遍历vector容器，取出每一名选手进行循环，将十个评分存放到deque容器中
3. 用sort算法对deque容器进行排序，去除最高分和最低分
4. deque容器遍历，累加总分
5. 获取平均分
# 示例
```cpp
#include <iostream>
#include <string>
#include <vector>
#include <deque>
#include <algorithm>//包含算法模块，用于排序
#include <ctime>//包含时间模块，用于生成随机数种子
using namespace std;

//案例：评委打分

//创建选手类
class Person
{
public:
	string m_name;
	int m_score;

	Person(string name, int score)
	{
		this->m_name = name;
		this->m_score = score;
	}
};
//创建选手函数
void create_person(vector<Person>& v)
{
	string name_seed = "ABCDE";//创建姓名种子
	//遍历创建选手
	for (int i = 0; i < 5; i++)
	{
		//创建选手姓名
		string name = "Player";
		name += name_seed[i];
		//创建选手对象
		Person p(name, 0);
		//选手填充容器
		v.push_back(p);
	}
}
//评委打分函数
void set_score(vector<Person>& v)
{
	for (vector<Person>::iterator v_it = v.begin(); v_it != v.end(); v_it++)
	{
		//将评分放入deque容器
		deque<int> d;
		//10个评委打分
		for (int i = 0; i < 10; i++)
		{
			//60~100随机值
			int score = rand() % 41 + 60;//rand() % 41为0~40随机值，再加60为0~100随机值
			d.push_back(score);//打分放入容器
		}

		//打印分数
		cout << v_it->m_name << ":" << endl;
		for (deque<int>::iterator dit = d.begin(); dit != d.end(); dit++)
		{
			cout << *dit << " ";
		}
		cout << endl;

		//排序
		sort(d.begin(), d.end());
		//去除最高和最低分
		d.pop_front();
		d.pop_back();
		//获取平均分
		int sum = 0;
		for (deque<int>::iterator d_it = d.begin(); d_it != d.end(); d_it++)
		{
			sum += *d_it;//累加
		}
		int avg = sum / d.size();//已移除最高值和最低值d.size()=8
		//赋值给选手属性
		v_it->m_score = avg;
	}
}
//显示平均分
void show_score(vector<Person>& v)
{
	//通过迭代器遍历并打印vector
	for (vector<Person>::iterator it = v.begin(); it != v.end(); it++)
	{
		cout << "Player name: " << (*it).m_name << endl;
		cout << "Player score: " << (*it).m_score << endl;
	}
	cout << endl;
}
//测试函数
void test01()
{
	//随机数种子
	/*
	* srand：这是一个标准库函数，用于设置随机数生成器的种子。种子是影响随机数序列的初始值，不同的种子会产生不同的随机数序列。
	* time(NULL)：这是一个标准库函数，返回值是time_t类型,当前时间的秒数（从1970年1月1日00:00:00 UTC到当前时间的秒数）。
	NULL表示不需要指定具体的时间结构体，因此返回的是当前时间。
	* (unsigned int)：这是一个类型转换，将time(NULL)返回的值转换为无符号整数（unsigned int）类型。
	因为time(NULL)返回的时间值是time_t类型，而srand需要一个无符号整数类型的种子值，所以需要进行类型转换。
	*/
	srand((unsigned int)time(NULL));
	//创建5名选手
	vector<Person> v;
	create_person(v);
	set_score(v);
	show_score(v);

}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
