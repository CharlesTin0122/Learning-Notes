# 基本概念
- 功能：将数据进行链式存储
- 概念：链表（list）是一种物理存储单元上非连续的存储结构，数据元素的逻辑顺序是通过链表中的指针链接实现的。
- 链表组成：链表有一系列节点组成。
- 节点组成：1.储存数据元素的数据域；2.储存下一个节点地址的指针域；
![](attachments/36.C++提高编程(List容器)-1.png)
- STL中的链表是一个双向循环链表，每个节点既记录了上一个节点的地址又记录了下一个节点的地址。
- 由于链表的储存方式并不是连续的储存空间，因此链表中的迭代器只支持前移和后移，属于双向迭代器。
- 优点：
	- 可在任意位置快速进行插入和删除元素，修改指针即可，不像数组一样需要移动元素。
	- 采用动态储存分配，不会造成内存浪费和溢出。
- 缺点：
	- 链表并不是连续的储存空间，容器的遍历速度没有数组快。
	- 每个节点包含数据和指针，占用的空间比较大。
- List有一个重要的特点，插入和删除操作不会导致原List迭代器的失效，这在vector是不成立的，vector容量满了之后会重新指定更大的空间来储存，导致迭代器失效。
# 构建函数
## 功能描述
- 创建list容器
## 函数原型
- `list<t> lst;`//默认构造函数
- `list(begin, end);`//构造函数将区间内的元素拷贝给本身
- `list(n, elem);`//构造函数将n个element拷贝给本身
- `list(const list& lst);`//拷贝构造函数
## 示例
```cpp
#include <iostream>
#include <list>
using namespace std;

//list容器构造函数

//打印链表函数
void print_list(const list<int> lst)
{
	for (list<int>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << *l_it << endl;
	}
}
//测试函数
void test01()
{
	//创建容器
	list<int> lst = { 1,2,3,4,5,6 };
	//添加数据
	lst.push_back(10);
	lst.push_back(20);
	lst.push_back(30);
	lst.push_back(40);
	lst.push_back(50);
	//打印链表
	print_list(lst);
	//区间方式构造
	list<int> lst1(lst.begin(), lst.end());
	print_list(lst1);
	//拷贝构造
	list<int> lst2(lst1);
	print_list(lst2);
	//n个element
	list<int> lst3(10,999);
	print_list(lst3);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 赋值和交换
## 功能描述
- 给list容器进行赋值和交换操作
## 函数原型
- `assign(begin, end);`//区间赋值
- `assign(n, element);`//n个element赋值给本身
- `list& operator=(const list& lst);`//重载等号运算符
- `swap(lst);`//将lst与本身进行互换
## 示例
```cpp
#include <iostream>
#include <list>
using namespace std;

//list容器构造函数

//打印链表函数
void print_list(const list<int> lst)
{
	for (list<int>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << *l_it << endl;
	}
}
//测试函数
void test01()
{
	//创建容器
	list<int> lst = { 1,2,3,4,5,6 };
	print_list(lst);
	//等号运算符赋值
	list<int> lst1;
	lst1 = lst;
	print_list(lst1);
	//assign赋值
	list<int> lst2;
	lst2.assign(lst1.begin(), lst1.end());
	print_list(lst2);
	//n个element
	list<int> lst3;
	lst3.assign(3, 99);
	print_list(lst3);
	//交换
	lst3.swap(lst2);
	print_list(lst3);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 大小操作
## 函数原型
- `size();`//返回容器元素个数
- `empty();`//判断容器是否为空
- `resize(num);`//重新制定容器大小，变长0填充，变短移除
- `resize(n, elem)`//重新制定容器大小，变长elem填充，变短移除
## 示例
```cpp
#include <iostream>
#include <list>
using namespace std;

//list容器

//打印链表函数
void print_list(const list<int> lst)
{
	for (list<int>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << *l_it << endl;
	}
}
//测试函数
void test01()
{
	//创建容器
	list<int> lst = { 1,2,3,4,5,6 };
	print_list(lst);

	if (lst.empty())
	{
		cout << "链表为空" << endl;
	}
	else
	{
		cout << "链表不为空" << endl;
		cout << "元素个数为： " << lst.size() << endl;
	}
	//重新指定大小
	lst.resize(10, 999);
	print_list(lst);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 插入和移除
## 函数原型
- `push_back(elem);`//尾插
- `pop_back();`尾删
- `push_front(elem);`//首插
- `pop_front();`//首删
- `insert(pos, elem);`//在pos位置插入elem元素的拷贝，返回新数据位置
- `insert(pos, n, elem);`//在pos位置插入n个elem元素的拷贝，无返回
- `insert(pos, begin, end);`//在pos位置插入区间数据，无返回值
- `clear();`//清空容器
- `erase(begin, end);`//删除区间数据，返回下一个数据的位置
- `erase(pos);`//删除pos位置数据，返回下一个数据的位置
- `remove(elem);`//移除容器中所有与elem值匹配的元素
## 示例
```cpp
#include <iostream>
#include <list>
using namespace std;

//list容器的插入和删除

//打印链表函数
void print_list(const list<int> lst)
{
	for (list<int>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << *l_it << " ";
	}
	cout << endl;
}
//测试函数
void test01()
{
	//创建容器
	list<int> lst = { 1,2,3,4,5,6 };
	print_list(lst);

	lst.push_back(10);//尾插
	lst.push_front(20);//首插
	lst.pop_front();//首删
	lst.pop_back();//尾删
	lst.insert(lst.begin(), 30);//插入
	lst.erase(lst.begin()++);//删除
	lst.remove(4);//移除容器中所有4
	lst.clear();//清空容器
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 数据存取
## 功能描述
- 对list容器进行数据存取,链表无法通过索引和at()函数取值，只能通过迭代器前移和后移取值。
## 函数原型
- `front();`//返回第一个元素
- `back();`//返回最后一个元素
## 示例
```cpp
#include <iostream>
#include <list>
using namespace std;

//list容器的数据存取

//打印链表函数
void print_list(const list<int> lst)
{
	for (list<int>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << *l_it << " ";
	}
	cout << endl;
}
//测试函数
void test01()
{
	//创建容器
	list<int> lst = { 1,2,3,4,5,6 };
	print_list(lst);
	//获取首末元素
	cout << "首元素为： " << lst.front() << endl;
	cout << "末元素为： " << lst.back() << endl;
	//链表不是用连续线性空间储存数据，因此无法通过索引获取元素，只能通过迭代器递增或者递减获取元素
	list<int>::iterator it = lst.begin();
	cout << *it << endl;
	it++;//递增
	cout << *it << endl;
	it--;//递减
	cout << *it << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 反转和排序
## 函数原型
- `reverse();`//反转链表
- `sort();`//俩表排序
## 示例
```cpp
#include <iostream>
#include <list>
using namespace std;

//list容器的反转和排序

//打印链表函数
void print_list(const list<int> lst)
{
	for (list<int>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << *l_it << " ";
	}
	cout << endl;
}
//比较函数，用于设置排序
bool my_compare(int a, int b)
{
	//降序，就让第一个数大于第二个数
	return a > b;
}
//测试函数
void test01()
{
	//创建容器
	list<int> lst = { 1,2,3,4,5,6 };
	print_list(lst);
	//反转
	lst.reverse();
	print_list(lst);
	//排序
	lst.sort(); //默认排序规则为升序
	print_list(lst);
	//降序排列:提供一个比较函数作为参数
	lst.sort(my_compare);
	print_list(lst);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 排序案例
- 案例描述：将person自定义数据类型进行排序，属性有姓名 年龄身高；
- 排序规则：按照年龄进行升序，如果年龄相同则按照身高排序
## 示例：
```cpp
#include <iostream>
#include <list>
#include <string>
using namespace std;

//list容器的排序案例

//创建类
class Person
{
public:
	string m_name;
	int m_age;
	float m_height;

	Person(string name, int age, float height)
	{
		this->m_name = name;
		this->m_age = age;
		this->m_height = height;
	}
};
//打印链表函数
void print_list(const list<Person> lst)
{
	for (list<Person>::const_iterator l_it = lst.begin(); l_it != lst.end(); l_it++)
	{
		cout << l_it->m_name << " ";
		cout << l_it->m_age << " ";
		cout << l_it->m_height << " ";
		cout << endl;
	}

}
//指定排序规则
bool my_compare(Person& a, Person& b)
{
	//按年龄升序,如果年龄相同则按身高降序排序
	if (a.m_age == b.m_age)
	{
		return a.m_height > b.m_height;//身高降序
	}
	else
	{
		return a.m_age < b.m_age;//年龄升序
	}
	
}
//测试函数
void test01()
{
	//创建容器
	list<Person> lst;
	//创建对象
	Person p1("Jim", 19, 165.5);
	Person p2("Tom", 45, 173.7);
	Person p3("Lucy", 19, 178.4);
	Person p4("Lily", 19, 154.6);
	Person p5("Jerry", 16, 193.2);
	//插入数据
	lst.push_back(p1);
	lst.push_back(p2);
	lst.push_back(p3);
	lst.push_back(p4);
	lst.push_back(p5);

	print_list(lst);
	//排序
	cout << "------------排序后------------" << endl;
	lst.sort(my_compare);
	print_list(lst);

}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
