# 常用的拷贝和替换算法
- `copy();`//容器内指定范围的元素拷贝到另一容器
- `replace();`//将容器内指定范围的旧元素替换为新元素
- `replace_if();`//将容器内指定范围满足条件的旧元素替换为新元素
- `swap();`//互换两容器元素
## copy
### 功能描述
- 容器内指定范围的元素拷贝到另一容器
### 函数原型
- `copy(begin, end, dest);`
	- 参数1,2：源容器元素区间
	- 参数3：目标容器起始迭代器
## 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 copy

//打印函数
void print_vec(const vector<int>& v)
{
	for (const int& elem : v)
	{
		cout << elem << " ";
	}
	cout << endl;
}

//测试函数
void test01()
{
	vector<int> v1 = { 1,4,7,2,5,8,3,6,9 };//源容器
	vector<int> v2;//目标容器
	v2.resize(v1.size());//开辟目标容器的大小
	copy(v1.begin(), v1.end(), v2.begin());//拷贝数据
	print_vec(v2);//打印容器元素
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## replace
### 功能描述
- 将容器内的指定范围的旧元素全部替换为新元素
### 函数原型
- `replace(begin, end, old_vla, new_val);`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 replace

//打印函数
void print_vec(const vector<int>& v)
{
	for (const int& elem : v)
	{
		cout << elem << " ";
	}
	cout << endl;
}

//测试函数
void test01()
{
	vector<int> v = { 1,4,7,2,7,8,3,7,9 };//源容器
	replace(v.begin(), v.end(), 7, 999);//将容器中的7替换为999
	print_vec(v);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## replace_if
### 功能描述
- 将区间内满足条件的元素全部替换为指定元素
### 函数原型
- `replace(begin, end, _Pred, new_val);`
	- 参数1,2：容器区间
	- 参数3：谓词
	- 参数4：新值
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 replace_if

//打印函数
void print_vec(const vector<int>& v)
{
	for (const int& elem : v)
	{
		cout << elem << " ";
	}
	cout << endl;
}
//谓词函数
bool greater_5(const int& val)
{
	return val > 5;
}

//测试函数
void test01()
{
	vector<int> v = { 1,4,7,2,7,8,3,7,9 };//源容器
	replace_if(v.begin(), v.end(), greater_5, 999);//将容器中的大于5的数替换为999
	print_vec(v);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
## swap
### 功能描述
- 互换两个容器元素，要交换的容器为同种类型
### 函数原型
- `swap(container c1, container c2);`
### 示例
```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>
using namespace std;

//STL算法 swap

//打印函数
void print_vec(const vector<int>& v)
{
	for (const int& elem : v)
	{
		cout << elem << " ";
	}
	cout << endl;
}
//谓词函数
bool greater_5(const int& val)
{
	return val > 5;
}

//测试函数
void test01()
{
	
	vector<int> v1 = { 1,4,7,2,7,8 };//容器1
	vector<int> v2 = { 7,8,3,7,9 };//容器2
	//交换两个容器的元素
	swap(v1, v2);
	//打印元素
	print_vec(v1);
	print_vec(v2);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
