# 纯虚函数和抽象类
在多态中，通常父类中虚函数的实现是无意义的，主要都是调用子类中重写的函数实现。因此可讲虚函数改为纯虚函数。
语法：`virtual 返回值类型 函数名 (参数列表) = 0；`
当类中有了纯虚函数，这个类也成为抽象类

抽象类特点：
- 无法实例化对象
- 子类必须重写抽象类中的纯虚函数，否则也属于抽象类
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

// 纯虚函数和抽象类

// 实现抽象类
class Base
{
public:
	virtual void func() = 0;
};

// 实现子类
class Son : public Base
{
public:
	void func()
	{
		cout << "func函数被调用" << endl;
	};
};
// 测试函数
void test01()
{
	// 通过创建一个指向派生类对象的基类指针来调用纯虚函数。
	// 这样做的好处是可以通过基类指针调用派生类实现的函数，实现了多态性，即根据对象的实际类型来调用相应的函数。
	Base *base = new Son;
	base->func();
	delete base; // 释放堆区数据
	// 引用的方法调用纯虚函数
	Son son;
	Base &baseRef = son; // 使用引用
	baseRef.func();		 // 调用纯虚函数
}
// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");

	test01();

	return 0;
}
```
# 多态案例2-制作饮品
大致流程：煮水 - 冲泡 - 导入杯中 - 加入辅料
```cpp
#include <iostream>
#include <string>
using namespace std;

// 多态案例2-制作饮品

// 构建抽象类
class AbstractDrinking
{
public:
	// 构建纯虚函数
	virtual void boil() = 0;		   // 煮水
	virtual void brew() = 0;		   // 冲泡
	virtual void pour_in_cup() = 0;	   // 倒入杯中
	virtual void add_condiments() = 0; // 添加辅料
	// 制作饮品
	void make_drink()
	{
		boil();
		brew();
		pour_in_cup();
		add_condiments();
	}
};

// 冲泡咖啡
class Coffee : public AbstractDrinking
{
public:
	virtual void boil()
	{
		cout << "煮冷水" << endl;
	}
	virtual void brew()
	{
		cout << "冲泡咖啡" << endl;
	}
	virtual void pour_in_cup()
	{
		cout << "倒入杯中" << endl;
	}
	virtual void add_condiments()
	{
		cout << "加糖" << endl;
	}
};

// 冲泡茶叶
class Tea : public AbstractDrinking
{
public:
	virtual void boil()
	{
		cout << "煮冷水" << endl;
	}
	virtual void brew()
	{
		cout << "冲泡茶叶" << endl;
	}
	virtual void pour_in_cup()
	{
		cout << "倒入杯中" << endl;
	}
	virtual void add_condiments()
	{
		cout << "加枸杞" << endl;
	}
};
// 执行函数
void do_Work(AbstractDrinking *abs)
{
	abs->make_drink();
	delete abs;
}
// 测试函数
void test()
{
	do_Work(new Coffee);
	cout << "-------------------------" << endl;
	do_Work(new Tea);
};
// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");

	test();

	return 0;
}
```
# 虚析构和纯虚析构
概念：
- 多态使用时，如果子类中有属性开辟到堆区，那么父类指针在释放时无法调用子类的析构函数。
- 解决方法：将父类中的析构函数改为**虚析构**或者**纯虚析构**
虚析构和纯虚析构的共性：
- 可以解决父类指针释放子类对象
- 都需要具体的函数实现
虚析构和纯虚析构的区别：
- 如果是纯虚析构，则该类属于抽象类，无法实例化对象
虚析构语法：
`virtual ~类名(){}`
纯虚析构类内声明：
`virtual ~类名() = 0`
纯虚析构类外实现：
`类名::~类名(){}`
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

// 虚析构和纯虚析构

// 构建抽象类
class Animal
{
public:
	Animal()
	{
		cout << "Animal构造函数调用" << endl;
	}
	// virtual关键字表示虚析构函数，析构父类对象时析构子类对象。
	virtual ~Animal()
	{
		cout << "Animal析构函数调用" << endl;
	}

	// 纯虚析构函数声明
	// virtual ~Animal() = 0;

	// 构建纯虚函数
	virtual void speak() = 0;
};

// 纯虚析构函数实现
//  Animal::~Animal()
//  {
//  	cout << "Animal纯虚析构函数调用" << endl;
//  }

// 冲泡咖啡
class Cat : public Animal
{
public:
	string *m_name; // 创建名字指针变量
	// 构建函数
	Cat(string name)
	{
		cout << "Cat构造函数调用" << endl;
		// 在堆区创建字符串数据，返回指针赋予给指针变量
		m_name = new string(name);
	}
	// 析构函数,释放堆区内存
	~Cat()
	{
		if (m_name != NULL)
		{
			cout << "Cat析构函数调用" << endl;
			delete m_name;
			m_name = NULL;
		}
	}
	virtual void speak()
	{
		cout << *m_name << " Meow" << endl;
	}
};
// 测试函数
void test01()
{
	Animal *animal = new Cat("Tom"); // 父类指针指向子类对象
	animal->speak();				 // 调用子类函数
	// 释放父类指针调用父类析构函数,不会调用子类的析构函数
	// 子类如果有堆区数据，会造成内存泄漏，因此需要使用虚析构(Line:16)
	delete animal; // 释放堆区数据
}
// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");

	test01();

	return 0;
}
```
> [!INFO] INFO
> - 虚析构或者纯虚析构函数是用来解决通过父类指针释放子类对象
> - 如果子类中没有堆区数据，可以不写虚析构或纯虚析构函数
> - 拥有纯虚析构函数的类也属于抽象类。
# 电脑组装案例
案例描述：
- 电脑的主要组成部件为CPU，GPU，Memory 
- 每个零件封装出抽象基类，提供不同厂商的不通零件
- 创建电脑类，让电脑工作，调用每个零件的接口
- 测试时组装三台不通电脑
示例：
```cpp
#include <iostream>
#include <string>
using namespace std;

// 电脑组装案例

// 构建抽象类
class CPU
{
public:
	virtual void calculate() = 0;
};

class GPU
{
public:
	virtual void display() = 0;
};

class Memory
{
public:
	virtual void storage() = 0;
};
// 实现计算机类
class Computer
{
private:
	CPU *m_cpu;
	GPU *m_gpu;
	Memory *m_memory;

public:
	Computer(CPU *cpu, GPU *gpu, Memory *memory)
	{
		m_cpu = cpu;
		m_gpu = gpu;
		m_memory = memory;
	}
	// 析构函数，析构Computer对象的同时析构三个电脑零件对象
	~Computer()
	{
		if (m_cpu != NULL)
		{
			delete m_cpu;
			m_cpu = NULL;
		}
		if (m_gpu != NULL)
		{
			delete m_gpu;
			m_gpu = NULL;
		}
		if (m_memory != NULL)
		{
			delete m_memory;
			m_memory = NULL;
		}
	}

	void work()
	{
		m_cpu->calculate();
		m_gpu->display();
		m_memory->storage();
	}
};

// Intel厂商派生类
class IntelCPU : public CPU
{
public:
	virtual void calculate()
	{
		cout << "Intel CPU开始计算了" << endl;
	}
};
class IntelGPU : public GPU
{
public:
	virtual void display()
	{
		cout << "Intel GPU开始计算了" << endl;
	}
};
class IntelMemory : public Memory
{
public:
	virtual void storage()
	{
		cout << "Intel Memory开始储存了" << endl;
	}
};

// Lenovo厂商派生类
class LenovoCPU : public CPU
{
public:
	virtual void calculate()
	{
		cout << "Lenovo CPU开始计算了" << endl;
	}
};
class LenovoGPU : public GPU
{
public:
	virtual void display()
	{
		cout << "Lenovo GPU开始计算了" << endl;
	}
};
class LenovoMemory : public Memory
{
public:
	virtual void storage()
	{
		cout << "Lenovo Memory开始储存了" << endl;
	}
};

// 测试函数
void test01()
{
	// 创建英特尔电脑
	CPU *intel_cpu = new IntelCPU;
	GPU *intel_gpu = new IntelGPU;
	Memory *intel_memory = new IntelMemory;
	Computer *intel_computer = new Computer(intel_cpu, intel_gpu, intel_memory);
	intel_computer->work();
	delete intel_computer;
	cout << "----------------------------------------" << endl;
	// 创建联想电脑
	Computer *lenovo_computer = new Computer(new LenovoCPU, new LenovoGPU, new LenovoMemory);
	lenovo_computer->work();
	delete lenovo_computer;
	cout << "----------------------------------------" << endl;
	// 创建组装电脑
	Computer *mix_computer = new Computer(new IntelCPU, new LenovoGPU, new LenovoMemory);
	mix_computer->work();
	delete mix_computer;
}
// 主函数
int main()
{
	// 设置控制台输出为UTF-8编码，以支持中文输出
	system("chcp 65001>nul");
	//执行测试函数
	test01();
	//按任意键继续
	system("Pause");
	return 0;
}
```
