# set集合容器基本概念

简介：
- 所有元素都会在插入时自动排序
本质：
- set/multiset属于关联式容器，底层结构是二叉树实现的
set和multiset区别
- set不允许容器有重复的元素
- multiset容器允许容器中有重复的元素
# set的构造和赋值
## 构造函数
- `set<T> st;`//默认构造函数
- `set(const set& st);`//拷贝构造函数
## 赋值
- `set& operator=(const set& st);`//重载等号操作符
- `set.insert(elem);`//插入数据
## 示例
```cpp
#include <iostream>
#include <set>
using namespace std;

//set容器的构造和赋值

//打印set函数
void print_set(set<int>& st)
{
	for (set<int>::iterator it = st.begin(); it != st.end(); it++)
	{
		cout << *it << " ";
	}
	cout << endl;
}
//测试函数
void test01()
{
	set<int> st1 = {20, 7, 40, 50, 15};
	st1.insert(3);
	st1.insert(6);
	st1.insert(7);
	st1.insert(2);
	st1.insert(4);
	//自动排序，自动去重
	print_set(st1);
	//拷贝构造
	set<int> st2(st1);
	print_set(st2);
	//赋值操作
	set<int> st3;
	st3 = st2;
	print_set(st3);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 大小和交换
## 函数原型
- `size();`
- `empty();`
- `swap(st);`
## 示例
```cpp
#include <iostream>
```

# 插入和删除
## 函数原型
- `insert(elem);`//插入元素
- `clear();`//清空容器
- `erase(pos);`//移除pos迭代器元素，返回下一个元素的迭代器
- `erase(begin, end);`//移除迭代器区间元素，返回下一个元素的迭代器
- `erase(elem);`//移除容器中值为element的元素
## 示例
```cpp
#include <iostream>
#include <set>
using namespace std;

//set容器的构造和赋值

//打印set函数
void print_set(set<int>& st)
{
	for (set<int>::iterator it = st.begin(); it != st.end(); it++)
	{
		cout << *it << " ";
	}
	cout << endl;
}
//测试函数
void test01()
{
	set<int> st1 = {20, 7, 40, 50, 15};
	//插入
	st1.insert(3);
	st1.insert(6);
	st1.insert(7);
	st1.insert(2);
	st1.insert(4);

	print_set(st1);
	//删除
	st1.erase(st1.begin());//移除首位
	print_set(st1);

	st1.erase(7);//移除值为7的元素
	print_set(st1);

	st1.erase(st1.begin(), st1.end());//移除区间
	print_set(st1);

	st1.clear();//清空

}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 查找和统计
## 函数原型
- `find(key);`//查找key是否存在，若存在，返回该元素的迭代器；若不在，返回set.end();
- `count(key);`//返回key的元素个数
## 示例
```cpp
#include <iostream>
#include <set>
using namespace std;

//set容器的查找和统计

//打印set函数
void print_set(set<int>& st)
{
	for (const auto& elem : st)
	{
		cout << elem << " ";
	}
	cout << endl;
}
//测试函数
void test01()
{
	set<int> st1 = {20, 7, 40, 50, 15};
	//查找
	set<int>::iterator pos = st1.find(7);
	//查找key是否存在，若存在，返回该元素的迭代器；若不在，返回set.end();
	if (pos != st1.end())
	{
		cout << "找到元素 " << *pos << endl;
	}
	else
	{
		cout << "未找到元素" << endl;
	}
	//统计
	int num = st1.count(40);
	cout << num << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# set和multiset区别
- set不允许容器有重复的元素,multiset容器允许有重复的元素
- set插入数据的的同时会返回插入的结果，表示插入是否成功
- multiset不会监测数据，因此可以重复插入数据
- `set.insert()`的返回值是pair(对组)，pair包含两个值：1.迭代器，插入成功返回迭代器；2.布尔值，返回插入是否成功；
- `multiset.insert()`返回值是迭代器
## 示例
```cpp
#include <iostream>
#include <set>
using namespace std;

//set和multset

//打印set函数
void print_set(set<int>& st)
{
	for (const auto& elem : st)
	{
		cout << elem << " ";
	}
	cout << endl;
}
//测试函数
void test01()
{
	//set

	set<int> st1 = {20, 7, 40, 50, 15};
	print_set(st1);

	//insert的返回值是pair(对组),pair包含两个值：1.迭代器，插入成功返回迭代器；2.布尔值，返回插入是否成功
	pair<set<int>::iterator, bool> ret = st1.insert(8);
	//pair.first是迭代器；pair.second是布尔值；
	if (ret.second)
	{
		cout << "插入成功" << endl;
		cout << *ret.first << endl;
	}
	else
	{
		cout << "插入失败" << endl;
		cout << *ret.first << endl;
	}
	print_set(st1);
	//再次插入
	ret = st1.insert(40);
	if (ret.second)
	{
		cout << "插入成功" << endl;
		cout << *ret.first << endl;
	}
	else
	{
		cout << "插入失败" << endl;
		cout << *ret.first << endl;
	}
	print_set(st1);

	//multiset

	multiset<int> ms = { 1,2,3,4,5 };
	//multiset允许插入重复值,返回该位置的迭代器
	ms.insert(10);
	ms.insert(10);
	for (const auto& elem : ms)
	{
		cout << elem << " ";
	}
	cout << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# pair（对组）

# set容器排序
## 学习目标
- set容器默认排序规则为从小到大，尝试改变排序规则
## 技术要点
- 利用仿函数，可以改变排序规则
## 示例
- set存放内置数据类型
```cpp
#include <iostream>
#include <set>
using namespace std;

//set容器排序


//仿函数实现set容器降序排列
class MyCompare
{
public:
	//重载()运算符,使这个类的对象表现得像一个函数。
	//const：表明这个成员函数不会修改类的任何成员变量。
	bool operator()(int a , int b) const
	{
		return a > b;
	}
};

//测试函数
void test01()
{
	set<int> s1 = {2, 8, 6, 4, 9, 7};

	for (const auto& elem : s1)
	{
		cout << elem << " ";
	}
	cout << endl;
	//指定排序规则为降序，利用仿函数
	//当insert需要比较两个元素时，就会调用MyCompare的operator()函数，依据 a > b 进行排序。
	set<int, MyCompare> s2;
	s2.insert(10);
	s2.insert(40);
	s2.insert(20);
	s2.insert(50);
	s2.insert(30);

	for (const auto& elem : s2)
	{
		cout << elem << " ";
	}
	cout << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
- set存放自定义数据类型
```cpp
#include <iostream>
#include <set>
using namespace std;

//set容器自定义数据类型排序

//创建类
class Person
{
public:
	string m_name;
	int m_age;

	Person(string name, int age)
	{
		this->m_name = name;
		this->m_age = age;
	}
};

//自定义数据类型必须指定排序规则，仿函数实现set容器降序排列
class MyCompare
{
public:
	//重载()运算符,使这个类的对象表现得像一个函数。
	//const：表明这个成员函数不会修改类的任何成员变量。
	bool operator()(const Person& a , const Person& b) const
	{
		return a.m_age > b.m_age;
	}
};

//测试函数
void test01()
{
	set<Person, MyCompare> s1;

	//创建对象
	Person p1("Jim", 19);
	Person p2("Tom", 45);
	Person p3("Lucy", 27);
	Person p4("Lily", 7);
	Person p5("Jerry", 16);

	s1.insert(p1);
	s1.insert(p2);
	s1.insert(p3);
	s1.insert(p4);
	s1.insert(p5);

	for (const auto& elem : s1)
	{
		cout << elem.m_name << " ";
		cout << elem.m_age << " ";
		cout << endl;
	}

}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
