# map容器的概念
## 简介
- map中所有元素都是pair
- pair中第一个元素为key（键），起到索引的作用，第二个元素为value（值）
- 所有元素都会根据元素的键自动排序
## 本质
- map/multimap属于关联式容器，底层结构是二叉树实现的
## 优点
- 可以根据key快速找到value
## map和multimap的区别
- map不允许容器中有重复的key元素
- multimap允许容器中有重复的key元素
# map的构造和赋值
## 函数原型
### 构造
- `map<T1, T2> mp;`//默认构造
- `map(const map& mp);`//拷贝构造函数
### 赋值
- `map& operator=(const map& mp);`//重载等号运算符
## 示例
```cpp
#include <iostream>
#include <map>
using namespace std;

//map容器的构造和赋值

//打印map函数
void print_map(const map<int, string>& mp)
{
	for (const pair<int, string>& elem : mp)
	{
		cout << elem.first << ": ";
		cout << elem.second << endl;
	}
}


//测试函数
void test01()
{
	//创建容器
	map<int, string> mp;
	//插入元素，插入值是一个匿名pair对象
	mp.insert(pair<int, string>(4, "Lucy"));
	mp.insert(pair<int, string>(1, "Tom"));
	mp.insert(pair<int, string>(5, "Lily"));
	mp.insert(pair<int, string>(3, "Jim"));
	mp.insert(pair<int, string>(2, "Jerry"));
	//打印输出
	print_map(mp);
	//拷贝构造
	map<int, string> mp1(mp);
	print_map(mp1);
	//赋值操作
	map<int, string> mp2;
	mp2 = mp1;
	print_map(mp2);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# map容器的大小和交换
## 函数原型
- `size();`//返回元素数量
- `empty();`//判断容器是否为空
- `swap(st);`//交换两个容器
## 示例
```cpp
#include <iostream>
#include <map>
using namespace std;

//map容器的大小和交换

//打印map函数
void print_map(const map<int, string>& mp)
{
	for (const pair<int, string>& elem : mp)
	{
		cout << elem.first << ": ";
		cout << elem.second << endl;
	}
}
//测试函数
void test01()
{
	//创建容器
	map<int, string> mp;
	//插入元素，插入值是一个匿名pair对象
	mp.insert(pair<int, string>(4, "Lucy"));
	mp.insert(pair<int, string>(3, "Jim"));
	mp.insert(pair<int, string>(5, "Lily"));
	//容器大小
	if (mp.empty())
	{
		cout << "容器为空" << endl;
	}
	else
	{
		cout << "容器不为空" << endl;
		cout << "元素个数为： " << mp.size() << endl;
	}
	//容器元素交换
	map<int, string> mp1;

	mp1.insert(pair<int, string>(1, "Tom"));
	mp1.insert(pair<int, string>(2, "Jerry"));

	cout << "交换前" << endl;
	print_map(mp1);

	mp1.swap(mp);

	cout << "交换后" << endl;
	print_map(mp1);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 插入和删除
## 函数原型
- `insert(elem);`//插入元素
- `clear();`//清除元素
- `erase(pos);`//移除迭代器所指元素，返回下一个元素的迭代器
- `erasr(begin, end);`//删除区间元素，返回下个元素的迭代器
- `erase(key);`//移除容器中key键的元素
## 示例
```cpp
#include <iostream>
#include <map>
using namespace std;

//map容器的插入和移除

//打印map函数
void print_map(const map<int, string>& mp)
{
	for (const pair<int, string>& elem : mp)
	{
		cout << elem.first << ": ";
		cout << elem.second << endl;
	}
}
//测试函数
void test01()
{
	//创建容器
	map<int, string> mp;
	//插入元素，三种插入方法
	mp.insert(pair<int, string>(4, "Lucy"));//pair匿名对象
	mp.insert(make_pair(3, "Jim"));//make_pair函数
	mp.insert(map<int, string>::value_type(5, "Lily"));//value_type函数
	mp[1] = "Jim";//[]赋值，不建议使用

	cout << mp[2] << endl;//查询不存在的元素，会插入此元素，值为0或者空
	print_map(mp);
	//元素移除
	mp.erase(mp.begin());//移除首元素
	print_map(mp);
	mp.erase(3);//移除key为3的元素
	print_map(mp);

	mp.clear();//清空容器
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 查找和统计
## 函数原型
- `find(key);`//查找key元素是否存在。若存在，返回该元素的迭代器；若不在，返回set.end();
- `count(key);`//统计key的元素个数
## 示例
```cpp
#include <iostream>
#include <map>
using namespace std;

//map容器的查找和统计

//打印map函数
void print_map(const map<int, string>& mp)
{
	for (const pair<int, string>& elem : mp)
	{
		cout << elem.first << ": ";
		cout << elem.second << endl;
	}
}
//测试函数
void test01()
{
	//创建容器
	map<int, string> mp;

	mp.insert(pair<int, string>(4, "Lucy"));
	mp.insert(pair<int, string>(3, "Jim"));
	mp.insert(pair<int, string>(5, "Lily"));
	mp.insert(pair<int, string>(1, "Tom"));
	mp.insert(pair<int, string>(2, "Jerry"));
	//查找
	map<int, string>::iterator pos = mp.find(3);//查找key为3的元素，返回该元素的迭代器,未找到返回map.end()
	if (pos != mp.end())
	{
		cout << "找到该元素" << endl;
		cout << "该元素值为： " << pos->second << endl;
	}
	else
	{
		cout << "未找到元素" << endl;
	}
	//统计,multimap允许重复key
	multimap<int, string> mmp;
	mmp.insert(pair<int, string>(4, "Lucy"));
	mmp.insert(pair<int, string>(3, "Jim"));
	mmp.insert(pair<int, string>(4, "Lily"));
	mmp.insert(pair<int, string>(1, "Tom"));
	mmp.insert(pair<int, string>(4, "Jerry"));
	int num = mmp.count(4);
	cout << "num =  " << num << endl;
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
# 排序
## 学习目标
- map容器排序规则为按照key值从小到大排序，掌握如何改变排序规则
## 主要技术点
- 利用仿函数，改变排序规则
## 示例
```cpp
#include <iostream>
#include <map>
using namespace std;

//map容器的查找和统计

//仿函数，用于改变排序规则
class MyCompare
{
public:
	//仿函数，重载()运算符,比较两个key的值，使其降序排列
	bool operator()(int a, int b)const
	{
		return a > b;
	}
};

//打印map函数
void print_map(const map<int, string, MyCompare>& mp)
{
	for (const auto& elem : mp)
	{
		cout << elem.first << ": ";
		cout << elem.second << endl;
	}
}

//测试函数
void test01()
{
	//创建容器,改变排序规则仿函数放在模板参数列表
	map<int, string, MyCompare> mp;

	mp.insert(pair<int, string>(4, "Lucy"));
	mp.insert(pair<int, string>(3, "Jim"));
	mp.insert(pair<int, string>(5, "Lily"));
	mp.insert(pair<int, string>(1, "Tom"));
	mp.insert(pair<int, string>(2, "Jerry"));

	print_map(mp);
}

// 主函数
int main()
{
	test01();
	// 按任意键继续
	system("Pause");
	return 0;
}
```
