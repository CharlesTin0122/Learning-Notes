#include <cstdlib> // malloc, free
#include <iostream>
#include <utility> // std::forward

// 自定义分配器
template <typename T>
struct MyAllocator
{
    // 分配内存
    T *allocate(size_t size) { return static_cast<T *>(malloc(sizeof(T) * size)); }
    // 释放内存
    void deallocate(T *ptr)
    {
        if (ptr) free(ptr);
    }
    // 构造对象
    template <typename... Types>
    void construct(T *ptr, Types &&...args)
    {
        new (ptr) T(std::forward<Types>(args)...);
    }
    // 销毁对象
    void destroy(T *ptr)
    {
        if (ptr) ptr->~T();
    }
};

// 容器实现
template <typename T, typename Alloc = MyAllocator<T>>
class MyVector
{
  public:
    MyVector() : start_(nullptr), size_(0), idx_(0) {}

    ~MyVector()
    {
        clear();                       // 销毁所有对象
        allocator_.deallocate(start_); // 释放内存
    }

    void reserve(size_t size)
    {
        if (size <= size_) return;                // 如果请求大小不超过现有大小，直接返回
        T *new_start = allocator_.allocate(size); // 分配新内存
        // 将现有对象移动到新内存（如果有对象）
        for (size_t i = 0; i < idx_; ++i)
        {
            allocator_.construct(new_start + i, std::move(start_[i]));
            allocator_.destroy(start_ + i);
        }
        allocator_.deallocate(start_); // 释放旧内存
        start_ = new_start;
        size_ = size;
    }

    template <typename Type>
    void push_back(Type &&val)
    {
        ensure_capacity(); // 检查并扩容
        allocator_.construct(start_ + idx_, std::forward<Type>(val));
        ++idx_;
    }

    template <typename... Types>
    void emplace_back(Types &&...args)
    {
        ensure_capacity(); // 检查并扩容
        allocator_.construct(start_ + idx_, std::forward<Types>(args)...);
        ++idx_;
    }

    void clear()
    {
        // 销毁所有对象
        for (size_t i = 0; i < idx_; ++i)
        {
            allocator_.destroy(start_ + i);
        }
        idx_ = 0; // 重置索引
    }

    size_t size() const { return idx_; }
    size_t capacity() const { return size_; }

  private:
    void ensure_capacity()
    {
        if (idx_ >= size_)
        {
            reserve(size_ == 0 ? 1 : size_ * 2); // 初始分配或扩容为两倍
        }
    }

    T *start_;        // 数组起始指针
    size_t size_;     // 容量
    size_t idx_;      // 当前元素数量
    Alloc allocator_; // 分配器
};

// 测试类
class Test
{
  public:
    Test(int a) { std::cout << "Test(int)" << std::endl; }                 // 构造函数
    Test(int a, int b) { std::cout << "Test(int, int)" << std::endl; }     // 构造函数
    Test(const Test &) { std::cout << "Test(const Test &)" << std::endl; } // 拷贝构造函数
    Test(Test &&) { std::cout << "Test(Test &&)" << std::endl; }           // 移动构造函数

    ~Test() { std::cout << "~Test()" << std::endl; } // 析构函数
};

int main()
{

    Test t1(10);

    MyVector<Test> vec;
    vec.reserve(100);

    std::cout << "----------------------------" << std::endl;
    // 直接插入对象，调用拷贝构造函数，两者没有区别
    vec.push_back(t1);
    vec.emplace_back(t1);
    // 插入临时对象，调用移动构造函数，两者也没有区别
    vec.push_back(Test(20));
    vec.emplace_back(Test(20));
    std::cout << "----------------------------" << std::endl;
    // 给emplace函数传入对象构造所需要的参数，直接在容器底层构造对象。
    vec.emplace_back(20);
    vec.emplace_back(30, 40);
}
