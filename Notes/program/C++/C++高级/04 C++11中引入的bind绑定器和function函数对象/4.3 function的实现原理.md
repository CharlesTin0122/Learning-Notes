```cpp
#include <iostream>
#include <string>

// function的实现
template <typename R, typename... A> // ...表示可变参，参数数量未知
class MyFunction
{
public:
    // 函数指针类型
    using funcPtr = R (*)(A...);

    // 构造函数，传入一个函数指针
    MyFunction(funcPtr func) : _func(func) {}

    // ()重载,返回值类型为R,参数为A类型的arg
    R operator()(A... arg) { return _func(arg...); }

private:
    funcPtr _func; // 函数指针
};

void hello(std::string str) { std::cout << str << std::endl; }
int sum(int a, int b) { return a + b; }

// 主函数
int main()
{
    MyFunction<void, std::string> func1(hello);
    func1("Hello World!"); // func1.operator()("Hello World!")

    MyFunction<int, int, int> func2(sum);
    std::cout << func2(10, 20) << std::endl;

    return 0;
}
```
