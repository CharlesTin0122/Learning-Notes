#include <algorithm>
#include <cstdlib>
#include <ctime>
#include <functional>
#include <iostream>
#include <vector>

template <typename T>
void showContainer(const T &con)
{
    auto it = con.begin();
    while (it != con.end())
    {
        std::cout << *it << " ";
        ++it;
    }
    std::cout << std::endl;
}
// 实现一元函数对象类
template <typename Compare, typename T>
class monadic_predicate
{
public:
    monadic_predicate(Compare comp, T val) : _comp(comp), _val(val) {}
    bool operator()(const T &second) { return _comp(_val, second); }

private:
    Compare _comp;
    T _val;
};

// 实现绑定器,函数接收一个二元函数对象和一个值，返回一个一元函数对象
template <typename Compare, typename T>
monadic_predicate<Compare, T> my_bind1st(Compare comp, const T &val)
{
    return monadic_predicate<Compare, T>(comp, val);
}

// 实现 find_if,模板参数1：迭代器。参数2：一元函数对象
template <typename Iterator, typename Compare>
Iterator my_find_if(Iterator first, Iterator last, Compare comp)
{
    while (first != last)
    {
        if (comp(*first)) { return first; }
        first++;
    }
    return last;
}
// 主函数
int main()
{
    std::vector<int> vec;
    srand((unsigned int)time(nullptr)); // 随机数种子
    // 填充容器
    for (int i = 0; i < 20; ++i)
    {
        vec.push_back(rand() % 100 + 1);
    }

    // 利用二元函数对象greater，降序排列
    std::sort(vec.begin(), vec.end(), std::greater<int>());
    showContainer(vec);

    // 将70按序插入容器，在降序容器中找第一个小于70的数字，并插入该位置
    // 使用绑定器，将greater函数对象的第一个值绑定为70，使a > b 变为 70 > b
    auto it = std::find_if(vec.begin(), vec.end(),
                           my_bind1st(std::greater<int>(), 70));
    // 如果找到该位置，插入70元素
    if (it != vec.end()) { vec.insert(it, 70); }
    showContainer(vec);

    return 0;
}
