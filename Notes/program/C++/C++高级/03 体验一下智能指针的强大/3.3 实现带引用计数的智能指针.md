# 概念
- 带引用计数的智能指针：可以实现多个智能指针管理同一资源
- 带引用计数：给每一个对象资源匹配一个引用计数
	- 当一个智能指针引用一个资源的时候资源的引用计数加一。
	- 当一个只能指针不再使用资源的时候（指针出作用域）。引用计数减一。
	- 当引用计数为0的时候，资源释放。

# 实现一个带引用计数的智能指针
```cpp
#include <iostream>

// 资源的引用计数类
template <typename T>
class RefCount
{
public:
    RefCount(T *ptr = nullptr) : _ptr(ptr), _count(ptr ? 1 : 0)
    {
        std::cout << "RefCount()" << std::endl;
    }

    void addRef() { _count++; }
    int delRef() { return --_count; }
    int count() const { return _count; }

private:
    T *_ptr;
    int _count;
};

// 实现带引用计数的智能指针
template <typename T>
class SmartPtr
{
public:
    SmartPtr(T *ptr = nullptr) : _ptr(ptr)
    {
        std::cout << "SmartPtr()" << std::endl;
        _refCntPtr = new RefCount<T>(_ptr);
    }

    SmartPtr(const SmartPtr<T> &other)
        : _ptr(other._ptr), _refCntPtr(other._refCntPtr)
    {
        std::cout << "SmartPtr(&other)" << std::endl;
        if (_refCntPtr) { _refCntPtr->addRef(); }
    }

    ~SmartPtr()
    {
        std::cout << "~SmartPtr()" << std::endl;
        if (_refCntPtr && _refCntPtr->delRef() == 0)
        {
            delete _refCntPtr; // 释放引用计数对象
            delete _ptr;       // 释放资源
        }
    }

    SmartPtr<T> &operator=(const SmartPtr<T> &other)
    {
        std::cout << "&operator=()" << std::endl;
        if (this != &other)
        {
            if (_refCntPtr->delRef() == 0)
            {
                delete _refCntPtr; // 释放原引用计数对象
                delete _ptr;       // 释放原资源
            }
            _ptr = other._ptr;
            _refCntPtr = other._refCntPtr;
            if (_refCntPtr) { _refCntPtr->addRef(); } // 引用计数增加
        }
        return *this;
    }

    T &operator*()
    {
        std::cout << "operator*()" << std::endl;
        return *_ptr;
    }

    T *operator->()
    {
        std::cout << "operator->()" << std::endl;
        return _ptr;
    }

    RefCount<T> *getRefCount() const { return _refCntPtr; }

private:
    T *_ptr;
    RefCount<T> *_refCntPtr;
};

// 主函数
int main()
{
    SmartPtr<int> ptr1(new int(10));
    std::cout << ptr1.getRefCount()->count() << std::endl;

    SmartPtr<int> ptr2(ptr1);
    std::cout << ptr1.getRefCount()->count() << std::endl;

    SmartPtr<int> ptr3;
    ptr3 = ptr2;
    std::cout << ptr1.getRefCount()->count() << std::endl;

    *ptr1 = 20;
    std::cout << "ptr2: " << *ptr2 << " ptr3: " << *ptr3 << std::endl;

    return 0;
}
```
