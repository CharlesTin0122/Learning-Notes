# 多线程访问共享对象的安全问题
```cpp
#include <chrono>
#include <iostream>
#include <memory>
#include <thread>

class A
{
public:
    A() { std::cout << "A()" << std::endl; }
    ~A() { std::cout << "~A()" << std::endl; }
    void testA() { std::cout << "A::testA()" << std::endl; }
};
// 子线程
void handler01(std::weak_ptr<A> q)
{
    // q在访问A对象的时候，需要侦测A对象是否存活。
    std::shared_ptr<A> r = q.lock();
    if (r) { r->testA(); }
    else { std::cout << "A对象已经析构,无法访问" << std::endl; }
}
// mian线程
int main()
{
    {
        std::shared_ptr<A> p(new A);
        std::thread t1(handler01, std::weak_ptr<A>(p));
        t1.detach();

        std::this_thread::sleep_for(std::chrono::seconds(2));
    }

    // 主线程睡眠20秒
    std::this_thread::sleep_for(std::chrono::seconds(20));

    return 0;
}
```
