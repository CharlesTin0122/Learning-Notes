当一个类中的数据在堆上储存时，拷贝构造函数和等号运算符重载都需要另外开辟内存空间来储存数据，当函数处理数据要返回对象的时候，需要创建临时对象，临时对象造成大量的空间浪费。
```cpp
#include <cstring>
#include <iostream>
// String类
class String
{
public:
    // 构造函数
    String(const char *p = nullptr)
    {
        std::cout << "String()" << std::endl;
        // 如果地址不为空
        if (p != nullptr)
        {
            // strlen()方法计算地址指向字符数组有效长度，结尾还有'\0'，所以+1
            _data = new char[strlen(p) + 1];
            // 复制数据
            strcpy(_data, p);
        }
        // 如果地址为空
        else
        {
            // 开辟一个字节空间，值为'\0'
            _data = new char[1];
            *_data = '\0';
        }
    }
    // 拷贝构造
    String(const String &other)
    {
        std::cout << "String(other)" << std::endl;

        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
    }
    // 析构函数
    ~String()
    {
        std::cout << "~String()" << std::endl;

        delete[] _data;
        _data = nullptr;
    }
    // 赋值运算符重载
    String &operator=(const String &other)
    {
        std::cout << "operator=()" << std::endl;

        if (this == &other) { return *this; }
        delete[] _data;
        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
        return *this;
    }

    // 获取字符串长度,有效字符个数不包括'\0'
    int length() const { return strlen(_data); }
    // 返回字符串数组
    const char *c_str() const { return _data; }

private:
    char *_data;
    // 友元
};

// 全局函数
String getString(String &str)
{
    const char *str_ptr = str.c_str();
    String tmp_str(str_ptr); // String()
    return tmp_str; // String(other),调用拷贝构造函数在main函数栈帧构造临时对象，然后析构~String()
}

int main()
{
    String str1("Hello World!"); // String()
    String str2;                 // String()
    str2 = getString(str1);      // 临时对象给str2赋值，operator=()，~String()析构临时对象

    std::cout << str2.c_str() << std::endl;
}
```
