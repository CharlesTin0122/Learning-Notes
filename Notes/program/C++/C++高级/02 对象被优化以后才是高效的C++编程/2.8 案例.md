#include <cstddef>
#include <cstdlib>
#include <iostream>
#include <stdexcept>

// 实现容器的空间配置器，容器底层内存开辟释放，对象构造析构，都使用空间适配器来实现
template <typename T>
struct MyAllocator
{
    // 内存开辟
    T *allocate(size_t size)
    {
        // 使用malloc函数只开辟内存，不构造对象
        T *ptr = (T *)malloc(sizeof(T) * size);
        if (!ptr) { throw std::bad_alloc(); }
        return ptr;
    }
    // 内存释放
    void deallocate(T *ptr) { free(ptr); }
    // // 左值对象构造,在ptr地址构造值为val的对象
    // void construct(T *ptr, const T &val)
    // {
    //     // 定位new,在ptr地址构造值为val的对象，调用T类型对象的拷贝构造函数
    //     new (ptr) T(val);
    // }
    // // 右对象构造,在ptr地址构造值为val的对象
    // void construct(T *ptr, const T &&val)
    // {
    //     // 定位new,在ptr地址构造值为val的对象，调用T类型对象的拷贝构造函数
    //     // 右值引用本身是一个左值，想要调用T类型的右值拷贝构造函数，需要std::move(val)将其转化为右值
    //     new (ptr) T(std::move(val));
    // }

    template <typename Ty> // Ty 是一个模板参数，表示可以接收任何类型。这使得函数可以接受左值和右值。
    // Ty &&val 是一个转发引用（也称为完美转发），允许函数接受左值或右值，同时保持其原有的值类别（左值或右值）。
    void construct(T *ptr, Ty &&val)
    {
        // 定位new,在ptr地址构造值为val的对象，调用T类型对象的拷贝构造函数
        // std::forward类型完美转发，能够识别类型是左值还是右值
        new (ptr) T(std::forward<Ty>(val));
    }
    // 对象析构
    void destroy(T *ptr)
    {
        // 调用T类型的析构函数
        ptr->~T();
    }
};

// 实现vector类,ALLOC是非类型模板参数：空间配置器，意味着它在编译期必须是一个常量
template <typename T, typename ALLOC = MyAllocator<T>>
class MyVector
{
public:
    // 构造函数
    MyVector(int size = 10)
    {
        // 使用空间配置器开辟内存
        //_first = new T[size];
        _first = _allocator.allocate(size);
        _last = _first;
        _end = _first + size;
    }
    // 拷贝构造函数
    MyVector(const MyVector<T> &other)
    {
        // 计算参数对象长度
        int otherSize = other._end - other._first;
        // 分配新内存
        //_first = new T[otherSize];
        _first = _allocator.allocate(otherSize);
        // 拷贝数据
        int otherLen = other._last - other._first;
        for (int i = 0; i < otherLen; i++)
        {
            // 利用空间配置器构建对象
            //  this->_first[i] = other._first[i];
            _allocator.construct(_first + i, other._first[i]);
        }

        this->_last = this->_first + otherLen;
        this->_end = this->_first + otherSize;
    }
    // 赋值运算符重载
    MyVector<T> &operator=(const MyVector<T> &other)
    {
        // 防止自赋值
        if (this == &other) return *this;
        // 删除旧数据
        // delete[] _first;
        // 利用空间配置器析构有效元素
        for (T *p = _first; p != _last; ++p)
        {
            _allocator.destroy(p);
        }
        // 利用空间配置器释放堆上的数组内存
        _allocator.deallocate(_first);
        // 填充新数据
        // 计算参数对象长度
        int otherSize = other._end - other._first;
        // 分配新内存
        //_first = new T[otherSize];
        _first = _allocator.allocate(otherSize);
        // 拷贝数据
        int otherLen = other._last - other._first;
        for (int i = 0; i < otherLen; i++)
        {
            // 利用空间配置器构建对象
            //  this->_first[i] = other._first[i];
            _allocator.construct(_first + i, other._first[i]);
        }

        this->_last = this->_first + otherLen;
        this->_end = this->_first + otherSize;

        return *this;
    }
    ~MyVector()
    {
        // 利用空间配置器析构有效元素
        // delete[] _first;
        for (T *p = _first; p != _last; ++p)
        {
            _allocator.destroy(p);
        }
        // 利用空间配置器释放堆上的数组内存
        _allocator.deallocate(_first);
        _first = _last = _end = nullptr;
    }
    // //  添加左值元素
    // void push(const T &val)
    // {
    //     if (this->full()) this->expand();
    //     // 在_last地址构造一个值为val的对象
    //     //*_last++ = val;
    //     _allocator.construct(_last, val);
    //     _last++;
    // }
    // // 添加右值元素
    // void push(const T &&val)
    // {
    //     if (this->full()) this->expand();

    //     // 右值引用变量本身是一个左值，要想调用construct(T *ptr, const T &&val)，需要使用std::move(val)将其转化为右值
    //     _allocator.construct(_last, std::move(val));
    //     _last++;
    // }

    template <typename Ty> // Ty 是一个模板参数，表示可以接收任何类型。这使得函数可以接受左值和右值。
    // Ty &&val 是一个转发引用（也称为完美转发），允许函数接受左值或右值，同时保持其原有的值类别（左值或右值）。
    void push(Ty &&val)
    {
        if (full()) expand();
        // std::move移动语义，得到右值类型
        // std::forward类型完美转发，能够识别类型是左值还是右值
        /* std::forward<Ty>(val) 用于将 val 转发给构造函数。std::forward 可以根据 val
         * 的类型（左值或右值）选择适当的构造方式，从而调用正确的构造函数，确保效率和语义的正确性。
         */
        _allocator.construct(_last, std::forward<Ty>(val));
        _last++;
    }
    // 弹出元素
    void pop()
    {
        if (this->empty()) throw std::runtime_error("vector is empty!");
        // 析构要删除的元素，并且_last指针自减
        --_last;
        _allocator.destroy(_last);
    }
    // 返回末尾元素
    T back() const
    {
        if (this->empty()) throw std::runtime_error("vector is empty!");
        return *(_last - 1);
    }
    bool full() const { return _last == _end; }
    bool empty() const { return _last == _first; }
    int size() const { return _last - _first; }

    void print() const
    {
        for (int i = 0; i < this->size(); i++)
        {
            std::cout << _first[i] << " ";
        }
        std::cout << std::endl;
    }

private:
    T *_first;        // 指向数组起始位置
    T *_last;         // 指向数组中有效元素的后继位置
    T *_end;          // 指向数组空间的后继位置
    ALLOC _allocator; // 定义容器的空间配置器对象
    // 扩容
    void expand()
    {
        // 获取新数组容量和长度
        int newSize = (_end - _first) * 2;
        int len = _last - _first;
        // 申请新空间
        // T *newData = new T[newSize];
        T *newData = _allocator.allocate(newSize);
        // 赋值
        for (int i = 0; i < len; i++)
        {
            // newData[i] = _first[i]
            _allocator.construct(newData + i, _first[i]);
        }
        // delete[] _first;
        for (T *p = _first; p != _last; ++p)
        {
            _allocator.destroy(p);
        }
        _allocator.deallocate(_first);

        _first = newData;
        _last = _first + len;
        _end = _first + newSize;
    }
};

int main()
{
    MyVector<int> vec;

    vec.push(rand() % 100);
    std::cout << std::endl;
}
