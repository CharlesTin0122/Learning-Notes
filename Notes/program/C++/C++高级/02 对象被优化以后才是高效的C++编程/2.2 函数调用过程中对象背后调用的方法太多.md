```cpp
#include <iostream>

// 测试类
class Test
{
public:
    Test(int val = 5) : ma(val) { std::cout << "Test(int)" << std::endl; }
    Test(const Test &other) : ma(other.ma) { std::cout << "Test(other)" << std::endl; }
    Test &operator=(const Test &other)
    {
        std::cout << "operator=" << std::endl;
        ma = other.ma;
        return *this;
    }
    ~Test() { std::cout << "~Test()" << std::endl; }

    int getData() const { return ma; }

private:
    int ma;
};

Test getObj(Test t)
{
    int val = t.getData();
    Test tmp(val);
    return tmp;
}

// 主函数
int main()
{
    Test t1; // Test()
    Test t2; // Test()
    /* 函数调用中实参作为值传递时，实参到形参，是初始化过程，要调用拷贝构造函数。实参作为引用或指针传递时，是赋值过程。
     * 变量接收函数的返回值时，通常是初始化的过程，而不是赋值。
     * return tmp 要在main函数的栈帧上构建一个临时对象，然后析构tmp,析构形参t。getObj函数执行完毕
     * main函数栈帧上的临时对象给t2赋值，调用operator=，然后临时对象析构
     */
    t2 = getObj(t1); // Test(Other),Test(),Test(Other)，~Test(),~Test(),operator=，~Test(),

    return 0;
}
```
