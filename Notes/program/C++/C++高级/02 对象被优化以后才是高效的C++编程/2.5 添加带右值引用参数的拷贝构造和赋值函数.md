# 左值引用和右值引用
-  左值引用用来绑定左值。左值是指内存中的有明确地址的确定对象。
```cpp
//左值引用
int a = 10;
int &b = a;
//先由20生成临时量，再由常引用来引用临时量
//int tmp = 20; const int &c = tmp;
const int &c = 20;
```
- 右值引用用来绑定到右值。右值是指临时对象或字面量
```cpp
//右值引用引用字面量
//int tmp = 13; int &&d = tmp;
int &&d = 13;
//右值引用接收临时对象
string &&e = string("aaa");
```
- 一个右值引用变量本身是一个左值。内存中的有明确地址和名称。
# 带右值引用参数的拷贝构造和赋值函数
```cpp
#include <cstring>
#include <iostream>

// String类
class String
{
public:
    // 构造函数
    String(const char *p = nullptr)
    {
        std::cout << "String()" << std::endl;
        // 如果地址不为空
        if (p != nullptr)
        {
            // strlen()方法计算地址指向字符数组有效长度，结尾还有'\0'，所以+1
            _data = new char[strlen(p) + 1];
            // 复制数据
            strcpy(_data, p);
        }
        // 如果地址为空
        else
        {
            // 开辟一个字节空间，值为'\0'
            _data = new char[1];
            *_data = '\0';
        }
    }
    // 左值引用参数拷贝构造
    String(const String &other)
    {
        std::cout << "String(other&)" << std::endl;

        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
    }
    // 右值引用参数拷贝构造，other引用的是临时对象
    String(String &&other)
    {
        std::cout << "String(other&&)" << std::endl;
        _data = other._data;   // 接收对方指针
        other._data = nullptr; // 对方指针置空
    }
    // 析构函数
    ~String()
    {
        std::cout << "~String()" << std::endl;

        delete[] _data;
        _data = nullptr;
    }
    // 左值引用参数赋值运算符重载
    String &operator=(const String &other)
    {
        std::cout << "operator=(&)" << std::endl;

        if (this == &other) { return *this; }
        delete[] _data;
        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
        return *this;
    }
    // 右值引用参数的赋值运算符重载
    String &operator=(String &&other)
    {
        std::cout << "operator=(&&)" << std::endl;

        if (this == &other) { return *this; }
        delete[] _data;

        this->_data = other._data; // 接收对方指针
        other._data = nullptr;     // 对方指针置空
        return *this;
    }

    // 获取字符串长度,有效字符个数不包括'\0'
    int length() const { return strlen(_data); }
    // 返回字符串数组
    const char *c_str() const { return _data; }

private:
    char *_data;
    friend String operator+(const String &str1, const String &str2);
    friend std::ostream &operator<<(std::ostream &out, const String &str);
};
// 全局加法运算符重载,返回一个临时对象用于调用右值引用参数的拷贝构造函数
String operator+(const String &str1, const String &str2)
{
    std::cout << "operator+()" << std::endl;

    // 计算字符串相加后的长度
    int size = str1.length() + str2.length() + 1;
    // 构建对象
    String tmp;
    // 内存开辟空间
    tmp._data = new char[size];
    // 拷贝str1到新数据
    strcpy(tmp._data, str1._data);
    // 添加str2到新数据
    strcat(tmp._data, str2._data);
    // 返回加好的字符串对象
    return tmp; // tmp是一个临时对象要调用右值引用参数的拷贝构造函数在main函数栈帧上构造str3
}
// 输出运算符重载
std::ostream &operator<<(std::ostream &out, const String &str)
{
    std::cout << "operator<<()" << std::endl;
    out << str._data;
    return out;
}
// 全局函数
String getString(String &str)
{
    const char *str_ptr = str.c_str();
    String tmp_str(str_ptr); // String()
    return tmp_str; // 返回值tmp_str是一个临时对象，调用右值参数拷贝构造函数String(other&&),在main函数栈帧构造临时对象，然后析构~String()
}

int main()
{
    String str1 = "Hello ";
    String str2 = "World!";
    String str3 = str1 + str2;

    std::cout << str3 << std::endl;
}
```
