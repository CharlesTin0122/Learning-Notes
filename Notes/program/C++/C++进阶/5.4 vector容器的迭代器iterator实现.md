#include <cstddef>
#include <cstdlib>
#include <iostream>
#include <stdexcept>

// 实现容器的空间配置器，容器底层内存开辟释放，对象构造析构，都使用空间适配器来实现
template <typename T>
struct MyAllocator
{
    // 内存开辟
    T *allocate(size_t size)
    {
        // 使用malloc函数只开辟内存，不构造对象
        T *ptr = (T *)malloc(sizeof(T) * size);
        if (!ptr) { throw std::bad_alloc(); }
        return ptr;
    }
    // 内存释放
    void deallocate(T *ptr) { free(ptr); }
    // 对象构造,在ptr地址构造值为val的对象
    void construct(T *ptr, const T &val)
    {
        // 定位new,在ptr地址构造值为val的对象，调用T类型对象的拷贝构造函数
        new (ptr) T(val);
    }
    // 对象析构
    void destroy(T *ptr)
    {
        // 调用T类型的析构函数
        ptr->~T();
    }
};

// 实现vector类,ALLOC是非类型模板参数：空间配置器，意味着它在编译期必须是一个常量
template <typename T, typename ALLOC = MyAllocator<T>>
class MyVector
{
public:
    // 构造函数
    MyVector(int size = 10)
    {
        // 使用空间配置器开辟内存
        //_first = new T[size];
        _first = _allocator.allocate(size);
        _last = _first;
        _end = _first + size;
    }
    // 拷贝构造函数
    MyVector(const MyVector<T> &other)
    {
        // 计算参数对象长度
        int otherSize = other._end - other._first;
        // 分配新内存
        //_first = new T[otherSize];
        _first = _allocator.allocate(otherSize);
        // 拷贝数据
        int otherLen = other._last - other._first;
        for (int i = 0; i < otherLen; i++)
        {
            // 利用空间配置器构建对象
            //  this->_first[i] = other._first[i];
            _allocator.construct(_first + i, other._first[i]);
        }

        this->_last = this->_first + otherLen;
        this->_end = this->_first + otherSize;
    }
    // 析构函数
    ~MyVector()
    {
        // 利用空间配置器析构有效元素
        // delete[] _first;
        for (T *p = _first; p != _last; ++p)
        {
            _allocator.destroy(p);
        }
        // 利用空间配置器释放堆上的数组内存
        _allocator.deallocate(_first);
        _first = _last = _end = nullptr;
    }
    // 赋值运算符重载
    MyVector<T> &operator=(const MyVector<T> &other)
    {
        // 防止自赋值
        if (this == &other) return *this;
        // 删除旧数据
        // delete[] _first;
        // 利用空间配置器析构有效元素
        for (T *p = _first; p != _last; ++p)
        {
            _allocator.destroy(p);
        }
        // 利用空间配置器释放堆上的数组内存
        _allocator.deallocate(_first);
        // 填充新数据
        // 计算参数对象长度
        int otherSize = other._end - other._first;
        // 分配新内存
        //_first = new T[otherSize];
        _first = _allocator.allocate(otherSize);
        // 拷贝数据
        int otherLen = other._last - other._first;
        for (int i = 0; i < otherLen; i++)
        {
            // 利用空间配置器构建对象
            //  this->_first[i] = other._first[i];
            _allocator.construct(_first + i, other._first[i]);
        }

        this->_last = this->_first + otherLen;
        this->_end = this->_first + otherSize;

        return *this;
    }
    //[]运算符重载
    T &operator[](int index)
    {
        if (index < 0 || index >= size())
        {
            throw std::runtime_error("out of range index!");
        }
        return _first[index];
    }
    // 添加元素
    void push(const T &val)
    {
        if (this->full()) this->expand();
        // 在_last地址构造一个值为val的对象
        //*_last++ = val;
        _allocator.construct(_last, val);
        _last++;
    }
    // 弹出元素
    void pop()
    {
        if (this->empty()) throw std::runtime_error("vector is empty!");
        // 析构要删除的元素，并且_last指针自减
        --_last;
        _allocator.destroy(_last);
    }
    // 返回末尾元素
    T back() const
    {
        if (this->empty()) throw std::runtime_error("vector is empty!");
        return *(_last - 1);
    }
    bool full() const { return _last == _end; }
    bool empty() const { return _last == _first; }
    int size() const { return _last - _first; }

    void print() const
    {
        for (int i = 0; i < this->size(); i++)
        {
            std::cout << _first[i] << " ";
        }
        std::cout << std::endl;
    }
    // 迭代器实现为容器的嵌套类
    class iterator
    {
    public:
        // 构造函数，接收一个指针，生成迭代器对象
        iterator(T *ptr = nullptr) : _ptr(ptr) {}
        // 不等运算符!=重载
        bool operator!=(const iterator &it) const { return _ptr != it._ptr; }
        // 前置自增++a运算符重载，
        void operator++() { _ptr++; }
        // 解引用运算符*重载
        T &operator*() { return *_ptr; }
        const T &operator*() const { return *_ptr; }

    private:
        T *_ptr;
    };
    // 提供begin和end方法
    iterator begin() { return iterator(_first); }
    iterator end() { return iterator(_last); }

private:
    T *_first;        // 指向数组起始位置
    T *_last;         // 指向数组中有效元素的后继位置
    T *_end;          // 指向数组空间的后继位置
    ALLOC _allocator; // 定义容器的空间配置器对象
    // 扩容
    void expand()
    {
        // 获取新数组容量和长度
        int newSize = (_end - _first) * 2;
        int len = _last - _first;
        // 申请新空间
        // T *newData = new T[newSize];
        T *newData = _allocator.allocate(newSize);
        // 赋值
        for (int i = 0; i < len; i++)
        {
            // newData[i] = _first[i]
            _allocator.construct(newData + i, _first[i]);
        }
        // delete[] _first;
        for (T *p = _first; p != _last; ++p)
        {
            _allocator.destroy(p);
        }
        _allocator.deallocate(_first);

        _first = newData;
        _last = _first + len;
        _end = _first + newSize;
    }
};

int main()
{
    // 定义对象
    MyVector<int> vec;
    for (int i = 0; i < 20; i++)
    {
        vec.push(rand() % 100);
    }
    vec.print();
    // 随机访问
    for (int i = 0; i < vec.size(); ++i)
    {
        std::cout << vec[i] << " ";
    }
    std::cout << std::endl;
    // 迭代器使用
    auto it = vec.begin();
    for (; it != vec.end(); ++it)
    {
        std::cout << *it << " ";
    }
    std::cout << std::endl;
    // foreach遍历
    for (const int &val : vec)
    {
        std::cout << val << " ";
    }
    std::cout << std::endl;
}
