# 顺序栈案例
```cpp
#include <cstdlib>
#include <iostream>
#include <stdexcept>

// 类模板，顺序栈
template <typename T = int>
class MyStack // 模板名称
{
public:
    MyStack(int size = 10) : _data(new T[size]), _top(0), _size(size) {}
    MyStack(const MyStack<T> &other) : _top(other._top), _size(other._size)
    {
        _data = new T[_size];
        for (int i = 0; i < _top; i++)
        {
            _data[i] = other._data[i];
        }
    }
    MyStack &operator=(const MyStack<T> &other)
    {
        // 防止自赋值
        if (this == &other) return *this;

        // 先分配新数据，再删除旧数据
        T *newData = new T[other._size];
        for (int i = 0; i < other._top; i++)
        {
            newData[i] = other._data[i];
        }

        delete[] _data;
        _data = newData;

        _top = other._top;
        _size = other._size;

        return *this;
    }
    ~MyStack() { delete[] _data; }

    void push(const T &val)
    {
        if (this->full()) this->expand();

        _data[_top++] = val;
    }
    void pop()
    {
        if (this->empty()) throw std::runtime_error("Stack is empty!");
        _top--;
    }
    T top() const
    {
        if (this->empty()) throw std::runtime_error("Stack is empty!");
        return _data[_top - 1];
    }
    bool full() const { return _top == _size; }
    bool empty() const { return _top == 0; }
    void print()
    {
        for (int i = 0; i < _top; i++)
        {
            std::cout << _data[i] << " ";
        }
        std::cout << std::endl;
    }

private:
    T *_data;
    int _top;
    int _size;

    void expand()
    {
        T *newData = new T[_size * 2];
        for (int i = 0; i < _top; i++)
        {
            newData[i] = _data[i];
        }

        delete[] _data;
        _data = newData;

        _size *= 2;
    }
};

int main()
{
    MyStack<> stack;
    for (int i = 0; i < 10; i++)
    {
        stack.push(rand() % 100);
    }
    stack.print();
    return 0;
}

```
# 要点
- 类模板经过实例化得到模板类，模板类实例化得到对象。
- 类模板的选择性实例化：实例化对象调用了哪些方法，在实例化的时候才会实例化这些方法，没有用到的方法不会实例化。
- `template <typename T = int>`类模板的默认类型为int，调用时如果是int类型可省略参数`MyStack<> stack;`
