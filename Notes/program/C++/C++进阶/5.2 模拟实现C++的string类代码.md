#include <cstring>
#include <iostream>
#include <ostream>
// String类
class String
{
public:
    // 构造函数
    String(const char *p = nullptr)
    {
        // 如果地址不为空
        if (p != nullptr)
        {
            // strlen()方法计算地址指向字符数组有效长度，结尾还有'\0'，所以+1
            _data = new char[strlen(p) + 1];
            // 复制数据
            strcpy(_data, p);
        }
        // 如果地址为空
        else
        {
            // 开辟一个字节空间，值为'\0'
            _data = new char[1];
            *_data = '\0';
        }
    }
    // 拷贝构造
    String(const String &other)
    {
        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
    }
    // 析构函数
    ~String()
    {
        delete[] _data;
        _data = nullptr;
    }
    // 赋值运算符重载
    String &operator=(const String &other)
    {
        if (this == &other) { return *this; }
        delete[] _data;
        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
        return *this;
    }
    // 比较运算符重载,strcmp()函数
    bool operator>(const String &str) const
    {
        return strcmp(this->_data, str._data) > 0;
    }
    bool operator<(const String &str) const
    {
        return strcmp(this->_data, str._data) < 0;
    }
    bool operator==(const String &str) const
    {
        return strcmp(this->_data, str._data) == 0;
    }
    //[]运算符重载
    char &operator[](int index) { return _data[index]; }
    const char &operator[](int index) const { return _data[index]; }
    // 获取字符串长度,有效字符个数不包括'\0'
    int length() const { return strlen(_data); }
    // 返回字符串数组
    const char *c_str() const { return _data; }

private:
    char *_data;
    // 友元
    friend String operator+(const String &str1, const String &str2);
    friend std::ostream &operator<<(std::ostream &out, const String &str);
};
// 全局加法运算符重载
String operator+(const String &str1, const String &str2)
{
    // 计算字符串相加后的长度
    int size = str1.length() + str2.length() + 1;
    // 构建对象
    String tmp;
    // 内存开辟空间
    tmp._data = new char[size];
    // 拷贝str1到新数据
    strcpy(tmp._data, str1._data);
    // 添加str2到新数据
    strcat(tmp._data, str2._data);
    // 返回加好的字符串对象
    return tmp;
}
// 输出运算符重载
std::ostream &operator<<(std::ostream &out, const String &str)
{
    out << str._data;
    return out;
}
int main()
{
    // 加法运算符
    String str1;
    String str2 = "aaa";
    String str3 = "bbb";
    String str4 = str2 + str3;
    String str5 = str2 + "ccc";
    String str6 = "ddd" + str2;
    // 输出运算符
    std::cout << str6 << std::endl;
    // 比较运算符
    if (str5 > str6) { std::cout << "str5 > str6" << std::endl; }
    else { std::cout << "str5 < str6" << std::endl; }
    //[]运算符
    int len = str6.length();
    for (int i = 0; i < len; i++)
    {
        std::cout << str6[i] << " ";
    }
    std::cout << std::endl;

    // String -> char*类型转化
    char buffer[1024] = {0};
    strcpy(buffer, str6.c_str());
    std::cout << buffer << std::endl;
}
