# 容器迭代器
```cpp
    std::string str = "Hello World!";
    std::string::iterator it = str.begin();
    for (; it != str.end(); ++it)
    {
        std::cout << *it << std::endl;
    }
```
- 功能：提供一种统一的方式来透明的遍历容器，迭代器可以透明的访问容器内部元素的值，不需要知道容器底层的数据结构。
- 容器迭代器是容器类的一个嵌套类，`std::string::iterator`，每种容器都有自己的迭代器
- 对象str有一个`begin()`方法，返回的是底层数据的首元素位置迭代器表示；
- 对象str有一个`end()`方法，返回的是底层数据的末尾元素后继位置迭代器表示；
- 迭代器对象 `it`用于接收迭代器，`++it`表示将迭代器位置后移一位，`it != str.end()`,便可遍历所有元素。
- `*it`将迭代器解引用，获取容器迭代器位置的元素。
- STL的泛型算法algorithm是一组全局的函数，可以给所有容器使用，泛型算法接受的参数基本都是迭代器，能够统一遍历所有容器的元素。

# string类型的迭代器实现
```cpp
#include <cstring>
#include <iostream>
#include <ostream>
// String类
class String
{
public:
    // 构造函数
    String(const char *p = nullptr)
    {
        // 如果地址不为空
        if (p != nullptr)
        {
            // strlen()方法计算地址指向字符数组有效长度，结尾还有'\0'，所以+1
            _data = new char[strlen(p) + 1];
            // 复制数据
            strcpy(_data, p);
        }
        // 如果地址为空
        else
        {
            // 开辟一个字节空间，值为'\0'
            _data = new char[1];
            *_data = '\0';
        }
    }
    // 拷贝构造
    String(const String &other)
    {
        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
    }
    // 析构函数
    ~String()
    {
        delete[] _data;
        _data = nullptr;
    }
    // 赋值运算符重载
    String &operator=(const String &other)
    {
        if (this == &other) { return *this; }
        delete[] _data;
        _data = new char[strlen(other._data) + 1];
        strcpy(_data, other._data);
        return *this;
    }
    // 比较运算符重载,strcmp()函数
    bool operator>(const String &str) const
    {
        return strcmp(this->_data, str._data) > 0;
    }
    bool operator<(const String &str) const
    {
        return strcmp(this->_data, str._data) < 0;
    }
    bool operator==(const String &str) const
    {
        return strcmp(this->_data, str._data) == 0;
    }
    //[]运算符重载
    char &operator[](int index) { return _data[index]; }
    const char &operator[](int index) const { return _data[index]; }
    // 获取字符串长度,有效字符个数不包括'\0'
    int length() const { return strlen(_data); }
    // 返回字符串数组
    const char *c_str() const { return _data; }
    // 迭代器嵌套类
    class iterator
    {
    public:
        // 构造函数，接收一个指针
        iterator(char *ptr = nullptr) : _iter(ptr) {}
        // 不等运算符!=重载
        bool operator!=(const iterator &it) { return _iter != it._iter; }
        // 前置自增++a运算符重载
        void operator++() { ++_iter; }
        // 解引用*运算符重载
        char &operator*() { return *_iter; }

    private:
        // 迭代器底层用指针实现，也可以使用索引实现
        char *_iter;
    };
    // 提供begin和end方法，返回类型是iterator迭代器，是容器的方法，不是迭代器的方法

    // 返回的是底层数据的首元素位置迭代器表示
    iterator begin() { return iterator(_data); }
    // 返回的是底层数据的末尾元素后继位置迭代器表示,指向'\0'位置
    iterator end() { return iterator(_data + length()); }

private:
    char *_data;
    // 友元
    friend String operator+(const String &str1, const String &str2);
    friend std::ostream &operator<<(std::ostream &out, const String &str);
};
// 全局加法运算符重载
String operator+(const String &str1, const String &str2)
{
    // 计算字符串相加后的长度
    int size = str1.length() + str2.length() + 1;
    // 构建对象
    String tmp;
    // 内存开辟空间
    tmp._data = new char[size];
    // 拷贝str1到新数据
    strcpy(tmp._data, str1._data);
    // 添加str2到新数据
    strcat(tmp._data, str2._data);
    // 返回加好的字符串对象
    return tmp;
}
// 输出运算符重载
std::ostream &operator<<(std::ostream &out, const String &str)
{
    out << str._data;
    return out;
}
int main()
{

    // String容器迭代器
    String str = "Hello World!";
    String::iterator it = str.begin();
    for (; it != str.end(); ++it)
    {
        std::cout << *it << " ";
    }
    std::cout << std::endl;
    // C++11，foreach的方式遍历容器内部元素的值
    // 底层是通过迭代器进行遍历的，只有类中包含迭代器，begin和end方法才可以使用
    for (const char ch : str)
    {
        std::cout << ch << " ";
    }
    std::cout << std::endl;
}
```
