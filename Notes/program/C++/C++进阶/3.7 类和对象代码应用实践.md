# 编写类string的构造函数，析构函数和赋值函数
```cpp
#include <cstring>
#include <iostream>

class MyString
{
public:
    // 构造函数
    MyString(const char *str = nullptr)
    {
        if (str != nullptr)
        {
            // C 风格字符串末尾为空字符'\0'，对应的就是整数0，所以要增加一个字节
            _data = new char[strlen(str) + 1];
            strcpy(this->_data, str);
        }
        else
        {
            // 空字符串，空字符'\0'占用1字节
            _data = new char[1];
            *_data = '\0';
        }
    }
    // 拷贝构造函数
    MyString(const MyString &other)
    {
        _data = new char[strlen(other._data) + 1];
        strcpy(this->_data, other._data);
    }
    // 析构函数
    ~MyString()
    {
        delete[] _data;
        _data = nullptr;
    }
    // 重载=运算符,返回值为MyString类型的引用,为了连续赋值
    MyString &operator=(const MyString &other)
    {
        // 如果当前对象地址与传入对象相同，返回当前对象
        if (this == &other) { return *this; }
        // 删除旧数据
        delete[] _data;
        // 分配内存储存新数据
        _data = new char[strlen(other._data) + 1];
        strcpy(this->_data, other._data);
        // 返回当前对象
        return *this;
    }
    void print() { std::cout << _data << std::endl; }

private:
    char *_data;
};

int main()
{
    // 调用构造函数
    MyString str1;
    MyString str2("Hello");
    MyString str3 = "World";
    // 调用拷贝构造函数
    MyString str4(str2);
    // 调用赋值重载函数
    MyString str5 = str3;

    str4.print();
    str5.print();
    // 连续调用调用赋值重载函数
    str1 = str2 = str3;
    str1.print();
    str2.print();
    str3.print();

    return 0;
}
```
# 编写循环队列
```cpp
#include <cstdlib>
#include <iostream>
#include <stdexcept>

// 循环队列
class MyQueue
{
public:
    // 构造函数
    MyQueue(int size = 5)
        : _front(0), _rear(0), _size(size > 0 ? size : 5),
          _data(new int[size > 0 ? size : 5])
    {
    }

    // 拷贝构造函数
    MyQueue(const MyQueue &other)
        : _data(new int[other._size]), _front(other._front), _rear(other._rear),
          _size(other._size)
    {
        copy_data_from(other);
    }

    // 重载赋值运算符
    MyQueue &operator=(const MyQueue &other)
    {
        if (this == &other) return *this;
        delete[] _data;
        _data = new int[other._size];
        _front = other._front;
        _rear = other._rear;
        _size = other._size;
        copy_data_from(other);
        return *this;
    }

    // 析构函数
    ~MyQueue() { delete[] _data; }

    // 添加数据
    void push(int val)
    {
        if (full()) resize();
        _data[_rear] = val;
        _rear = (_rear + 1) % _size;
    }

    // 弹出数据
    void pop()
    {
        if (empty()) throw std::runtime_error("Queue is empty, cannot pop.");
        _front = (_front + 1) % _size;
    }

    // 队首元素
    int front() const
    {
        if (empty())
        {
            throw std::runtime_error(
                "Queue is empty, cannot access front element.");
        }
        return _data[_front];
    }

    // 是否已满
    bool full() const { return (_rear + 1) % _size == _front; }

    // 是否为空
    bool empty() const { return _front == _rear; }

    // 打印元素
    void print() const
    {
        int index = _front;
        while (index != _rear)
        {
            std::cout << _data[index] << " ";
            index = (index + 1) % _size;
        }
        std::cout << std::endl;
    }

private:
    int *_data;
    int _front;
    int _rear;
    int _size;

    // 重订大小
    void resize()
    {
        int newSize = _size * 2;
        int *newData = new int[newSize];

        int count = size(); // 当前有效元素个数
        for (int i = 0; i < count; ++i)
        {
            newData[i] = _data[(_front + i) % _size];
        }

        delete[] _data;
        _data = newData;
        _front = 0;
        _rear = count;
        _size = newSize;
    }

    // 拷贝数据
    void copy_data_from(const MyQueue &other)
    {
        int count = other.size();
        for (int i = 0; i < count; ++i)
        {
            _data[i] = other._data[(other._front + i) % other._size];
        }
        _rear = count;
    }

    // 计算当前队列中的有效元素个数
    int size() const { return (_rear + _size - _front) % _size; }
};
```
