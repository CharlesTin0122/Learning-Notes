# 概念
底层是红黑树实现，增删查$O(log_2^n)$
- set：集合
- multiset：多重集合
- map：映射
- multimap：多重映射
# set案例
- 自定义对象填充set
```cpp
#include <iostream>
#include <set>
#include <string>

// 学生类
class Students
{
public:
    Students(int id, std::string name) : _id(id), _name(name) {}
    // 小于运算符重载，常方法
    bool operator<(const Students &stu) const { return this->_id < stu._id; }

private:
    int _id;
    std::string _name;
    friend std::ostream &operator<<(std::ostream &out, const Students &stu);
};
// 输出运算符重载函数,因为对象是个右值，只能提供全局运算符重载
std::ostream &operator<<(std::ostream &out, const Students &stu)
{
    out << "id: " << stu._id << " name: " << stu._name << std::endl;
    return out;
}
// 主函数
int main()
{
    // 向一个有序数据结构存储自定义类型要提供小于运算符重载
    std::set<Students> set;
    set.insert(Students(1001, "ZhangSan"));
    set.insert(Students(1002, "LiSi"));

    for (const Students &stu : set)
    {
        std::cout << stu << std::endl;
    }
}
```
# map案例
```cpp
#include <iostream>
#include <map>
#include <string>
#include <utility>

// 学生类
class Students
{
public:
    Students(int id = 0, std::string name = "") : _id(id), _name(name) {}

private:
    int _id;
    std::string _name;
    friend std::ostream &operator<<(std::ostream &out, const Students &stu);
};
// 输出运算符重载函数,因为对象是个右值，只能提供全局运算符重载
std::ostream &operator<<(std::ostream &out, const Students &stu)
{
    out << "id: " << stu._id << " name: " << stu._name << std::endl;
    return out;
}
// 主函数
int main()
{
    std::map<int, Students> map;
    map.insert({1001, Students(1001, "ZhangSan")});
    map.insert({1002, Students(1002, "LiSi")});
    map.insert({1003, Students(1003, "WangWu")});

    map.erase(1001);

    std::cout << map[1002] << std::endl;

    for (const std::pair<int, Students> &p : map)
    {
        std::cout << p.first << " : " << p.second << std::endl;
    }
}
```
