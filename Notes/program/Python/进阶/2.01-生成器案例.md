# maya 案例
- 遍历大量节点
```python
import maya.cmds as cmds

def transform_generator():
    """生成器：逐个产生场景中的变换节点"""
    all_transforms = cmds.ls(type='transform')  # 获取所有变换节点
    for node in all_transforms:
        if cmds.objExists(node):  # 过滤不存在的节点
            yield node  # 暂停并产生一个节点

# 使用生成器
for transform in transform_generator():
    print(f"处理节点: {transform}")
    # 示例操作：打印其位置
    pos = cmds.xform(transform, q=True, t=True)
    print(f"  位置: {pos}")
```
- 遍历网格顶点
```python
import maya.cmds as cmds
import random

def vertex_generator(mesh):
    """生成器：逐个产生网格顶点及其索引"""
    vertex_count = cmds.polyEvaluate(mesh, vertex=True)
    for i in range(vertex_count):
        pos = cmds.pointPosition(f"{mesh}.vtx[{i}]", world=True)
        yield i, pos  # 产生索引和位置

# 使用生成器（假设场景有 'pSphere1'）
mesh = 'pSphere1'
for idx, pos in vertex_generator(mesh):
    # 随机偏移 Z 坐标
    new_z = pos[2] + random.uniform(-0.1, 0.1)
    cmds.xform(f"{mesh}.vtx[{idx}]", t=[0, 0, new_z], ws=True)
    print(f"顶点 {idx} 新 Z: {new_z}")
```
- 动画关键帧生成
```python
import maya.cmds as cmds
import math

def keyframe_generator(obj, start_frame, end_frame, attribute='translateX'):
    """生成器：逐帧产生正弦波关键帧值"""
    for frame in range(start_frame, end_frame + 1):
        cmds.currentTime(frame)  # 设置当前帧
        value = math.sin(frame * 0.1) * 5  # 正弦波幅度 5
        yield frame, value
        # 设置关键帧（可选，在循环外处理）
        cmds.setKeyframe(obj, attribute=attribute, value=value, time=frame)

# 使用生成器（假设对象 'ball'）
obj = cmds.polySphere(name='ball')[0]
for frame, value in keyframe_generator(obj, 1, 100, 'translateX'):
    print(f"帧 {frame}: {attribute} = {value}")
    # 上面的 yield 已设置关键帧
```
- 批处理文件
```python
import maya.cmds as cmds
import os

def file_importer(directory):
    """生成器：逐个产生并导入 OBJ 文件"""
    for filename in os.listdir(directory):
        if filename.endswith('.obj'):
            filepath = os.path.join(directory, filename)
            imported = cmds.file(filepath, i=True, type='OBJ', ignoreVersion=True)
            yield imported[0]  # 产生导入的节点

# 使用生成器（假设目录 'C:/assets/' 有 OBJ 文件）
for imported_node in file_importer('C:/assets/'):
    print(f"导入: {imported_node}")
    # 示例：缩放导入对象
    cmds.scale(0.5, 0.5, 0.5, imported_node)
```
# pymel结合生成器高级案例
在 **PyMEL** 中，生成器（generators）结合其 **面向对象（OO）** 设计和 **自动类型转换** 特性，可以实现非常**高级、优雅且高效**的 Maya 场景操作。

相比 `maya.cmds`，PyMEL 的对象（如 `pymel.core.PyNode`）支持链式调用、属性访问、上下文管理等，配合生成器可构建**流式数据管道（streaming pipeline）**，特别适合处理大型场景、复杂动画、批量资产处理等生产级任务。

### 1. **链式过滤 + 生成器：智能节点流（Node Stream）**

**场景**：只处理可见的、未锁定的、选中层的 `mesh` 节点。

```python
def visible_mesh_stream():
    """生成器：流式产生符合条件的 mesh 节点"""
    for obj in pm.ls(type='mesh', long=True):
        transform = obj.getParent()
        if (transform.visibility.get() and 
            not transform.isLocked() and 
            transform.layer.get() in pm.ls(selection=True)):
            yield transform

# 使用：流式处理
for mesh in visible_mesh_stream():
    print(f"处理: {mesh}")
    mesh.scale.set([1.1, 1.1, 1.1])  # 放大 10%
```

**优势**：
- 惰性加载：10万网格也只占一个对象内存
- 可组合：可嵌套多个生成器
- 支持 `break` 中断，节省时间

### 2. **`yield from` + 递归遍历层级结构（Hierarchy Traversal）**

**场景**：递归遍历组节点，生成所有子孙 `transform`。

```python
def walk_hierarchy(root):
    """生成器：深度优先遍历所有子节点"""
    if not root:
        return
    yield root
    for child in root.getChildren(type='transform'):
        yield from walk_hierarchy(child)  # Python 3.3+ 语法

# 使用
root_grp = pm.PyNode('character_GRP')
for node in walk_hierarchy(root_grp):
    print(node)
```

**高级技巧**：结合 `send()` 实现**可控遍历**（交互式选择）

```python
def controllable_walk(root):
    node = root
    while node:
        action = yield node  # 暂停，等待外部 send 指令
        if action == 'skip':
            continue
        children = node.getChildren(type='transform')
        node = children[0] if children else None

# 使用
gen = controllable_walk(pm.PyNode('root'))
first = next(gen)
print("当前:", first)

# 跳过当前子树
gen.send('skip')

# 继续
next(gen)
```

### 3. **生成器 + 上下文管理：安全批量操作（Undo Chunk）**

**场景**：批量修改 1000 个对象，需支持 **单次 Undo**

```python
from contextlib import contextmanager

@contextmanager
def undo_chunk():
    pm.undoInfo(openChunk=True)
    try:
        yield
    finally:
        pm.undoInfo(closeChunk=True)

def batch_scale_generator(objects, scale_factor):
    for obj in objects:
        obj.scale.set(scale_factor)
        yield obj  # 边改边产出

# 使用
with undo_chunk():
    for scaled_obj in batch_scale_generator(pm.ls(type='transform')[:1000], [2,2,2]):
        pass  # 所有操作在一个 undo 中
```

### 4. **生成器管道（Generator Pipeline）：多阶段处理**

**场景**：筛选 → 变形 → 导出 UV 快照

```python
def mesh_stream():
    for m in pm.ls(type='mesh'):
        yield m.getParent()

def deform_stream(source):
    for obj in source:
        # 随机噪声变形
        for vtx in obj.vtx:
            pos = vtx.getPosition(space='world')
            pos.y += pm.util.noise(pos.x * 0.1) * 0.5
            vtx.setPosition(pos, space='world')
        yield obj

def export_uv_stream(source):
    for obj in source:
        uv_snap = pm.polyUVSet(obj, copy=True, uvSetName='temp_snap')
        pm.select(obj)
        pm.runtime.UVSnapshot(
            file=f"/tmp/{obj}.png",
            xResolution=1024, yResolution=1024
        )
        yield obj

# 构建管道
pipeline = export_uv_stream(
    deform_stream(
        mesh_stream()
    )
)

# 执行
for result in pipeline:
    print(f"完成: {result}")
```

**优势**：模块化、可测试、可重用、内存极省。

### 5. **生成器 + `send()`：实时交互工具（Interactive Tool）**

**场景**：用户逐个确认是否删除空组

```python
def confirm_delete_empty_groups():
    for grp in pm.ls(type='transform'):
        if not grp.getChildren() and grp.getShapes() == []:
            confirm = yield grp  # 暂停，等待用户决定
            if confirm is True:
                pm.delete(grp)
                print(f"已删除: {grp}")
            elif confirm is False:
                print(f"保留: {grp}")
            # confirm 为 None 时继续询问

# 使用
gen = confirm_delete_empty_groups()
first = next(gen)
print(f"发现空组: {first}，删除？")

# 模拟用户输入
gen.send(True)   # 删除
gen.send(False)  # 保留
```

**可扩展为 GUI 工具**（Qt/PySide）：
```python
# 在按钮点击时：
gen.send(True)
```

### 6. **生成器 + `pymel` 属性流：动画曲线采样**

**场景**：对动画曲线采样，生成每帧值（支持插值）

```python
def anim_curve_sampler(curve, start, end, step=1):
    """生成器：逐帧采样动画曲线"""
    for frame in range(int(start), int(end) + 1, step):
        value = curve.eval(frame)
        yield frame, value

# 使用
curve = pm.PyNode('pCube1.translateX').inputs()[0]  # 假设有动画曲线
for frame, val in anim_curve_sampler(curve, 1, 100):
    print(f"Frame {frame}: {val}")
```

### 7. **生成器 + `itertools`：高级组合操作**

```python
import itertools

def selected_meshes():
    return (s.getParent() for s in pm.ls(selection=True, type='mesh'))

def locked_meshes():
    return (m for m in pm.ls(type='transform') if m.isLocked())

# 合并流
all_problematic = itertools.chain(selected_meshes(), locked_meshes())

# 去重 + 排序
unique_sorted = sorted(set(all_problematic), key=lambda x: x.name())

for obj in unique_sorted:
    print(obj)
```

## 推荐工具函数（复制粘贴用）

```python
def pm_ls_generator(**kwargs):
    """PyMEL 版 ls 生成器（惰性）"""
    for node in pm.ls(**kwargs):
        yield node

def pm_children_generator(parent, type=None):
    """递归子节点生成器"""
    for child in parent.getChildren(type=type):
        yield child
        yield from pm_children_generator(child, type=type)
```
