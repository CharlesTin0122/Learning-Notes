# 特点
- 每一个节点除了数据域，还有两个指针域，next指针域指向下个节点，pre指针域指向前一个节点
- 头节点的pre指针指向末尾节点，末尾节点的next指针指向头节点

# 代码
```cpp
#include <iostream>
// 定义链表节点
struct Node
{
    int data_; // 数据域
    // 定义前后指针域
    Node *pre_;
    Node *next_;
    // 构造函数
    Node(int data = 0) : data_(data),
                         pre_(nullptr),
                         next_(nullptr)
    {
    }
};
// 定义双向循环链表类DoublyCircularLinkedList
class DoublyCircularLinkedList
{
private:
    Node *head_; // 指向头节点
public:
    // 构造函数
    DoublyCircularLinkedList()
    {
        head_ = new Node(); // 构建头节点
        // 头节点前后地址域指向自身
        head_->pre_ = head_;
        head_->next_ = head_;
    }
    // 析构函数
    ~DoublyCircularLinkedList()
    {
        Node *cur = head_->next_; // 定义游标节点
        // 遍历链表,直到节点的地址域指向头节点跳出循环
        while (cur->next_ != head_)
        {
            // 逐个删除节点
            head_->next_ = cur->next_; // 头节点的后继节点指向当前节点的后继节点
            cur->next_->pre_ = head_;  // 当前节点的后继节点的前指针域指向头节点

            delete cur;         // 删除节点
            cur = head_->next_; // 后移游标节点
        }
        // 最终删除头节点
        delete head_;
        head_ = nullptr;
    }
    // 头插法
    void insertHead(int val)
    {
        // 根据参数构造新节点
        Node *node = new Node(val);

        // 插入节点
        node->next_ = head_->next_; // node的后指针域指向头节点的后继节点，如果头节点无后继节点，则为nullptr
        node->pre_ = head_;         // node的前指针域指向头节点
        head_->next_->pre_ = node;  // 第二个节点的前指针域指向node
        head_->next_ = node;        // 头节点的后指针域指向node
    }
    // 尾插法
    void insertTail(int val)
    {
        // 根据参数构造新节点
        Node *node = new Node(val);
        // 定义游标节点，用于找到尾节点
        Node *tail = head_->pre_;
        // 执行尾插
        node->pre_ = tail;
        tail->next_ = node;
        // 链接首尾节点
        node->next_ = head_;
        head_->pre_ = node;
    }
    // 查找节点
    bool find(int val) const
    {
        Node *cur = head_->next_;
        while (cur != head_)
        {
            if (cur->data_ == val)
            {
                return true;
            }
            else
            {
                cur = cur->next_;
            }
        }
        return false;
    }
    // 删除节点
    void remove(int val)
    {
        // 定义游标节点
        Node *cur = head_->next_;
        // 遍历链表
        while (cur != head_)
        {
            // 如果节点数据域分与val值，执行删除
            if (cur->data_ == val)
            {
                // 要删除节点的前趋节点的后指针域指向要删除节点的后继节点
                cur->pre_->next_ = cur->next_;
                // 要删除节点的后继节点的前指针域指向要删除节点的前趋节点
                cur->next_->pre_ = cur->pre_;
                // 删除节点
                delete cur;
                return;
            }
            else
            {
                cur = cur->next_;
            }
        }
    }
    // 输出函数
    void show() const
    {
        Node *cur = head_->next_;
        while (cur != head_)
        {
            std::cout << cur->data_ << " ";
            cur = cur->next_;
        }
        std::cout << std::endl;
    }
};
int main()
{
    DoublyCircularLinkedList list;
    list.insertHead(666);
    list.insertTail(10);
    list.insertTail(20);
    list.insertTail(30);
    list.insertTail(40);
    list.insertTail(50);
    list.insertHead(50);

    list.show();

    list.remove(50);
    list.show();

    return 0;
}
```
