# 特点
- 每个节点除了数据域，还有一个指针域指向下个节点
- 末尾节点指向头节点

# 代码
```cpp
#include <iostream>

// 定义节点结构体
struct Node
{
    int data_;   // 数据域
    Node *next_; // 指针域
    // 节点构造函数
    Node(int data = 0) : data_(data), next_(nullptr) {}
};
// 定义单向循环链表类CircularLinkedList
class CirLink
{
private:
    Node *head_; // 头节点
    Node *tail_; // 尾节点
public:
    // 链表构造函数，构造一个带有虚拟头节点的单向循环链表
    CirLink()
    {
        // 三个指针指向头节点，头、尾、自身地址域
        head_ = new Node();   // 生成头节点
        tail_ = head_;        // 尾节点也指向头节点
        head_->next_ = head_; // 头节点的地址域也指向头节点(自己指向自己)
    }
    // 析构函数
    ~CirLink()
    {
        // 定义游标指针指向第一个节点
        Node *p = head_->next_;
        // 遍历循环链表，直到游标节点指向头节点跳出
        while (p != head_)
        {
            head_->next_ = p->next_; // 头节点地址域指向第二个节点，把第一个节点剥离出来
            delete p;                // 清除第一个节点
            p = head_->next_;        // 游标指针指向新的第一个节点
        }
        // 跳出循环后删除虚拟头节点
        delete head_;
    }
    // 尾插法,因为设置了尾节点指针，所以尾插节点不必遍历链表，所以时间复杂度为O(1)
    void indertTail(int val)
    {
        // 根据参数值构造新节点
        Node *new_node = new Node(val);
        // 尾插节点
        new_node->next_ = head_; // 新节点的地址域指向头节点
        tail_->next_ = new_node; // 尾节点的地址域指向新节点
        tail_ = new_node;        // 更新尾节点指针指向新节点
    }
    // 头插法
    void insertHead(int val)
    {
        // 根据参数值构造新节点
        Node *new_node = new Node(val);
        // 头插节点
        new_node->next_ = head_->next_; // 新节点地址域指向第一个节点
        head_->next_ = new_node;        // 头节点指向新节点
        // 如果链表中只包含一个我们插入的节点，那么此节点是尾节点
        if (new_node->next_ == head_)
        {
            tail_ = new_node;
        }
    }
    // 删除节点
    void remove(int val)
    {
        // 定义双指针
        Node *prev = head_;
        Node *curr = head_->next_;
        // 遍历链表，当游标节点移动到头节点时跳出
        while (curr != head_)
        {
            // 当游标节点的值等于val时
            if (curr->data_ == val)
            {
                // 执行删除节点操作
                prev->next_ = curr->next_; // 要删除节点的前趋地址域指向要删除节点的后继
                delete curr;               // 删除节点
                // 如果移除的时尾节点，重定义尾节点
                if (prev->next_ == head_)
                {
                    tail_ = prev;
                }
                return;
            }
            else
            {
                // 后移双指针
                prev = prev->next_;
                curr = curr->next_;
            }
        }
    }
    // 查询
    Node *find(int val) const
    {
        // 定义游标节点
        Node *curr = head_->next_;
        // 遍历链表
        while (curr != head_)
        {
            // 如果游标节点的值等于参数值，返回该节点
            if (curr->data_ == val)
            {
                return curr;
            }
            // 推进游标
            curr = curr->next_;
        }
        return nullptr;
    }
    // 修改
    void update(int ori_val, int rep_val)
    {
        Node *node = find(ori_val); // 找到该节点
        node->data_ = rep_val;      // 修改该节点
    }
    // 输出
    void show() const
    {
        // 定义游标节点
        Node *curr = head_->next_;
        // 遍历链表
        while (curr != head_)
        {
            std::cout << curr->data_ << " ";
            curr = curr->next_;
        }
        std::cout << std::endl;
    }
};
// 主函数
int main()
{
    // 实例化链表
    CirLink clink;
    // 测试插入
    clink.insertHead(0);
    clink.indertTail(10);
    clink.indertTail(20);
    clink.indertTail(30);
    clink.indertTail(40);
    clink.indertTail(50);
    clink.insertHead(100);
    clink.show();
    // 测试移除
    clink.remove(20);
    clink.show();
    // 查找测试
    Node *ret = clink.find(30);
    if (ret)
    {
        std::cout << "Find it" << std::endl;
    }
    else
    {
        std::cout << "Not find it" << std::endl;
    }
    // 测试替换
    clink.update(40, 999);
    clink.show();
}
```

# 案例
## 约瑟夫环问题
### 问题描述
[LCR 187. 破冰游戏](https://leetcode.cn/problems/yuan-quan-zhong-zui-hou-sheng-xia-de-shu-zi-lcof/)

社团共有 `num` 位成员参与破冰游戏，编号为 `0 ~ num-1`。成员们按照编号顺序围绕圆桌而坐。社长抽取一个数字 `target`，从 0 号成员起开始计数，排在第 `target` 位的成员离开圆桌，且成员离开后从下一个成员开始计数。请返回游戏结束时最后一位成员的编号。

**示例 1：**

**输入：**num = 7, target = 4
**输出：**1

**示例 2：**

**输入：**num = 12, target = 5
**输出：**0

**提示：**

- `1 <= num <= 10^5`
- `1 <= target <= 10^6`
### 解题思路
- 利用链表进行模拟
- 注意此题的解法主要用于练习链表的使用，模拟法需要循环删除 n−1 轮，每轮在链表中寻找删除节点需要 m 次访问操作（链表线性遍历），因此总体时间复杂度为 O(nm)，此时间复杂度是不可接受的。
### 代码
```cpp
#include <iostream>
// 定义链表节点
struct ListNode
{
    int val;
    ListNode *next;
    ListNode(int x) : val(x), next(nullptr) {}
};
// 解题函数
int iceBreakingGame(int num, int target)
{
    // 如果只有一个人，返回其编号0
    if (num == 1)
        return 0;

    // 创建一个环形链表
    ListNode *head = new ListNode(0); // 创建头节点
    ListNode *prev = head;            // 创建游标节点prev
    // 遍历人数生成链表，从1开始，因为头节点已生成完毕
    for (int i = 1; i < num; ++i)
    {
        prev->next = new ListNode(i); // 游标节点的地址域指向新生成的节点
        prev = prev->next;            // 更新游标节点的到下个节点
    }
    prev->next = head; // 尾节点指向头节点，形成环

    // 开始消除数字
    ListNode *curr = head; // 定义游标节点指向头节点
    // 遍历链表，直到只剩最后一个节点，退出循环
    while (curr->next != curr)
    {
        // 此时，prev指向尾节点，curr指向头节点
        // 移动双指针到第target个人（举例：从1开始移动6次找到第7个人）
        for (int i = 1; i < target; ++i)
        {
            prev = curr;
            curr = curr->next;
        }
        // 删除 curr 节点
        prev->next = curr->next;
        delete curr;
        curr = prev->next;
    }
    int result = curr->val;
    delete curr; // 删除最后一个节点
    return result;
}

int main()
{
    int n = 5; // 示例值，可以修改
    int m = 3; // 示例值，可以修改
    std::cout << iceBreakingGame(n, m) << std::endl;
    return 0;
}
```
