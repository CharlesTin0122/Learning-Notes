import maya.api.OpenMaya as om


def maya_useNewAPI():
    """告诉maya调用的是maya Python API 2.0"""
    pass


class HelloWorldCmd(om.MPxCommand):
    """插件类"""

    COMMAND_NAME = "HelloWorld"  # 命令名称常量

    def __init__(self):
        """构造函数"""
        super().__init__()

    def doIt(self, args):
        """插件执行逻辑"""
        print("Hello World")

    @classmethod
    def creator(cls):
        """返回类的实例"""
        return cls()


def initializePlugin(plugin):
    """加载插件并注册命令"""
    vendor = "Charles Tian"
    version = "1.0.0"
    plugin_fn = om.MFnPlugin(plugin, vendor, version)
    try:
        plugin_fn.registerCommand(HelloWorldCmd.COMMAND_NAME, HelloWorldCmd.creator)
    except Exception as e:
        om.MGlobal.displayError(
            f"Failed to register command {HelloWorldCmd.COMMAND_NAME}:{e}"
        )


def uninitializePlugin(plugin):
    """卸载插件，并取消注册命令"""
    plugin_fn = om.MFnPlugin(plugin)
    try:
        plugin_fn.deregisterCommand(HelloWorldCmd.COMMAND_NAME)
    except Exception as e:
        om.MGlobal.displayError(
            f"Failed to deregister command {HelloWorldCmd.COMMAND_NAME}:{e}"
        )
