# 概述
- 参见[11.0-MAnimCurve](基础知识/11.0-MAnimCurve.md)和[11.1-MFnAnimCurve](基础知识/11.1-MFnAnimCurve.md)
# 案例
```python
import traceback
import maya.api.OpenMaya as om
import maya.api.OpenMayaAnim as oma


def create_anim_curve(obj: om.MObject, attr_name: str) -> oma.MFnAnimCurve:
    """
    根据对象和属性名称创建动画曲线.

    Args:
        obj (om.MObject): 要创建动画曲线的对象.
        attr_name (str): 要创建动画曲线的属性.

    Returns:
        oma.MFnAnimCurve: 创建的动画曲线函数集.
    """
    plug = find_plug(obj, attr_name)
    if not plug or plug.isConnected:
        return None

    anim_curve_fn = oma.MFnAnimCurve()
    try:
        # 创建动画曲线，未指定曲线类型，由maya决定其类型
        anim_curve_fn.create(plug, oma.MFnAnimCurve.kAnimCurveUnknown)
    except Exception:
        traceback.print_exc()
        return None

    return anim_curve_fn


def get_anim_curve_fn(obj: om.MObject, attr_name: str) -> oma.MFnAnimCurve:
    """
    根据对象和属性名称获取其对应动画曲线的函数集

    Args:
        obj (om.MObject): 要查找函数集的对象.
        attr_name (str): 要查找函数集的属性名称.

    Returns:
        oma.MFnAnimCurve or None: 如果找到返回函数集，如果没有找到返回None
    """
    plug = find_plug(obj, attr_name)  # 找到插头
    if not plug:
        return None
    source_plug = plug.source()  # 获取插头的源插头
    source_node = source_plug.node()  # 获取 MPlug 的 MObject
    anim_curve_fn = oma.MFnAnimCurve()  # 创建动画曲线函数集
    # 如果对象和函数集对应，则设置函数集对象然后返回函数集
    if anim_curve_fn.hasObj(source_node):
        anim_curve_fn.setObject(source_node)
        return anim_curve_fn
    return None


def find_plug(obj: om.MObject, attr_name: str) -> om.MPlug:
    """
    根据对象和属性名称找到插头
    Args:
        obj (om.MObject): 要查找插头的对象.
        attr_name (str): 属性的名称.
    Returns:
        om.MPlug: 如果找到返回插头，如果没有找到返回None
    """
    depend_fn = om.MFnDependencyNode(obj)  # 获取对象的DependNode函数集
    try:
        plug = depend_fn.findPlug(attr_name, False)  # 查找插头
    except Exception:
        om.MGlobal.displayWarning(
            f"Attribute not found: {depend_fn.name()}.{attr_name}"
        )
        return None
    return plug  # 返回插头


if __name__ == "__main__":
    # 获取选择列表
    selection = om.MGlobal.getActiveSelectionList()
    # 获取选择列表第一项
    if selection.length() > 0:
        obj = selection.getDependNode(0)
        # 获取动画曲线函数集
        anim_curve_fn = get_anim_curve_fn(obj, "translateX")
        # 如果没有则创建动画曲线
        if not anim_curve_fn:
            anim_curve_fn = create_anim_curve(obj, "translateX")
        # 打印动画曲线名称
        if anim_curve_fn:
            print(anim_curve_fn.name())
        else:
            print("An anim curve is not connected")
    else:
        om.MGlobal.displayError("An object is not selected")

```
