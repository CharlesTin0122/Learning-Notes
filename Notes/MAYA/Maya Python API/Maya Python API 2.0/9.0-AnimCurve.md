## 基础功能应用
```python
import traceback
import maya.api.OpenMaya as om
import maya.api.OpenMayaAnim as oma


def create_anim_curve(obj: om.MObject, attr_name: str) -> oma.MFnAnimCurve:
    """
    根据对象和属性名称创建动画曲线.

    Args:
        obj (om.MObject): 要创建动画曲线的对象.
        attr_name (str): 要创建动画曲线的属性.

    Returns:
        oma.MFnAnimCurve: 创建的动画曲线函数集.
    """
    plug = find_plug(obj, attr_name)
    if not plug or plug.isConnected:
        return None

    anim_curve_fn = oma.MFnAnimCurve()
    try:
        # 创建动画曲线，未指定曲线类型，由maya决定其类型
        anim_curve_fn.create(plug, oma.MFnAnimCurve.kAnimCurveUnknown)
    except Exception:
        traceback.print_exc()
        return None

    return anim_curve_fn


def insert_key(obj: om.MObject, attr_name: str) -> int | None:
    """为动画曲线插入帧"""
    current_time = get_current_time()  # 获取当前时间栏时间
    anim_curve_fn = get_anim_curve_fn(obj, attr_name)  # 获取动画曲线函数集
    # 如果未获取到函数集则创建一条曲线，如果曲线创建失败返回None
    if not anim_curve_fn:
        anim_curve_fn = create_anim_curve(obj, attr_name)
        if not anim_curve_fn:
            return None
    # 获取当前时间动画曲线上的帧索引
    index = anim_curve_fn.find(current_time)
    # 如果当前时间没有帧，则插入帧（保持动画曲线形状）
    # 不能使用if not index,因为0也可能是帧索引
    if index is None:
        # insertKey(time, breakdown=False, change=None) ->int
        # 插入帧并保持曲线形状
        return anim_curve_fn.insertKey(current_time)
    # 其余情况 返回None
    return None


def remove_key(obj: om.MObject, attr_name: str) -> None:
    """移除当前时间关键帧"""
    anim_curve_fn = get_anim_curve_fn(obj, attr_name)
    if not anim_curve_fn:
        return
    index = anim_curve_fn.find(get_current_time())
    # 不能使用 if index:,因为0也可能是帧索引
    if not index is None:
        anim_curve_fn.remove(index)


def set_key_value(anim_curve_fn: oma.MFnAnimCurve, index: int, value: float) -> None:
    """设置关键帧的值"""
    anim_curve_fn.setValue(index, value)


def set_key_time(anim_curve_fn: oma.MFnAnimCurve, index: int, time: om.MTime) -> None:
    """设置关键帧的时间, setInput而不是setTime
    注意：设置帧时间不能超过前后帧的范围，否则会导致index失效
    如果确实要修改时间超过前后帧的范围，可以先删除帧，然后在新位置重新插入
    """
    anim_curve_fn.setInput(index, time)


def display_anim_curve_info(anim_curve_fn: oma.MFnAnimCurve) -> None:
    """打印动画曲线信息"""
    print("动画曲线信息:")
    print(f"动画曲线类型: {anim_curve_fn.animCurveType}")
    print(f"帧总数: {anim_curve_fn.numKeys}")
    print(f"是否静态曲线：{anim_curve_fn.isStatic}")
    print(f"前无限类型： {anim_curve_fn.preInfinityType}")
    print(f"后无限类型： {anim_curve_fn.postInfinityType}")


def display_key_info(anim_curve_fn: oma.MFnAnimCurve, index: int) -> None:
    """打印帧信息"""
    print(f"frame index：{index}")
    print(f"frame time：{anim_curve_fn.input(index).value}")
    print(f"frame value：{anim_curve_fn.value(index)}")
    print(f"breakdown：{anim_curve_fn.isBreakdown(index)}")
    print(f"tangents_locked：{anim_curve_fn.tangentsLocked(index)}")
    print(f"inTangents_type：{anim_curve_fn.inTangentType(index)}")
    print(f"inTangebts_value：{anim_curve_fn.getTangentXY(index, True)}")
    print(f"outTangents_type：{anim_curve_fn.outTangentType(index)}")
    print(f"outTangebts_value：{anim_curve_fn.getTangentXY(index, False)}")


"""-------------------------------------辅助函数-----------------------------------------"""


def get_anim_curve_fn(obj: om.MObject, attr_name: str) -> oma.MFnAnimCurve:
    """
    根据对象和属性名称获取其对应动画曲线的函数集

    Args:
        obj (om.MObject): 要查找函数集的对象.
        attr_name (str): 要查找函数集的属性名称.

    Returns:
        oma.MFnAnimCurve or None: 如果找到返回函数集，如果没有找到返回None
    """
    plug = find_plug(obj, attr_name)  # 找到插头
    if not plug:
        return None
    source_plug = plug.source()  # 获取插头的源插头
    source_node = source_plug.node()  # 获取 MPlug 的 MObject
    anim_curve_fn = oma.MFnAnimCurve()  # 创建动画曲线函数集
    # 如果对象和函数集对应，则设置函数集对象然后返回函数集
    if anim_curve_fn.hasObj(source_node):
        anim_curve_fn.setObject(source_node)
        return anim_curve_fn
    return None


def get_current_time() -> om.MTime:
    """获取当前时间"""
    return oma.MAnimControl.currentTime()


def find_plug(obj: om.MObject, attr_name: str) -> om.MPlug:
    """
    根据对象和属性名称找到插头
    Args:
        obj (om.MObject): 要查找插头的对象.
        attr_name (str): 属性的名称.
    Returns:
        om.MPlug: 如果找到返回插头，如果没有找到返回None
    """
    depend_fn = om.MFnDependencyNode(obj)  # 获取对象的DependNode函数集
    try:
        plug = depend_fn.findPlug(attr_name, False)  # 查找插头
    except Exception:
        om.MGlobal.displayWarning(
            f"Attribute not found: {depend_fn.name()}.{attr_name}"
        )
        return None
    return plug  # 返回插头


if __name__ == "__main__":
    # 获取选择列表
    selection = om.MGlobal.getActiveSelectionList()
    # 获取选择列表第一项
    if selection.length() > 0:
        obj = selection.getDependNode(0)
        anim_curve_fn = get_anim_curve_fn(obj, "translateZ")
        if anim_curve_fn:
            for i in range(anim_curve_fn.numKeys):
                display_key_info(anim_curve_fn, i)

    else:
        om.MGlobal.displayError("An object is not selected")
```

## 动画曲线修改命令
```python
import maya.api.OpenMaya as om
import maya.api.OpenMayaAnim as oma
import pymel.core as pm


def maya_useNewAPI():
    """告诉maya调用的是maya Python API 2.0"""
    pass


class AnimCurveCmd(om.MPxCommand):
    COMMAND_NAME = "animCurveCmd"

    CREATE_CURVE_FLAG = ["-cc", "-createCurve"]
    DELETE_CURVE_FLAG = ["-dc", "-deleteCurve"]
    SET_KEY_FLAG = ["-set", "-setKey"]
    REMOVE_KEY_FLAG = ["-rem", "-removeKey"]

    def __init__(self):
        super().__init__()

    def doIt(self, arg_list):
        # 解析参数列表
        try:
            arg_db = om.MArgDatabase(self.syntax(), arg_list)
        except Exception:
            self.displayError("Invalid arguments")
            raise
        # 获取参数列表中的对象列表
        selection_list = arg_db.getObjectList()
        # 获取列表第一项的 MObject
        self.selected_obj = selection_list.getDependNode(0)
        # 获取对象的类型不是kTransform，报错
        if self.selected_obj.apiType() != om.MFn.kTransform:
            raise RuntimeError("Selected object is not a transform node")

        raise RuntimeError("A valid flag is required")

    def isUndoable(self):
        return True

    def undoIt(self):
        pass

    def redoIt(self):
        pass

    def create_anim_curve(self, obj: om.MObject, attr: str) -> bool:
        """是否创建动画曲线"""
        return False

    @classmethod
    def create_syntax(cls):
        syntax = om.MSyntax()

        syntax.addFlag(*cls.CREATE_CURVE_FLAG)
        syntax.addFlag(*cls.DELETE_CURVE_FLAG)
        syntax.addFlag(*cls.SET_KEY_FLAG)
        syntax.addFlag(*cls.REMOVE_KEY_FLAG)
        # 选择列表作为参数，可使用场景内的对象名称作为参数
        # 参数1：选择列表，参数2：场景内选择的最小数量，参数3：场景内选择的最大数量
        syntax.setObjectType(om.MSyntax.kSelectionList, 1, 1)
        # 使用maya场景内的选择列表作为默认参数
        syntax.useSelectionAsDefault(True)

        return syntax

    @classmethod
    def creator(cls):
        return cls()


def initializePlugin(plugin):
    """加载插件并注册命令"""
    vendor = "Charles Tian"
    version = "1.0.0"

    plugin_fn = om.MFnPlugin(plugin, vendor, version)
    try:
        plugin_fn.registerCommand(
            AnimCurveCmd.COMMAND_NAME, AnimCurveCmd.creator, AnimCurveCmd.create_syntax
        )
    except Exception as e:
        om.MGlobal.displayError(
            f"Failed to register command: {AnimCurveCmd.COMMAND_NAME}, {e}"
        )


def uninitializePlugin(plugin):
    """卸载插件，并取消注册命令"""
    plugin_fn = om.MFnPlugin(plugin)
    try:
        plugin_fn.deregisterCommand(AnimCurveCmd.COMMAND_NAME)
    except Exception as e:
        om.MGlobal.displayError(
            f"Failed to deregister command: {AnimCurveCmd.COMMAND_NAME}, {e}"
        )


if __name__ == "__main__":
    pass
```
