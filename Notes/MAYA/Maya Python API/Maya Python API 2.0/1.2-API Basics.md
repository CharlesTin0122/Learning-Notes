# 获取选择对象类型以及对应函数集
```python
import maya.api.OpenMaya as om

# 获取场景内的选择列表,返回：MSelectionList Object
selection = om.MGlobal.getActiveSelectionList()
# 遍历选择列表
for i in range(selection.length()):
    if i > 0:
        print("--------")
    # MSelectionList Object的 getDependNode 方法返回 对应索引的 MObject
    obj = selection.getDependNode(i)
    # 输出类型名称： kMesh， kJoint， kTransform
    print(f"API Type: {obj.apiTypeStr}")
    # 如果当前对象有MFnDependencyNode函数集，则调用打印对象名称
    if obj.hasFn(om.MFn.kDependencyNode):
        depend_fn = om.MFnDependencyNode(obj)
        print(f"Dependency Node: {depend_fn.name()}")
        # 如果当前对象有MFnTransform函数集，则输出其位置
        if obj.hasFn(om.MFn.kTransform):
            transform_fn = om.MFnTransform(obj)
            print(f"Translation：{transform_fn.translation(om.MSpace.kTransform)}")
        # 如果当前对象有MFnMesh函数集，则输出其顶点位置
        elif obj.hasFn(om.MFn.kMesh):
            mesh_fn = om.MFnMesh(obj)
            print(f"Mesh Vertices: {mesh_fn.getVertices()}")
        # 如果当前对象类型为kJoint，则使用MFnDagNode输出其名称（API 2.0 中没有 MFnJoint 类）
        elif obj.hasFn(om.MFn.kJoint):
            joint_fn = om.MFnDagNode(obj)
            print(f"Joint Name:{joint_fn.name()}")

```
# 获取和设置属性
## 获取单个数值
```python
import maya.api.OpenMaya as om

# 定义节点名称和属性名称
node_name = "pCube1"
attribute_name = "translateY"
# 通过选择列表获取场景对象
selection_list = om.MSelectionList()
selection_list.add(node_name)
# 通过选择列表获取对象的MObject
obj = selection_list.getDependNode(0)
# 如果对象有MFnTransform函数集，则实例化函数集
if obj.hasFn(om.MFn.kTransform):
    transform_fn = om.MFnTransform(obj)
    # 通过函数集获取非网络化的 MPlug，通常用于直接访问属性值
    plug = transform_fn.findPlug(attribute_name, False)
    # 查询插头数值
    attr_value = plug.asDouble()
    # 打印数值
    print(f"{plug}: {attr_value}")
    # 设置属性值
    plug.setDouble(3.14)

```
## 获取组合数值

```python
import maya.api.OpenMaya as om

# 定义节点名称和属性名称
node_name = "pCube1"
attribute_name = "translate"
# 通过选择列表获取场景对象
selection_list = om.MSelectionList()
selection_list.add(node_name)
# 通过选择列表获取对象的MObject
obj = selection_list.getDependNode(0)
# 如果对象有MFnTransform函数集，则实例化函数集
if obj.hasFn(om.MFn.kTransform):
    transform_fn = om.MFnTransform(obj)
    # 通过函数集获取非网络化的 MPlug，通常用于直接访问属性值
    plug = transform_fn.findPlug(attribute_name, False)
    # 查询插头是否为复合属性（x,y,z）
    if plug.isCompound:
        # 遍历复合属性
        for i in range(plug.numChildren()):
            # 获取复合属性子属性
            child_plug = plug.child(i)
            # 查询子属性数值
            attr_value = child_plug.asDouble()
            # 打印数值
            print(f"{child_plug}: {attr_value}")

```

## 使用函数集方法查询和设置属性
```python
import maya.api.OpenMaya as om

# 定义节点名称和属性名称
node_name = "pCube1"
attribute_name = "translate"
# 通过选择列表获取场景对象
selection_list = om.MSelectionList()
selection_list.add(node_name)
# 通过选择列表获取对象的MObject
obj = selection_list.getDependNode(0)
# 如果对象有MFnTransform函数集，则实例化函数集
if obj.hasFn(om.MFn.kTransform):
    transform_fn = om.MFnTransform(obj)
    # 通过函数集获取属性
    translation = transform_fn.translation(om.MSpace.kTransform)
    # 打印属性
    print(translation)
    # 设置属性
    translation[1] = 3.0
    transform_fn.setTranslation(translation, om.MSpace.kTransform)

```
