import maya.api.OpenMaya as om
import maya.api.OpenMayaUI as omui


def maya_useNewAPI():
    """告诉maya调用的是maya Python API 2.0"""
    pass


class HelloWorldNode(omui.MPxLocatorNode):
    """插件类"""

    TYPE_NAME = "HelloWorld"  # 命令名称常量
    TYPE_ID = om.MTypeId(0x0007F7F7)

    def __init__(self):
        """构造函数"""
        super().__init__()

    @classmethod
    def creator(cls):
        """返回类的实例"""
        return cls()

    @classmethod
    def initialize(cls):
        """用于初始化节点属性"""
        pass


def initializePlugin(plugin):
    """加载插件并注册命令"""
    vendor = "Charles Tian"
    version = "1.0.0"
    plugin_fn = om.MFnPlugin(plugin, vendor, version)
    try:
        plugin_fn.registerNode(
            HelloWorldNode.TYPE_NAME,
            HelloWorldNode.TYPE_ID,
            HelloWorldNode.creator,
            HelloWorldNode.initialize,
            om.MPxNode.kLocatorNode,
        )
    except Exception as e:
        om.MGlobal.displayError(
            f"Failed to register Node {HelloWorldNode.TYPE_NAME}:{e}"
        )


def uninitializePlugin(plugin):
    """卸载插件，并取消注册命令"""
    plugin_fn = om.MFnPlugin(plugin)
    try:
        plugin_fn.deregisterNode(HelloWorldNode.TYPE_ID)
    except Exception as e:
        om.MGlobal.displayError(
            f"Failed to deregister Node {HelloWorldNode.TYPE_NAME}:{e}"
        )
