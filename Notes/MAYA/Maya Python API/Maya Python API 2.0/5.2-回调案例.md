# 案例一
- 用到的类和函数
	-  om.MEventMessage.addEventCallback：用于事件回调：新场景打开，时间栏改变，选择对象改变
	- om.MSceneMessage.addCallback：用于场景回调：导入前回调，导入后回调
	- omui.MUiMessage.addCameraChangedCallback：用于UI回调：摄影机改变
	- om.MConditionMessage.addConditionCallback：用于条件回调：播放状态的真与假
	- om.MTimerMessage.addTimerCallback：用于时间回调：每隔多长时间调用回调函数
```python
import maya.api.OpenMaya as om
import maya.api.OpenMayaAnim as oma  # 用于获得当前帧数
import maya.api.OpenMayaUI as omui


def maya_useNewAPI():
    """告诉maya调用的是maya Python API 2.0"""
    pass


# 储存回调ID，用于卸载插件时移除回调
callback_id_list: list = []


# 打开场景时的回调函数
def on_new_scene(client_data):
    print("New scene opened")


# 导入前回调函数
def before_import(client_data):
    print("Import pre-prosessing")


# 导入后回调函数
def after_import(client_data):
    print("Import post-processing")


# 当时间栏时间改变回调函数
def on_time_changed(client_data):
    """获取时间栏当前帧
    * MAnimControl 是 OpenMayaAnim 模块中的一个类，用于控制 Maya 的动画播放和时间轴。
    * MAnimControl.currentTime() 是一个方法，返回当前时间轴上的时间点，返回值是一个 MTime 对象。
    * asUnits() 是 MTime 对象的方法，用于将时间值转换为指定单位。
    * om.MTime.uiUnit() 获取 Maya 用户界面中设置的时间单位（通常由用户在 Maya 的偏好设置中定义，例如帧、秒、分钟等）。
    """
    frame = oma.MAnimControl.currentTime().asUnits(om.MTime.uiUnit())
    print(f"Time changed: {frame}")


# 当选择对象改变
def on_selection_changed(lient_data):
    print("Selection changed")


# 当摄影机改变
def on_camera_changed(model_panel: str, *args):
    print(f"Camera: {model_panel} changed")


# 当播放状态改变
def on_playing_state_changed(is_playing, client_data):
    print(f"Playing state changed:{is_playing}")


# 每隔多长时间触发,参数：已用时间， 触发间隔时间，用户数据
def on_timer_fire(elapsed_time, pre_exec_time, client_data):
    print("Timer fired")


# 加载插件
def initializePlugin(plugin):
    # global关键字用于在函数内部声明一个变量为全局变量，以便在函数中访问或修改全局作用域中的变量。
    global callback_id_list
    """加载插件并注册命令"""
    vendor = "Charles Tian"
    version = "1.0.0"
    om.MFnPlugin(plugin, vendor, version)
    # 添加打开新场景回调
    callback_id_list.append(
        om.MEventMessage.addEventCallback("NewSceneOpened", on_new_scene)
    )
    # 添加时间栏改变回调，注意："timeChanged"首字母小写
    callback_id_list.append(
        om.MEventMessage.addEventCallback("timeChanged", on_time_changed)
    )
    # 添加选择对象改变回调
    callback_id_list.append(
        om.MEventMessage.addEventCallback("SelectionChanged", on_selection_changed)
    )

    # 添加导入前处理回调
    callback_id_list.append(
        om.MSceneMessage.addCallback(om.MSceneMessage.kBeforeImport, before_import)
    )
    # 添加导入后处理回调
    callback_id_list.append(
        om.MSceneMessage.addCallback(om.MSceneMessage.kAfterImport, after_import)
    )
    # 添加modelPanel改变回调，modelPanel是MayaUI中的一个控件Widget，用于渲染和显示 3D 场景内容。
    # modelPanel5是当前persp摄像机所绑定的modelPane
    callback_id_list.append(
        omui.MUiMessage.addCameraChangedCallback("modelPanel5", on_camera_changed)
    )
    # 添加播放状态改变回调
    callback_id_list.append(
        om.MConditionMessage.addConditionCallback(
            "palyingBack", on_playing_state_changed
        )
    )
    # 添加间隔时间触发回调,每隔2.5秒触发
    callback_id_list.append(om.MTimerMessage.addTimerCallback(2.5, on_timer_fire))


# 卸载插件
def uninitializePlugin(plugin):
    """卸载插件，并取消注册命令"""
    # 移除所有回调
    global callback_id_list
    om.MMessage.removeCallbacks(callback_id_list)
    callback_id_list = []
```
