# 概述

编写自定义节点过程
	- node class
		- nitialize node function：初始化节点函数(`__init__`)，初始化所有节点属性
			- 使用`MFnAttribute`进行初始化
			- ![](Notes/MAYA/Maya%20Python%20API/attachments/90a2c4a9eeebb96c693be3d990bae324_MD5.jpeg)
			- 这里我们使用`MFnNumericAttribute`来创建数学属性，将返回属性的mObject
			- 属性必须被定义为类属性，而不是实例属性（在__init__函数内），是因为：
				- 在 Maya 的 Dependency Graph 节点中，属性（Attributes）是节点的静态定义，描述了节点的结构和数据接口。
				- initialize方法是静态方法无法调用实例属性，插件注册时调用一次，用于定义节点的属性和行为。MObject 被分配具体的属性定义（如浮点数、向量等），并且这些定义对所有节点实例是相同的。
				- 在 Dependency Graph 中，属性的连接（如 attributeAffects）是基于节点类型的，而不是单个实例。
			- 设置属性的特性
				- readable（可读）用于source（源属性），作为输出属性
				- writable（可写）用于destination（目标属性），作为输入属性
				- storable（可存）：输入属性可存，输出属性不可存
				- keyable（可剋帧）：属性是否可做动画
				- 总结：==输入属性以上均可开启，输出属性仅开启readable==
			- 将属性附加到节点
			- 设置属性影响（attributeAffects），让maya知道哪个输入属性改变了会影响哪个输出属性（脏传播）
				- ![](Notes/MAYA/Maya%20Python%20API/attachments/a63e07131cf1000686cc6241f12aacf4_MD5.jpeg)

		- cumpute function：计算函数，相当于创建命令中的`doIt`函数
		- creator function：返回节点指针
	- initialize plugin：加载插件
	- uninitialize plugin：卸载插件
# 案例
## 概述
- 我们要创建一个数学节点用于计算轮子的滚动，这个节点以轮子的半径和位移作为输出，以轮子的旋转角度作为输出
- 计算公式为：
$$
\text{rotation} = \left(\frac{translation}{2 * \pi * r} \right) * -360
$$
- 即移动距离除以周长得到旋转几周，再乘以每周360度

## 代码
```python
import maya.OpenMaya as om
import maya.OpenMayaMPx as mpx
import math

# 定义节点名称和位移ID
kPluginNodeName = "WheelRotateNode"
kPluginNodeId = om.MTypeId(0x87000)  # 使用唯一的 ID，避免冲突


# 节点类
class WheelRotateNode(mpx.MPxNode):
    # 定义属性，属性定义为类属性而不是实例属性
    inRadius = om.MObject()
    inTranslate = om.MObject()
    outRotate = om.MObject()

    def __init__(self):
        # 调用父类构造函数
        mpx.MPxNode.__init__(self)

    # 计算函数。节点核心逻辑,参数：1.MPlug:属性句柄，2.MDataBlock：节点数据容器
    def compute(self, plug: om.MPlug, dataBlock: om.MDataBlock):
        """
        rotate = translate / (2 * π * radius) * -360
        """
        # 检查maya请求拉取的 plug 是否是输出属性
        if plug == WheelRotateNode.outRotate:
            # 从节点容器dataBlock中获取输入属性,返回值为om.MDataHandle
            dataHandleRadius = dataBlock.inputValue(WheelRotateNode.inRadius)
            dataHandleTranslate = dataBlock.inputValue(WheelRotateNode.inTranslate)
            # 获取输入属性值
            inRadiusVal = dataHandleRadius.asFloat()
            inTranlateVal = dataHandleTranslate.asFloat()
            # 计算输出属性,进行除零判断
            if inRadiusVal == 0:
                outRotateVal = 0.0
            else:
                outRotateVal = inTranlateVal / (2 * math.pi * inRadiusVal) * -360
            # 设置输出属性
            dataHandleRotate = dataBlock.outputValue(WheelRotateNode.outRotate)
            dataHandleRotate.setFloat(outRotateVal)
            # 将输出MPlug设置为“净”
            dataBlock.setClean(plug)
        else:
            return om.kUnknownParameter

    # 返回节点实例指针
    @staticmethod
    def nodeCreator():
        return mpx.asMPxPtr(WheelRotateNode())

    # 节点属性初始化
    @staticmethod
    def nodeInitializer():
        # 创建数字属性函数集
        mFnAttr = om.MFnNumericAttribute()

        # 创建输入半径属性，参数:长名称，短名称，类型，初始值
        WheelRotateNode.inRadius = mFnAttr.create(
            "radius", "rad", om.MFnNumericData.kFloat, 0.0
        )
        # 设置属性的特性
        mFnAttr.setReadable(True)  # 可读
        mFnAttr.setWritable(True)  # 可写
        mFnAttr.setStorable(True)  # 可存
        mFnAttr.setKeyable(True)  # 可剋

        # 创建输入位移属性，会重置mFnAttr的值，我们可以重设特性
        WheelRotateNode.inTranslate = mFnAttr.create(
            "translate", "t", om.MFnNumericData.kFloat, 0.0
        )
        # 设置属性的特性
        mFnAttr.setReadable(True)  # 可读
        mFnAttr.setWritable(True)  # 可写
        mFnAttr.setStorable(True)  # 可存
        mFnAttr.setKeyable(True)  # 可剋

        # 创建输出旋转属性,不设置默认值，由计算得出
        WheelRotateNode.outRotate = mFnAttr.create(
            "Rotate", "rot", om.MFnNumericData.kFloat
        )
        # 设置属性的特性
        mFnAttr.setReadable(True)  # 可读
        mFnAttr.setWritable(False)  # 不可写
        mFnAttr.setStorable(False)  # 不可存
        mFnAttr.setKeyable(False)  # 不可剋

        # 为节点添加属性,addAttribute方法来自父类MPxNode
        WheelRotateNode.addAttribute(WheelRotateNode.inRadius)
        WheelRotateNode.addAttribute(WheelRotateNode.inTranslate)
        WheelRotateNode.addAttribute(WheelRotateNode.outRotate)

        # 设置属性影响(attributeAffects继承自父类MPxNode)，
        # 让maya知道哪个输入属性改变了会影响哪个输出属性（脏传播）
        WheelRotateNode.attributeAffects(
            WheelRotateNode.inRadius, WheelRotateNode.outRotate
        )
        WheelRotateNode.attributeAffects(
            WheelRotateNode.inTranslate, WheelRotateNode.outRotate
        )


# 加载和卸载插件
def initializePlugin(mobject):
    """
    注册插件到 Maya
    mobject参数是maya生成的句柄用来捕获命令实例的指针
    """
    mplugin = mpx.MFnPlugin(mobject)
    try:
        # 注册节点插件
        mplugin.registerNode(
            kPluginNodeName,  # 节点名称
            kPluginNodeId,  # 节点ID
            WheelRotateNode.nodeCreator,  # 节点创建函数，返回节点实例指针
            WheelRotateNode.nodeInitializer,  # 节点属性初始化函数
        )
    except Exception:
        om.MGlobal.displayError(f"Failed to register node: {kPluginNodeName}\n")
        raise


def uninitializePlugin(mobject):
    """
    从 Maya 卸载插件
    卸载插件只需命令名称即可
    """
    mplugin = mpx.MFnPlugin(mobject)
    try:
        # 卸载插件需要NodeId
        mplugin.deregisterNode(kPluginNodeId)
    except Exception:
        om.MGlobal.displayError(f"Failed to deregister node: {kPluginNodeName}\n")
        raise

```
- 当maya检测到节点输入节点发生改变，radius或者transform属性
- 会调用compute函数，compute函数会接受outRotate属性的Mplug和节点的MDataBlock
- 进行计算：通过MDataBlock获取输入属性的值，进行计算得到结果，设置为输出属性的值，并将输出属性的MPlug设置为“净”
