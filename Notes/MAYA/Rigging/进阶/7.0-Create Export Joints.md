import pymel.core as pc

def create_export_joints():
	source_jnt = pc.ls(sl=1, type="joint")
	exp_jnt_list = []
	for jnt in source_jnt:
		pc.select(cl=1)
		exp_jnt = pc.joint(name=f"{jnt}_exp")
		pc.matchTransform(exp_jnt, jnt, position=1, rotation = 1)
		pc.makeIdentity(exp_jnt, apply=1)
		pc.parentConstraint(jnt, exp_jnt, mo = 0)
		exp_jnt_list.append(exp_jnt)
	return exp_jnt_list

create_export_joints()
